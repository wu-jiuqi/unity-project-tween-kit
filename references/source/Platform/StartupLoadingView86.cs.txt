 IEnumerator PrefetchHomeLevel020(ContentLoadOperation operation,int generation,int index){
  try{
  yield return new WaitForSecondsRealtime(1);
  if(operation.Cancelled||generation!=viewGeneration||(CurrentPage!="home"&&CurrentPage!="chapters"))yield break;
  // Prepare the current playable level first; foreground entry joins it.
  prefetchMenuPage=null;prefetchLevelIndex=index;
  yield return operation.Run(GameContent.PrefetchLevel(index));
  if(operation.Succeeded&&contentLoad==null&&generation==viewGeneration&&(CurrentPage=="home"||CurrentPage=="chapters")){
   prefetchLevelIndex=-1;prefetchMenuPage="chapters";
   yield return operation.Run(GameContent.PreparePage("chapters"));
   if(!operation.Succeeded||contentLoad!=null||generation!=viewGeneration)yield break;
   Debug.Log("BOUNCE_MAP_PREFETCH ready");
   prefetchMenuPage="result-lose";yield return operation.Run(GameContent.PreparePage("result-lose"));
   if(!operation.Succeeded||contentLoad!=null||generation!=viewGeneration)yield break;
   if(index+1<OpeningCampaign.Count){prefetchMenuPage=null;prefetchLevelIndex=index+1;yield return operation.Run(GameContent.PrefetchLevel(index+1));}
  }
  }finally{FinishPrefetch(operation);}
  if(!operation.Cancelled)Debug.Log("BOUNCE_HOME_PREFETCH "+(operation.Succeeded?"ready":"retry-on-entry")+" level="+(index+1));
 }
