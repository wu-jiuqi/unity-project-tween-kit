// Shared with the original letter animation: same frequencies, damping and mesh
// deformation. Decorations use a softer strength and a small phase offset.
public static class JellyLogoMotion {
 public struct Pose {public Vector3 Scale;public float Lift,Rotation,Deformation;}
 public static Pose Sample(float elapsed,float remaining,float strength=1,float phase=0,float fadeIn=1){
  float envelope=Mathf.Exp(-.42f*elapsed)*Mathf.Clamp01(remaining/.65f)*strength*fadeIn;
  float wave=Mathf.Sin(elapsed*11.5f+phase);
  return new Pose{Scale=new Vector3(1+wave*.17f*envelope,1-wave*.14f*envelope,1),
   Lift=Mathf.Abs(Mathf.Sin(elapsed*5.75f+phase*.5f))*21*envelope,
   Rotation=Mathf.Sin(elapsed*8+phase)*3.8f*envelope,
   Deformation=Mathf.Sin(Mathf.Floor(elapsed*30)/30*11.5f+phase)*.42f*envelope};
 }
 public static void Deform(VertexHelper mesh,Rect bounds,float deformation){
  if(Mathf.Abs(deformation)<.001f)return;UIVertex vertex=default(UIVertex);
  for(int i=0;i<mesh.currentVertCount;i++){
   mesh.PopulateUIVertex(ref vertex,i);float y=Mathf.InverseLerp(bounds.yMin,bounds.yMax,vertex.position.y);
   vertex.position.x+=(y-.5f)*bounds.width*.065f*deformation;
   vertex.position.y+=Mathf.Abs(vertex.position.x/bounds.width)*bounds.height*.04f*deformation;mesh.SetUIVertex(vertex,i);
  }
 }
}
public sealed class JellyLogoDecoration:BaseMeshEffect {
 RectTransform rect;Vector2 rest;Vector3 restScale;Quaternion restRotation;
 float started,ends,strength,phase,deformation;bool running;
 public void Play(int index,float endAt,float amplitude){
  rect=(RectTransform)transform;rest=rect.anchoredPosition;restScale=rect.localScale;restRotation=rect.localRotation;
  started=Time.unscaledTime;ends=endAt;strength=amplitude;phase=index*.36f;running=true;
 }
 void Update(){
  if(!running)return;float now=Time.unscaledTime;
  if(now>=ends){ResetPose();return;}
  float elapsed=now-started;
  var pose=JellyLogoMotion.Sample(elapsed,ends-now,strength,phase,Mathf.SmoothStep(0,1,elapsed/.14f));
  rect.localScale=Vector3.Scale(restScale,pose.Scale);rect.anchoredPosition=rest+Vector2.up*pose.Lift;
  rect.localRotation=restRotation*Quaternion.Euler(0,0,pose.Rotation);deformation=pose.Deformation;graphic.SetVerticesDirty();
 }
 void ResetPose(){
  running=false;if(rect==null)return;rect.anchoredPosition=rest;rect.localScale=restScale;rect.localRotation=restRotation;
  deformation=0;if(graphic!=null)graphic.SetVerticesDirty();
 }
 public override void ModifyMesh(VertexHelper mesh){if(IsActive()&&rect!=null)JellyLogoMotion.Deform(mesh,rect.rect,deformation);}
 protected override void OnDisable(){ResetPose();base.OnDisable();}
}
public sealed class JellyGlyph:BaseMeshEffect {
 RectTransform rect;Vector2 rest;Sequence motion;float deformation,started,jellyStarted=-1,landingPitch=1;
 CanvasGroup visibility;public bool DropStarted{get;private set;}
 public bool DropComplete{get;private set;}public bool InSustainedPhase{get;private set;}
 public void ReleaseJelly(float release){jellyStarted=release;}
 // Hand-keyed impact poses at 30 fps, layered over the position tween.
 static readonly Vector3[] Poses={new Vector3(1.32f,.65f,-7),new Vector3(1.39f,.61f,-4),new Vector3(1.26f,.73f,4),new Vector3(.87f,1.22f,6),new Vector3(.82f,1.27f,3),new Vector3(.94f,1.12f,-3),new Vector3(1.12f,.87f,-4),new Vector3(1.15f,.85f,-1),new Vector3(.95f,1.07f,2),new Vector3(.97f,1.04f,1),new Vector3(1.04f,.96f,-1),new Vector3(1.02f,.98f,0),Vector3.one};
 public void Begin(int index){landingPitch=.94f+(index%4)*.035f;started=Time.unscaledTime;rect=(RectTransform)transform;rest=rect.anchoredPosition;rect.anchoredPosition=new Vector2(rest.x+(index%2==0?-22:22),1170f-((RectTransform)rect.parent).anchoredPosition.y+rect.rect.height+80);visibility=gameObject.AddComponent<CanvasGroup>();visibility.alpha=0;rect.localRotation=Quaternion.Euler(0,0,index%2==0?-13:12);rect.localScale=new Vector3(.87f,1.14f,1);
 motion=DOTween.Sequence().SetUpdate(true);motion.AppendInterval(.12f+index*.105f);motion.AppendCallback(()=>{DropStarted=true;visibility.alpha=1;});motion.Append(DOTween.To(()=>rect.anchoredPosition,v=>rect.anchoredPosition=v,rest,.5f).SetEase(Ease.InCubic));motion.Join(rect.DOLocalRotate(Vector3.zero,.5f));motion.AppendCallback(()=>StartCoroutine(Impact()));
 }
 IEnumerator Impact(){SoundService.Play("jelly_land",landingPitch);for(int frame=0;frame<Poses.Length;frame++){var pose=Poses[frame];rect.localScale=new Vector3(pose.x,pose.y,1);rect.localRotation=Quaternion.Euler(0,0,frame==Poses.Length-1?0:pose.z);deformation=frame==Poses.Length-1?0:Mathf.Sin(frame*1.2f)*(1-frame/(float)Poses.Length);graphic.SetVerticesDirty();yield return new WaitForSecondsRealtime(1f/30f);}DropComplete=true;while(jellyStarted<0)yield return null;InSustainedPhase=true;while(Time.unscaledTime-started<JellyLogoIntro.HoldDuration){float t=Time.unscaledTime-jellyStarted;float remaining=JellyLogoIntro.HoldDuration-(Time.unscaledTime-started);var pose=JellyLogoMotion.Sample(t,remaining);rect.localScale=pose.Scale;rect.anchoredPosition=rest+Vector2.up*pose.Lift;rect.localRotation=Quaternion.Euler(0,0,pose.Rotation);deformation=pose.Deformation;graphic.SetVerticesDirty();yield return null;}rect.anchoredPosition=rest;rect.localScale=Vector3.one;rect.localRotation=Quaternion.identity;deformation=0;graphic.SetVerticesDirty();}
 public override void ModifyMesh(VertexHelper mesh){if(IsActive()&&rect!=null)JellyLogoMotion.Deform(mesh,rect.rect,deformation);}
 protected override void OnDisable(){motion?.Kill();StopAllCoroutines();base.OnDisable();}
}
