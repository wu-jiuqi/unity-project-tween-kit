using System;using System.Collections.Generic;using UnityEngine;using UnityEngine.UI;
namespace BounceGarden {
public static class GardenArt70 {
 [Serializable]class Data{public Entry[] entries;}[Serializable]class Entry{public string key;public int[] rect;public float width,height,cx,cy,diameter;}
 static readonly Dictionary<string,Sprite> sprites=new Dictionary<string,Sprite>();static Dictionary<string,Entry> entries;
 static Entry Geometry(string key){if(entries==null){entries=new Dictionary<string,Entry>();foreach(var e in JsonUtility.FromJson<Data>(Resources.Load<TextAsset>("Art/GardenV70/geometry").text).entries)entries[e.key]=e;}return entries[key];}
 public static string OrbKey(StreakBooster k)=>k==StreakBooster.Pierce?"pierce":k==StreakBooster.Bomb?"bomb":"super";
 public static Sprite Get(string key){if(sprites.TryGetValue(key,out var s))return s;var t=Resources.Load<Texture2D>("Art/GardenV70/"+key);if(t==null)throw new Exception("Missing GardenV70 "+key);var e=Geometry(key);var r=e.rect;float sx=t.width/e.width,sy=t.height/e.height,h=(r[3]-r[1])*sy;return sprites[key]=Sprite.Create(t,new Rect(r[0]*sx,t.height-r[3]*sy,(r[2]-r[0])*sx,h),Vector2.one*.5f,100,0,SpriteMeshType.FullRect,key=="garden-button"?new Vector4(h*.5f,h*.48f,h*.5f,h*.48f):Vector4.zero);}
 
 public static void Orb(Image i,StreakBooster k,float diameter,float x,float y){var key=OrbKey(k);var e=Geometry(key);float scale=diameter/e.diameter;i.sprite=Get(key);i.type=UnityEngine.UI.Image.Type.Simple;i.preserveAspect=true;i.rectTransform.sizeDelta=new Vector2(e.width,e.height)*scale;i.rectTransform.anchoredPosition=new Vector2(x-(e.cx-e.width*.5f)*scale,y+(e.cy-e.height*.5f)*scale);}
 
 
 
}
}
