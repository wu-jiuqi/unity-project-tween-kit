using UnityEngine;
using UnityEngine.UI;
namespace BounceGarden {
// Pure presentation: neither the swinging stars nor their ropes have colliders.
public sealed class BattleScoreMeter:MonoBehaviour {
 GameSession game;int generation;RectTransform root;int[] thresholds;float clock;
 readonly RectTransform[] hinges=new RectTransform[3];readonly Image[] stars=new Image[3];
 readonly StarChargeView[] charges=new StarChargeView[3];bool victoryCounting;int displayScore;
 public bool ChargeSettled=>charges[0]!=null&&charges[0].Settled&&charges[1].Settled&&charges[2].Settled;
 public StarChargeView Charge(int i)=>charges[i];public int DisplayedScore=>victoryCounting?displayScore:game.Score;
 public void CountVictoryScore(int score){victoryCounting=true;displayScore=score;Place();}
 public RectTransform[] DetachVictoryStars(Transform destination){victoryCounting=false;var result=new RectTransform[3];for(int i=0;i<3;i++){result[i]=stars[i].rectTransform;result[i].SetParent(destination,true);hinges[i].gameObject.SetActive(false);}return result;}
 readonly float[] angle=new float[3],velocity=new float[3],lastBrick={-10,-10,-10};
 readonly float[] xs={-2.85f,0,2.85f};
 public int BallResponses{get;private set;}public int BrickResponses{get;private set;}
 public float Swing(int i)=>angle[i];public bool HasNumericLabels=>GetComponentsInChildren<TMPro.TMP_Text>().Length>0;
 public void Initialize(GameSession g,int best){game=g;generation=g.Generation;root=(RectTransform)transform;root.anchoredPosition=Vector2.zero;root.sizeDelta=new Vector2(1080,2340);
  int goals=g.IsOpening?System.Array.ConvertAll(g.OpeningTargets,t=>t.count).SumCounts():6;int unit=Mathf.Max(1000,Mathf.CeilToInt(goals*600f/1000)*1000);thresholds=new[]{unit,unit*2,unit*3};
  for(int i=0;i<3;i++){hinges[i]=UICraft.Root(root,"Vine pendant "+i,0,0,2,2);BattleJellyArt.Image(hinges[i],"vine-star-junction",0,0,24,16);BattleJellyArt.Image(hinges[i],"vine-star-stem",0,-36,18,72);stars[i]=BattleJellyArt.Image(hinges[i],"vine-star-off",0,-102,80,80);var edge=stars[i].gameObject.AddComponent<Shadow>();edge.effectColor=new Color(.24f,.38f,.14f,.30f);edge.effectDistance=new Vector2(1.8f,-2.5f);}
  for(int i=0;i<3;i++){charges[i]=stars[i].gameObject.AddComponent<StarChargeView>();int lower=i==0?0:thresholds[i-1];charges[i].Initialize(root,Mathf.Clamp01((g.Score-lower)/(float)(thresholds[i]-lower)));}game.VineBallContact+=OnBall;game.VineFallingContact+=OnFalling;Place();
 }
 Vector2 Local(Vector2 world){var canvas=GetComponentInParent<Canvas>();var screen=Camera.main.WorldToScreenPoint(BoardDisplay.Forward(world));RectTransformUtility.ScreenPointToLocalPointInRectangle(root,screen,canvas.renderMode==RenderMode.ScreenSpaceOverlay?null:canvas.worldCamera,out var p);return p;}
 void Place(){if(Camera.main==null)return;for(int i=0;i<3;i++){float dent=game.JellyWalls!=null?game.JellyWalls.Deformation(new Vector2(xs[i],JellyWallFrame.TopCenter),1):0;hinges[i].anchoredPosition=Local(new Vector2(xs[i],JellyWallFrame.TopCenter+dent));hinges[i].localRotation=Quaternion.Euler(0,0,angle[i]);if(charges[i]!=null){int lower=i==0?0:thresholds[i-1];var summary=GetComponentInParent<AppUI>()?.GetComponentInChildren<BattleHUDSummary>();Vector3 source=summary!=null?summary.ScoreLabel.transform.position:hinges[i].position;charges[i].Charge(Mathf.Clamp01((DisplayedScore-lower)/(float)(thresholds[i]-lower)),source);}}}
 bool Allowed=>game!=null&&game.Generation==generation&&game.IsPlaying&&!game.IsPaused;
 void Impulse(int i,float amount){velocity[i]=Mathf.Clamp(velocity[i]+amount,-240,240);}
 void OnBall(Vector2 point,float speed){if(!Allowed)return;BallResponses++;for(int i=0;i<3;i++){float distance=Mathf.Abs(point.x-xs[i]);float direction=point.x>=xs[i]?1:-1;float influence=.48f+.52f*Mathf.Exp(-distance*.5f);Impulse(i,direction*Mathf.Clamp(speed*16,75,190)*influence);}}
 void OnFalling(Vector2 from,Vector2 to,float size){if(!Allowed||Camera.main==null)return;var a=Local(from);var b=Local(to);float half=41*Mathf.Max(1,size);for(int i=0;i<3;i++){Vector2 p=root.InverseTransformPoint(stars[i].rectTransform.position);if(clock-lastBrick[i]<.16f||Mathf.Abs(a.x-p.x)>half+40||a.y+half+40<p.y||b.y-half-40>p.y)continue;lastBrick[i]=clock;BrickResponses++;Impulse(i,(a.x>=p.x?1:-1)*80);}}
 public void Advance(float dt){if(!Allowed)return;dt=Mathf.Clamp(dt,0,.05f);clock+=dt;for(int i=0;i<3;i++){if(!FeedbackProfile.Active.enabledFeedback){angle[i]=velocity[i]=0;continue;}for(int n=0;n<4;n++){float step=dt*.25f;velocity[i]+=(-angle[i]*30-velocity[i]*3.6f)*step;float next=angle[i]+velocity[i]*step;angle[i]=Mathf.Clamp(next,-36,36);if(next!=angle[i]&&Mathf.Sign(velocity[i])==Mathf.Sign(angle[i]))velocity[i]*=-.25f;}}Place();foreach(var charge in charges)if(charge!=null)charge.Advance(dt);}
 void LateUpdate(){if(victoryCounting){Place();foreach(var charge in charges)charge.Advance(game.PresentationDelta(Mathf.Min(.05f,Time.unscaledDeltaTime)));}else Advance(Time.deltaTime);}
 void OnDestroy(){if(game!=null){game.VineBallContact-=OnBall;game.VineFallingContact-=OnFalling;}}
}
}
