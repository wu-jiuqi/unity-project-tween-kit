using UnityEngine;
using UnityEngine.UI;
namespace BounceGarden {
// Pure presentation: neither the swinging stars nor their ropes have colliders.
public sealed class BattleScoreMeter:MonoBehaviour {
 GameSession game;int generation;RectTransform root;int[] thresholds;float clock;
 readonly Vector4[] ballSamples=new Vector4[32];readonly RectTransform[] anchors=new RectTransform[3];readonly RectTransform[] hinges=new RectTransform[3];readonly Image[] stars=new Image[3];
 readonly StarChargeView[] charges=new StarChargeView[3];bool victoryCounting;int displayScore;
 
 public StarChargeView Charge(int i)=>charges[i];public int DisplayedScore=>victoryCounting?displayScore:game.Score;
 public void CountVictoryScore(int score){victoryCounting=true;displayScore=score;Place();}
 public RectTransform[] DetachVictoryStars(Transform destination){victoryCounting=false;var result=new RectTransform[3];for(int i=0;i<3;i++){charges[i].PrepareVictoryTransfer();result[i]=stars[i].rectTransform;result[i].SetParent(destination,true);hinges[i].gameObject.SetActive(false);}return result;}
 readonly float[] angle=new float[3],velocity=new float[3],lastBrick={-10,-10,-10};
 readonly float[] xs={-2.15f,0,2.15f};
 public int BallResponses{get;private set;}public int BrickResponses{get;private set;}
 public float Swing(int i)=>angle[i];public float AngularSpeed(int i)=>velocity[i];public void QASetRotation(int i,float value){if(!RuntimeQA.IsIsolated)return;angle[i]=value;velocity[i]=0;Place();}
 public void Initialize(GameSession g,int best){game=g;generation=g.Generation;root=(RectTransform)transform;root.anchoredPosition=Vector2.zero;root.sizeDelta=new Vector2(1080,2340);
  int goals=g.IsOpening?System.Array.ConvertAll(g.OpeningTargets,t=>t.count).SumCounts():6;int unit=Mathf.Max(1000,Mathf.CeilToInt(goals*600f/1000)*1000);thresholds=new[]{unit,unit*2,unit*3};
  for(int i=0;i<3;i++){anchors[i]=UICraft.Root(root,"Star flower ground anchor "+i,0,0,2,2);hinges[i]=UICraft.Root(anchors[i],"Root-pivoted star flower "+i,0,0,2,2);stars[i]=BattleJellyArt.Image(hinges[i],"vine-star-off",0,0,90,82);var edge=stars[i].gameObject.AddComponent<Shadow>();edge.effectColor=new Color(.24f,.38f,.14f,.30f);edge.effectDistance=new Vector2(1.8f,-2.5f);}
  for(int i=0;i<3;i++){charges[i]=stars[i].gameObject.AddComponent<StarChargeView>();int lower=i==0?0:thresholds[i-1];charges[i].Initialize(root,Mathf.Clamp01((g.Score-lower)/(float)(thresholds[i]-lower)));}game.VineBallContact+=OnBall;game.VineFallingContact+=OnFalling;Place();
 }
 public int PassResponses{get;private set;}public int FirstThreshold=>thresholds[0];public RectTransform Flower(int i)=>anchors[i];public RectTransform FlowerHead(int i)=>stars[i].rectTransform;
 Vector2 Local(Vector2 world)=>FromDisplay(BoardDisplay.Forward(world));
 Vector2 FromDisplay(Vector2 display){var canvas=GetComponentInParent<Canvas>();var screen=Camera.main.WorldToScreenPoint(display);RectTransformUtility.ScreenPointToLocalPointInRectangle(root,screen,canvas.renderMode==RenderMode.ScreenSpaceOverlay?null:canvas.worldCamera,out var p);return p;}
 void Place(){if(Camera.main==null)return;float correction=root.lossyScale.x/Mathf.Max(.0001f,root.lossyScale.y);for(int i=0;i<3;i++){anchors[i].anchoredPosition=Local(new Vector2(xs[i],FlightGeometry.Top))-Vector2.up*(62*correction);anchors[i].localScale=new Vector3(1,correction,1);hinges[i].localRotation=Quaternion.identity;if(charges[i]!=null)charges[i].SetRotation(angle[i]);if(charges[i]!=null){int lower=i==0?0:thresholds[i-1];var summary=GetComponentInParent<AppUI>()?.GetComponentInChildren<BattleHUDSummary>();Vector3 source=summary!=null?summary.ScoreLabel.transform.position:anchors[i].position;charges[i].Charge(Mathf.Clamp01((DisplayedScore-lower)/(float)(thresholds[i]-lower)),source);}}}
 void BallPass(float dt){int count=game.CopyGrassBallSamples(ballSamples);for(int i=0;i<3;i++){Vector2 head=root.InverseTransformPoint(stars[i].transform.position);for(int j=0;j<count;j++){var b=ballSamples[j];var p=FromDisplay(new Vector2(b.x,b.y));float distance=Vector2.Distance(p,head)/100f;if(distance>=1.35f)continue;float w=1-distance/1.35f;float direction=Mathf.Abs(b.z)>.01f?Mathf.Sign(b.z):(p.x>=head.x?1:-1);float speed=new Vector2(b.z,b.w).magnitude;Impulse(i,direction*w*w*Mathf.Clamp(speed*10,60,160)*dt*8);if(w>.1f)PassResponses++;}}}
 bool Allowed=>game!=null&&game.Generation==generation&&game.IsPlaying&&!game.IsPaused;
 void Impulse(int i,float amount){velocity[i]=Mathf.Clamp(velocity[i]+amount*3.2f,-1440,1440);}
 void OnBall(Vector2 point,float speed){if(!Allowed)return;BallResponses++;for(int i=0;i<3;i++){float distance=Mathf.Abs(point.x-xs[i]);float direction=point.x>=xs[i]?1:-1;float influence=.48f+.52f*Mathf.Exp(-distance*.5f);Impulse(i,direction*Mathf.Clamp(speed*16,75,190)*influence);}}
 void OnFalling(Vector2 from,Vector2 to,float size){if(!Allowed||Camera.main==null)return;var a=Local(from);var b=Local(to);float half=41*Mathf.Max(1,size);for(int i=0;i<3;i++){Vector2 p=root.InverseTransformPoint(stars[i].rectTransform.position);if(clock-lastBrick[i]<.16f||Mathf.Abs(a.x-p.x)>half+40||a.y+half+40<p.y||b.y-half-40>p.y)continue;lastBrick[i]=clock;BrickResponses++;Impulse(i,(a.x>=p.x?1:-1)*80);}}
 public void Advance(float dt){if(!Allowed)return;dt=Mathf.Clamp(dt,0,.05f);clock+=dt;if(FeedbackProfile.Active.enabledFeedback)BallPass(dt);for(int i=0;i<3;i++){if(!FeedbackProfile.Active.enabledFeedback){angle[i]=velocity[i]=0;continue;}velocity[i]*=Mathf.Exp(-1.35f*dt);angle[i]+=velocity[i]*dt;}Place();foreach(var charge in charges)if(charge!=null)charge.Advance(dt);}
 void LateUpdate(){if(victoryCounting){Place();foreach(var charge in charges)charge.Advance(game.PresentationDelta(Mathf.Min(.05f,Time.unscaledDeltaTime)));}else Advance(Time.deltaTime);}
 void OnDestroy(){if(game!=null){game.VineBallContact-=OnBall;game.VineFallingContact-=OnFalling;}}
}
}
