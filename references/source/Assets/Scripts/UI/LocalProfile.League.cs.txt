 public bool LeagueUnlocked=>AlphabetUnlocked>=10;
 public LeagueSnapshot ReadLeague(bool offline=false){
  if(!LeagueUnlocked)return null;
  long now=LeagueClock.UtcSeconds;
  // Do not keep writing unchanged snapshots every frame; view requests once per configured interval.
  if(storageHealthy&&(offline||league.current==null||LeagueExclusiveCosmetics.NeedsRefresh(league,LeagueSettings)||now>league.current.lastObserved||(LeagueSettings.activeCompetition&&league.current.competitionVersion<LeagueEngine.ActiveCompetitionVersion)))LeagueTransaction(()=>{LeagueExclusiveCosmetics.RefreshCurrent(league,LeagueSettings);LeagueEngine.PrepareActive(league,LeagueSettings,now,Badges);LeagueEngine.Advance(league,now,Badges,offline);LeagueEngine.Open(league,LeagueSettings,now,Badges);},false);
  var p=league.current;if(p==null)return null;
  var entries=LeagueEngine.Standings(p,Badges);var player=entries.First(e=>e.player);
  return new LeagueSnapshot{config=p.config,periodId=p.id,remaining=Math.Max(0,p.endsAt-now),entries=entries,player=player,next=player.rank>1?entries[player.rank-2]:null,reward=league.rewards.FirstOrDefault(r=>r.eligible&&!r.claimed)};
 }

 void CommitCampaignLeague(int level,bool firstClear,int oldBadges){
  LastLeagueMedalGain=0;LastLeagueRankBefore=LastLeagueRankAfter=0;
  if(!LeagueUnlocked)return;
  long now=LeagueClock.UtcSeconds;
  LeagueEngine.PrepareActive(league,LeagueSettings,now,oldBadges);LeagueEngine.Advance(league,now,oldBadges,false);LeagueEngine.Open(league,LeagueSettings,now,oldBadges);
  var p=league.current;
  if(!p.config.calendarSeasons){LastLeagueRankBefore=LeagueEngine.Standings(p,oldBadges).First(e=>e.player).rank;LeagueEngine.Win(league,LeagueSettings,now,oldBadges,Badges,true);LastLeagueMedalGain=1;LastLeagueRankAfter=LeagueEngine.Standings(p,Badges).First(e=>e.player).rank;return;}
  long day=(now+8*3600)/86400;
  if(league.replayDay!=day){league.replayDay=day;league.replayLevels=new int[0];}
  bool eligible=firstClear||((league.replayLevels?.Length??0)<3&&!(league.replayLevels??new int[0]).Contains(level));
  // The permanent medal counter still grows on every win. Excluded wins shift the season baseline equally.
  if(!eligible){p.startMedals++;return;}
  if(!firstClear)league.replayLevels=(league.replayLevels??new int[0]).Concat(new[]{level}).ToArray();
  LastLeagueRankBefore=LeagueEngine.Standings(p,oldBadges).First(e=>e.player).rank;
  LeagueEngine.Win(league,LeagueSettings,now,oldBadges,Badges,true);LastLeagueMedalGain=1;LastLeagueRankAfter=LeagueEngine.Standings(p,Badges).First(e=>e.player).rank;
 }

 public bool ClaimLeagueReward(string periodId){
  var reward=league.rewards.FirstOrDefault(r=>r.periodId==periodId);if(reward==null||!reward.eligible||reward.claimed||committed.Contains(periodId+"/leagueReward")||(long)reward.coins+LeagueDuplicateCoins(reward)>int.MaxValue-Coins)return false;
  int duplicate=LeagueDuplicateCoins(reward);
  return LeagueTransaction(()=>{Coins+=reward.coins+duplicate;if(!string.IsNullOrEmpty(reward.avatarId)&&!ownedAvatars.Contains(reward.avatarId))ownedAvatars.Add(reward.avatarId);if(!string.IsNullOrEmpty(reward.frameId)&&!ownedFrames.Contains(reward.frameId))ownedFrames.Add(reward.frameId);reward.claimed=true;committed.Add(periodId+"/leagueReward");});
 }
