public static class LeagueSurface {
 // Size the visible face, not its transparent padding and baked shadow.
 public static void Fit(Image image,string key,float width,float height){var info=PastelArt.Info(key);float scale=height/info.visible[3];image.type=Image.Type.Sliced;image.preserveAspect=false;image.rectTransform.sizeDelta=new Vector2(width+(info.size[0]-info.visible[2])*scale,info.size[1]*scale);image.rectTransform.anchoredPosition=new Vector2((info.size[0]*.5f-info.visible[0]-info.visible[2]*.5f)*scale,(info.visible[1]+info.visible[3]*.5f-info.size[1]*.5f)*scale);image.pixelsPerUnitMultiplier=1/scale;}
}
public static class LeaguePortrait {
 public static RectTransform Create(Transform parent,LocalProfile profile,string resource,float x,float y,float size,bool own=false,string opponentFrame=null){
  string avatar=own?profile?.EquippedAvatarResource:resource;
  string frame=own?profile?.EquippedFrameResource:opponentFrame;
  if(own&&string.IsNullOrEmpty(frame))frame=CatCosmeticCatalog.Default.frame.resource;
  return CatCosmeticCatalog.Create(parent,avatar,frame,x,y,size);
 }
}
// Keep rects/hit targets stable; only visual children animate with a damped jelly pose.
public sealed class LeagueJelly:MonoBehaviour {
 Vector3 rest;Quaternion rotation;float time,delay;bool ready;public float duration=.65f;public int Pulses{get;private set;}
 public void Play(float wait=0){if(!ready){rest=transform.localScale;rotation=transform.localRotation;ready=true;}transform.localScale=rest;transform.localRotation=rotation;time=0;delay=wait;enabled=true;Pulses++;}
 void Update(){if(!FeedbackProfile.Active.enabledFeedback){Finish();return;}time+=Time.unscaledDeltaTime;if(time<delay)return;float t=(time-delay)/Mathf.Max(.05f,duration);if(t>=1){Finish();return;}float wave=Mathf.Sin(t*Mathf.PI*5)*Mathf.Exp(-4*t)*(1-t);transform.localScale=Vector3.Scale(rest,new Vector3(1+wave*.09f,1-wave*.12f,1));transform.localRotation=rotation*Quaternion.Euler(0,0,wave*2.2f);}
 void Finish(){if(ready){transform.localScale=rest;transform.localRotation=rotation;}enabled=false;}
 void OnDisable(){if(ready){transform.localScale=rest;transform.localRotation=rotation;}}
}

  var link=UICraft.Button(root,"",330,bottom+215,330,70,UICraft.Gold,rewards);link.name="League reward details";link.transform.Find("ButtonVisual").gameObject.SetActive(false);rewardLink=UICraft.Label(link.transform,"",0,0,330,60,29);
  pressureText=UICraft.Label(root,"",0,bottom+280,940,45,27,UICraft.Ink);
  var button=UICraft.Button(root,c.copy.play,0,bottom+110,840,156,UICraft.Gold,play,62);button.name="League play round";var face=button.transform.Find("ButtonVisual/Face").GetComponent<Image>();face.sprite=PastelArt.Get("button_yellow");face.type=Image.Type.Sliced;face.preserveAspect=false;face.color=new Color32(255,181,99,255);button.GetComponentInChildren<TMP_Text>().color=Color.white;
 }

 }
 void UpdateClock(){
  long seconds=Math.Max(0,snapshot.remaining-(long)(Time.unscaledTime-readTime));var remaining=TimeSpan.FromSeconds(seconds);
  countdown.text=(remaining.TotalDays>=1?((int)remaining.TotalDays)+"天":"")+remaining.Hours.ToString("00")+":"+remaining.Minutes.ToString("00")+":"+remaining.Seconds.ToString("00");nextClockTick=Time.unscaledTime+1;
 }
 public void FocusPlayer(){if(snapshot==null||scroll==null)return;float stride=snapshot.config.layout.rowHeight+snapshot.config.layout.rowGap;float desired=(snapshot.player.rank-.5f)*stride-viewport.rect.height*.5f;content.anchoredPosition=new Vector2(0,Mathf.Clamp(desired,0,Mathf.Max(0,content.rect.height-viewport.rect.height)));scroll.StopMovement();}

  string previousPeriod=snapshot?.periodId;snapshot=source.Read();if(snapshot==null)return;if(lastHeight<0||previousPeriod!=snapshot.periodId)Build();dirty=false;readTime=Time.unscaledTime;nextTick=readTime+snapshot.config.refreshSeconds;var c=snapshot.config;UpdateClock();podium.Bind(snapshot,profile);
  rankText.text=string.Format(c.copy.rank,snapshot.player.rank)+"  ·  "+snapshot.player.medals.ToString("N0")+"枚奖章";rewardLink.text=snapshot.reward!=null?"领取奖励 >":"前"+c.topRewardCount+"名奖励 >";
  float stride=c.layout.rowHeight+c.layout.rowGap;content.sizeDelta=new Vector2(990,snapshot.entries.Length*stride);
  var ids=new HashSet<string>(snapshot.entries.Select(e=>e.id));foreach(var id in rows.Keys.Where(id=>!ids.Contains(id)).ToArray()){rows[id].move?.Kill();rows[id].root.gameObject.SetActive(false);Destroy(rows[id].root.gameObject);rows.Remove(id);}
  foreach(var e in snapshot.entries){if(!rows.TryGetValue(e.id,out var row)){row=Create(e);rows.Add(e.id,row);row.root.anchoredPosition=new Vector2(0,-(e.rank-.5f)*stride);}
   row.name.text=e.name;row.rank.text=e.rank.ToString();row.rank.color=e.rank<=3?Color.white:UICraft.Ink;

   row.gift.gameObject.SetActive(e.rank<=3);if(e.rank<=3)row.gift.sprite=PastelArt.Get(e.rank==1?"chest_blush":e.rank==2?"chest_blue":"chest_lavender");
   string style=profile.EquippedAvatarResource+"/"+profile.EquippedFrameResource;if(e.player&&row.style!=style){row.style=style;row.portrait.gameObject.SetActive(false);Destroy(row.portrait.gameObject);row.portrait=LeaguePortrait.Create(row.visual,profile,"",-315,0,130,true);}
   Vector2 target=new Vector2(0,-(e.rank-.5f)*stride);if(row.oldRank!=e.rank){if(row.oldRank>0){row.direction.sprite=LeagueMotionArt.Get(row.oldRank>e.rank?"up":"down");row.direction.gameObject.SetActive(true);row.direction.rectTransform.DOKill();row.direction.rectTransform.anchoredPosition=new Vector2(95,-37+(row.oldRank>e.rank?18:-18));DOTween.To(()=>row.direction.rectTransform.anchoredPosition,v=>row.direction.rectTransform.anchoredPosition=v,new Vector2(95,-37),.6f).SetTarget(row.direction.rectTransform).SetUpdate(true);}
   if(e.player&&row.oldRank>e.rank&&row.oldRank>0)SoundService.Play("rank_up");row.move?.Kill();if(row.oldRank>0&&FeedbackProfile.Active.enabledFeedback)row.move=DOTween.To(()=>row.root.anchoredPosition,v=>row.root.anchoredPosition=v,target,.48f).SetEase(Ease.OutCubic).SetUpdate(true);else row.root.anchoredPosition=target;}

  }
  if(pressureText!=null)pressureText.text=LeagueEngine.Pressure(snapshot);

  // Season entry is scheduled by AppUI, independently of this page.
 }
}
