using TMPro;
using UnityEngine;
using UnityEngine.UI;
namespace BounceGarden {
// Local demonstration actors never enter the game session or grant real loot.
public sealed class LessonCollectionFlight:MonoBehaviour {
 int page;float age;Image token;TMP_Text count;readonly Image[] ingots=new Image[5];
 public void Begin(int id){page=id;
  var goal=UICraft.Root(transform,"Demonstration collection card",0,330,200,120);OnboardingArt.Draw(goal,"Empty target tag",1,3,0,0,210,130);
  var icon=SpriteFor(id);var item=UICraft.Box(goal,"Target identity",-45,-1,57,57,Color.white);item.sprite=icon;item.preserveAspect=true;item.raycastTarget=false;
  count=UICraft.Label(goal,"",35,-1,100,60,33,UICraft.Ink);
  token=UICraft.Box(transform,"Separate flying collectible",0,0,60,60,Color.white);token.sprite=icon;token.preserveAspect=true;token.raycastTarget=false;
  if(id==521)for(int i=0;i<5;i++)ingots[i]=OnboardingArt.Draw(transform,"Generated ingot "+i,6,4,(i-2)*90,-160,65,65);
 }
 public static Sprite SpriteFor(int id)=>id==511?OnboardingArt.Get(5,4):id==521?OnboardingArt.Get(6,4):id==531?OnboardingArt.Get(7,5):id==505||id==506||id==518?OnboardingArt.Get(4,5):OnboardingArt.Get(3,0);
 void Update(){age=Mathf.Repeat(age+Time.unscaledDeltaTime*(7f/2.2f),7f);float t=age%7,start=page==511?1.65f:page==521?5.15f:page==531?5.2f:page==518?5.8f:page==506?4.25f:page==505?2.25f:1.8f;
  if(page==511&&t>=3.5f)start=3.65f;
  float u=Mathf.Clamp01((t-start)/.65f);bool flying=t>=start&&t<start+.65f;token.enabled=flying;
  token.rectTransform.anchoredPosition=Vector2.Lerp(new Vector2(0,0),new Vector2(0,330),u)+Vector2.right*(Mathf.Sin(u*Mathf.PI)*70);token.rectTransform.localScale=Vector3.one*Mathf.Lerp(1,.55f,u);
  int initial=page==511?8:page==521?10:6;int collected=t>=start+.65f?(page==521?5:page==511&&t>3.5f?3:1):page==511&&t>3.5f?1:0;count.text=(initial-collected).ToString();
  if(page==521)for(int i=0;i<5;i++){ingots[i].enabled=t>=4.5f&&t<5.8f;ingots[i].rectTransform.anchoredPosition=new Vector2((i-2)*90,-160+Mathf.Sin(Mathf.Clamp01((t-4.5f)/.5f)*Mathf.PI)*45);}
 }
}
}
