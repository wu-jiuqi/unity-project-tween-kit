using System;
using UnityEngine;
using UnityEngine.UI;
namespace BounceGarden {
// Real-time celebration; reward preparation starts only after its callback.
public sealed class GoalCelebration:MonoBehaviour {
 public const float Duration=1.15f;
 AppUI app;GameSession game;Action completed;int generation;float age;
 RectTransform content,plaque,star,rays;Image dim;CanvasGroup group;
 readonly RectTransform[] sprays=new RectTransform[2];
 readonly Image[] sparks=new Image[14];readonly Vector2[] origins=new Vector2[14],velocities=new Vector2[14];
 bool suspended,focused=true,finished;Vector2Int viewport;Rect safe;int layoutRevision=-1;
 public float Elapsed=>age;
 public bool LayoutFits{get;private set;}
 public void Begin(AppUI owner,GameSession session,Action done){
  app=owner;game=session;generation=game.Generation;completed=done;
  var root=(RectTransform)transform;root.anchorMin=Vector2.zero;root.anchorMax=Vector2.one;root.sizeDelta=Vector2.zero;root.anchoredPosition=Vector2.zero;
  dim=gameObject.AddComponent<Image>();dim.color=new Color(.035f,.025f,.02f,0);dim.raycastTarget=true;
  content=UICraft.Root(transform,"Generated jelly celebration layers");group=content.gameObject.AddComponent<CanvasGroup>();group.blocksRaycasts=false;
  rays=Art("Warm golden radial light","halo",0,40,980,980);
  star=Art("Puffy candy star","star",0,190,230,230);
  plaque=Art("Blank jelly garden ribbon","ribbon",0,-50,900,390);
  var title=UICraft.Label(plaque,"目标达成！",0,5,650,110,84,UICraft.Ink);title.color=UICraft.Ink;
  var sub=UICraft.Label(content,"全部收集完成",0,-260,760,80,44,new Color(1,.96f,.84f));sub.color=new Color(1,.96f,.84f);
  sprays[0]=Art("Left confetti fan","confetti",-345,30,390,390);
  sprays[1]=Art("Right confetti fan","confetti",345,30,390,390);sprays[1].localScale=new Vector3(-1,1,1);
  for(int i=0;i<sparks.Length;i++){
   var image=UICraft.Box(content,"Independent golden star particle "+i,0,0,25+i%3*8,25+i%3*8,Color.white);
   image.sprite=GoalCelebrationArt.Get("star");image.type=Image.Type.Simple;image.preserveAspect=true;image.raycastTarget=false;sparks[i]=image;
   int side=i%2==0?-1:1;origins[i]=new Vector2(side*340,-50);velocities[i]=new Vector2(-side*(100+(i*37)%250),260+(i*67)%350);
  }
  Layout();Render();SoundService.Play("result_win_burst");
 }
 RectTransform Art(string name,string key,float x,float y,float w,float h){var image=UICraft.Box(content,name,x,y,w,h,Color.white);image.sprite=GoalCelebrationArt.Get(key);image.type=Image.Type.Simple;image.preserveAspect=true;image.raycastTarget=false;return image.rectTransform;}
 void Layout(){
  viewport=app.AlphabetViewport;safe=app.AlphabetSafeRect;layoutRevision=WeChatLayoutMetrics.Revision;var available=safe;
  if(WeChatLayoutMetrics.TryGetBlockedRect(viewport,out var blocked))available.yMax=Mathf.Min(available.yMax,blocked.yMin-12);
  var root=(RectTransform)transform;float width=available.width/Mathf.Max(1,viewport.x)*root.rect.width,height=available.height/Mathf.Max(1,viewport.y)*root.rect.height;
  float fit=Mathf.Min(width/1080,height/2340);content.localScale=Vector3.one*fit;
  content.anchoredPosition=new Vector2((available.center.x/Mathf.Max(1,viewport.x)-.5f)*root.rect.width,(available.center.y/Mathf.Max(1,viewport.y)-.5f)*root.rect.height);
  LayoutFits=fit>0&&980*fit<=width+.5f&&1120*fit<=height+.5f;
 }
 void Update(){
  if(game==null||game.Generation!=generation){completed=null;Destroy(gameObject);return;}
  if(suspended||(!focused&&!game.QAIgnoreFocusLoss))return;
  if(viewport!=app.AlphabetViewport||safe!=app.AlphabetSafeRect||layoutRevision!=WeChatLayoutMetrics.Revision||!LayoutFits)Layout();
  age+=Mathf.Min(Time.unscaledDeltaTime,.05f);Render();
  if(age<Duration||finished)return;finished=true;var next=completed;completed=null;
  gameObject.SetActive(false);Destroy(gameObject);next?.Invoke();
 }
 void Render(){
  float t=Mathf.Clamp01(age/Duration),fade=1-Mathf.SmoothStep(0,1,(age-.92f)/.23f);
  group.alpha=Mathf.Min(1,age/.10f)*fade;dim.color=new Color(.035f,.025f,.02f,.36f*Mathf.Min(1,age/.1f)*fade);
  bool moving=FeedbackProfile.Active.enabledFeedback;float motion=moving?t:0;
  float pop=moving?Mathf.LerpUnclamped(.7f,1,UICraft.OutBack(Mathf.Clamp01(age/.28f))):1;
  plaque.localScale=Vector3.one*pop;star.localScale=Vector3.one*(1+(moving?.045f*Mathf.Sin(Mathf.Clamp01(age/.4f)*Mathf.PI):0));
  rays.localRotation=Quaternion.Euler(0,0,motion*12);rays.localScale=Vector3.one*(.9f+.15f*motion);
  for(int i=0;i<sprays.Length;i++)sprays[i].anchoredPosition=new Vector2((i==0?-1:1)*(345+45*motion),30+110*motion);
  for(int i=0;i<sparks.Length;i++){sparks[i].enabled=moving;float u=Mathf.Max(0,age-.1f-(i%4)*.025f);sparks[i].rectTransform.anchoredPosition=origins[i]+velocities[i]*u+Vector2.down*(440*u*u);sparks[i].rectTransform.localRotation=Quaternion.Euler(0,0,i*27+u*140);sparks[i].color=new Color(1,1,1,Mathf.Clamp01(u/.08f)*Mathf.Clamp01((1.05f-u)/.3f));}
 }
 void OnApplicationPause(bool paused){suspended=paused;}
 void OnApplicationFocus(bool focus){focused=focus;}
 void OnDestroy(){completed=null;}
}
public static class GoalCelebrationArt {
 static readonly System.Collections.Generic.Dictionary<string,Sprite> cache=new System.Collections.Generic.Dictionary<string,Sprite>();
 public static Sprite Get(string key){
  if(cache.TryGetValue(key,out var sprite))return sprite;
  var texture=Resources.Load<Texture2D>("Art/GoalCelebrationV1/"+key);
  if(texture==null)throw new InvalidOperationException("Generated celebration art missing: "+key);
  sprite=Sprite.Create(texture,new Rect(0,0,texture.width,texture.height),Vector2.one*.5f,100,0,SpriteMeshType.FullRect);sprite.name="Generated celebration "+key;cache[key]=sprite;return sprite;
 }
}
}
