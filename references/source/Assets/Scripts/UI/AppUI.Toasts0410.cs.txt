  var center=new Vector3((safe.center.x/Mathf.Max(1,pixels.x)-.5f)*cw,((safe.center.y+safe.height*.048f)/Mathf.Max(1,pixels.y)-.5f)*ch,0);
  rt.anchoredPosition=(Vector2)parent.InverseTransformPoint(canvasRoot.TransformPoint(center));
 }
