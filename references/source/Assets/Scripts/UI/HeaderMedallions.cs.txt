 public static void Avatar(Transform parent,float x,float y,LocalProfile profile=null){var root=UICraft.Root(parent,"Avatar medallion",x,y,176,176);
 LeaguePortrait.Create(root,profile,"",0,0,176,true);
 }

 CenteredArt(visual,"button_circle_cream",148,new Vector2(101,100));CenteredArt(visual,"icon_settings",86);return button;
 }
