public sealed class ChapterHomeView:MonoBehaviour {
 RectTransform frame,bounds,background,oldBackground,avatarWidget,footer,quote;JellyChapterOrb orb;string theme,avatarResource,frameResource;static string previousTheme;bool initialized;
 public bool TransitionActive{get;private set;}
 public static void SetThemeImmediately(int chapter){previousTheme=chapter==0?"alphabet":chapter==1?"picnic":"snow";}
 public void Build(int next,Action chapters,Action start,Action avatar,Action settings,Action league,Action shop,Action<string> notify){
  theme=ChapterSceneArt.Theme(next);background=CreateBackground(transform,theme,"Live theme background");frame=UICraft.Root(transform,"Native home composition",0,0,853,1844);
  RefreshAvatar();var avatarHit=UICraft.Button(frame,"",-351,630.5f,140,146,UICraft.Green,avatar);avatarHit.name="Profile avatar";avatarHit.transform.Find("ButtonVisual").gameObject.SetActive(false);
  HeaderMedallions.Settings(frame,344,630.5f,settings);frame.Find("Settings medallion").localScale=Vector3.one*.9f;
  var logo=PastelLogo.Create(frame,0,630.5f,350,130);logo.name="Header wordmark";logo.gameObject.AddComponent<HomeWordmarkWiggle>();
  var carousel=ActivityCarousel.Create(frame,notify,route=>{if(route=="ranking")league();else if(route=="shop")shop();});var carouselRect=(RectTransform)carousel.transform;carouselRect.anchoredPosition=new Vector2(10,312);carouselRect.localScale=Vector3.one*.78f;
  var dots=frame.Find("Activity page indicators") as RectTransform;dots.anchoredPosition=new Vector2(0,67);dots.localScale=Vector3.one*.78f;
  footer=HomeProgress.PlayButton(frame,next,start).GetComponent<RectTransform>();footer.anchoredPosition=new Vector2(0,-788);footer.localScale=Vector3.one*.76f;
  
  quote=HomeFeedbackView.Create(frame).GetComponent<RectTransform>();quote.localScale=Vector3.one*.72f;quote.anchoredPosition=new Vector2(0,-104);
  bounds=UICraft.Root(transform,"Chapter orb safe bounds");orb=JellyChapterOrb.Create(bounds,chapters);
  if(previousTheme!=null&&previousTheme!=theme)StartCoroutine(Transition(previousTheme));previousTheme=theme;
 }
 static readonly Dictionary<string,Sprite> compressedBackgrounds=new Dictionary<string,Sprite>();
 RectTransform CreateBackground(Transform parent,string key,string name){
  var root=UICraft.Root(parent,name);var image=UICraft.Art(root,"NativeV50/background-"+key,0,0,853,1844);
  if(image==null){if(!compressedBackgrounds.TryGetValue(key,out var sprite)){var texture=Resources.Load<Texture2D>("Art/NativeV50/background-"+key);if(texture==null)throw new InvalidOperationException("Missing clean theme background "+key);sprite=Sprite.Create(texture,new Rect(0,0,texture.width,texture.height),Vector2.one*.5f,100,0,SpriteMeshType.FullRect);compressedBackgrounds.Add(key,sprite);}image=UICraft.Box(root,"NativeV50/background-"+key,0,0,853,1844,Color.white);image.sprite=sprite;image.type=Image.Type.Simple;}
  image.preserveAspect=false;var fit=image.gameObject.AddComponent<AspectRatioFitter>();fit.aspectMode=AspectRatioFitter.AspectMode.EnvelopeParent;fit.aspectRatio=853f/(key=="alphabet"?1843f:1844f);var canvas=GetComponentInParent<Canvas>();root.SetParent(canvas.transform,false);root.SetSiblingIndex(1);root.anchorMin=Vector2.zero;root.anchorMax=Vector2.one;root.offsetMin=root.offsetMax=Vector2.zero;return root;
 }
 IEnumerator Transition(string oldTheme){TransitionActive=true;oldBackground=CreateBackground(transform,oldTheme,"Theme dissolve background only");oldBackground.SetSiblingIndex(background.GetSiblingIndex()+1);var group=oldBackground.gameObject.AddComponent<CanvasGroup>();group.blocksRaycasts=false;float time=0;while(time<.8f){time+=Time.unscaledDeltaTime;float t=Mathf.Clamp01(time/.8f);group.alpha=1-t*t*(3-2*t);yield return null;}Destroy(oldBackground.gameObject);oldBackground=null;TransitionActive=false;orb?.Pulse();}
 public void Layout(float h,float reserve){var root=(RectTransform)transform;root.sizeDelta=new Vector2(1080,h);root.anchoredPosition=Vector2.zero;
  // The large settings medallion may tuck behind the capsule at its top;
  // keep its lower half visible instead of reserving an entire empty row.
  float lift=WeChatLayoutMetrics.Enabled?110:0;reserve=Mathf.Max(0,reserve-lift);
  // Fit the actual foreground extents, not the background's empty sky. The
  // highest header motion stays below the full-width WX capsule exclusion band.
  const float top=700,bottom=900;float fit=Mathf.Min(1080f/853,Mathf.Max(1,h-reserve-8)/(top+bottom));frame.localScale=Vector3.one*fit;frame.anchoredPosition=new Vector2(0,h*.5f-reserve-8-top*fit);
  footer.anchoredPosition=new Vector2(0,(-h*.5f+70+87.4f*fit-frame.anchoredPosition.y)/fit);
  quote.anchoredPosition=new Vector2(0,(h*.03f-reserve*.45f+lift*.55f-frame.anchoredPosition.y)/fit);
  bounds.sizeDelta=new Vector2(1080,h-reserve);bounds.anchoredPosition=new Vector2(0,-reserve*.5f);if(!initialized){orb.SetInitialPosition();initialized=true;}else orb.Relayout();RefreshAvatar();}
 void LateUpdate(){var app=GetComponentInParent<AppUI>();bool visible=app!=null&&app.CurrentPage=="home";if(background!=null)background.gameObject.SetActive(visible);if(oldBackground!=null)oldBackground.gameObject.SetActive(visible);}
 void OnDestroy(){if(background!=null)Destroy(background.gameObject);if(oldBackground!=null)Destroy(oldBackground.gameObject);}
 void RefreshAvatar(){var owner=GetComponentInParent<AppUI>();if(owner==null)return;var p=owner.Profile;if(avatarWidget!=null&&avatarResource==p.EquippedAvatarResource&&frameResource==p.EquippedFrameResource)return;avatarResource=p.EquippedAvatarResource;frameResource=p.EquippedFrameResource;if(avatarWidget!=null){avatarWidget.gameObject.SetActive(false);Destroy(avatarWidget.gameObject);}HeaderMedallions.Avatar(frame,-351,630.5f,p);avatarWidget=(RectTransform)frame.GetChild(frame.childCount-1);avatarWidget.localScale=Vector3.one*.72f;}
}
