using System;
using System.Collections;
using System.Collections.Generic;
using UnityEngine;
using UnityEngine.UI;
namespace BounceGarden {


public sealed class ChapterHomeView:MonoBehaviour {
 RectTransform frame,bounds,background,oldBackground,avatarWidget,footer,quote;JellyChapterOrb orb;string avatarResource,frameResource;bool initialized;
 
 
 
 
 
 
 public void Layout(float h,float reserve){var root=(RectTransform)transform;root.sizeDelta=new Vector2(1080,h);root.anchoredPosition=Vector2.zero;
  // The large settings medallion may tuck behind the capsule at its top;
  // keep its lower half visible instead of reserving an entire empty row.
  float lift=WeChatLayoutMetrics.Enabled?110:0;reserve=Mathf.Max(0,reserve-lift);
  // Fit the actual foreground extents, not the background's empty sky. The
  // highest header motion stays below the full-width WX capsule exclusion band.
  const float top=700,bottom=900;float fit=Mathf.Min(1080f/853,Mathf.Max(1,h-reserve-8)/(top+bottom));frame.localScale=Vector3.one*fit;frame.anchoredPosition=new Vector2(0,h*.5f-reserve-8-top*fit);
  footer.anchoredPosition=new Vector2(0,(-h*.5f+70+87.4f*fit-frame.anchoredPosition.y)/fit);
  quote.anchoredPosition=new Vector2(0,(h*.03f-reserve*.45f+lift*.55f-frame.anchoredPosition.y)/fit);
  bounds.sizeDelta=new Vector2(1080,h-reserve);bounds.anchoredPosition=new Vector2(0,-reserve*.5f);if(!initialized){orb.SetInitialPosition();initialized=true;}else orb.Relayout();RefreshAvatar();}
 void LateUpdate(){var app=GetComponentInParent<AppUI>();bool visible=app!=null&&app.CurrentPage=="home";if(background!=null)background.gameObject.SetActive(visible);if(oldBackground!=null)oldBackground.gameObject.SetActive(visible);}
 void OnDestroy(){if(background!=null)Destroy(background.gameObject);if(oldBackground!=null)Destroy(oldBackground.gameObject);}
 void RefreshAvatar(){var owner=GetComponentInParent<AppUI>();if(owner==null)return;var p=owner.Profile;if(avatarWidget!=null&&avatarResource==p.EquippedAvatarResource&&frameResource==p.EquippedFrameResource)return;avatarResource=p.EquippedAvatarResource;frameResource=p.EquippedFrameResource;if(avatarWidget!=null){avatarWidget.gameObject.SetActive(false);Destroy(avatarWidget.gameObject);}HeaderMedallions.Avatar(frame,-351,630.5f,p);avatarWidget=(RectTransform)frame.GetChild(frame.childCount-1);avatarWidget.localScale=Vector3.one*.72f;}
}
public sealed class ChapterRouteView:MonoBehaviour {
 JellyJourney80 route;public void Build(LocalProfile profile,Action back,Action<int> select,Action<string> notify){route=gameObject.AddComponent<JellyJourney80>();route.Build(profile,back,select,notify);}public void Layout(float h,float reserve,float topInset=0,float bottomInset=0,float bottomReserve=0){route?.Layout(h,reserve,topInset,bottomInset,bottomReserve);}
}
}
