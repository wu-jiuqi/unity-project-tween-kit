public partial class AppUI {
 StreakBooster pendingBoosterGuide;
 IEnumerator ShowBoosterUnlocks(int generation){
  if(Session.IsTutorial||Session.IsStreakTrial)yield break;
  for(int i=1;i<=3;i++){var k=(StreakBooster)i;if(!Profile.BoosterUnlocked(k,currentLevel))continue;if(!Profile.GrantBoosterUnlock(k,currentLevel)){Toast("道具赠送保存失败，请稍后重试。");continue;}
   if(!BattleToolsVisible||Profile.BoosterIntroSeen(k))continue;
   bool complete=false;var root=UICraft.Root(canvasRoot,"道具解锁介绍 "+k);alphabetIntroduction=root.gameObject.AddComponent<AlphabetIntroductionView>();alphabetIntroduction.Begin(this,BoosterIntroduction.Copy(k),()=>{Profile.MarkBoosterIntro(k);complete=true;});while(!complete&&generation==viewGeneration&&alphabetIntroduction!=null)yield return null;if(generation!=viewGeneration)yield break;alphabetIntroduction=null;
  }
  for(int i=1;i<=3;i++){var k=(StreakBooster)i;if(Profile.BoosterUnlocked(k,currentLevel)&&!Profile.BoosterGuideSeen(k)&&Session.StreakStock(k)>0){pendingBoosterGuide=k;break;}}
 }
 void ShowBoosterSelectionGuide(){if(!BattleToolsVisible){pendingBoosterGuide=StreakBooster.None;return;}var k=pendingBoosterGuide;pendingBoosterGuide=StreakBooster.None;if(k==StreakBooster.None||!Session.CanSelectStreakBooster)return;var root=UICraft.Root(hudRoot,"首次道具选择引导");root.gameObject.AddComponent<BoosterSelectionGuide>().Build(this,Session,k);}
 public void OpenInGameBoosterOffer(StreakBooster kind){
  if(!playing||Session.IsPaused||Session.IsTutorial||Session.IsStreakTrial||!Profile.BoosterUnlocked(kind,currentLevel))return;
  CancelAim();Session.SetPaused(true);OpenModal("补充"+StreakTrials.Name(kind),UICraft.Green,860,1000);
  var image=UICraft.Box(modal,"道具独立插图",0,185,230,230,Color.white);image.sprite=ScoreSkillArt.Booster(kind);image.type=Image.Type.Simple;image.preserveAspect=true;
  UICraft.Label(modal,StreakTrials.Rule(kind),0,-15,740,125,30,UICraft.Ink);UICraft.Label(modal,"金币 "+Profile.Coins+" · 已拥有 "+Profile.BoosterStock(kind)+" / 99",0,-145,740,70,32,UICraft.Ink);
  string transaction="ingame-v69/"+Guid.NewGuid().ToString("N");bool bought=false;int price=LocalProfile.BoosterPrice(kind);
  var button=UICraft.Button(modal,price+" 金币 · 购买 1 件",0,-260,650,110,UICraft.Green,()=>{if(bought)return;if(!Profile.PurchaseBooster(transaction,kind)){Toast(Profile.Coins<price?"金币不足":Profile.BoosterStock(kind)>=99?"背包已满":"购买保存失败，请重试。");return;}bought=true;SoundService.Play("purchase_success");RemoveModal(true);Session.NotifyBoosterStock();if(Session.CanSelectStreakBooster&&Session.SelectedStreakBooster==StreakBooster.None)Session.SelectStreakBooster(kind);Toast("已补充"+StreakTrials.Name(kind));},36);button.name="局内购买 "+kind;
  UICraft.Button(modal,"返回关卡",0,-405,480,90,UICraft.Blue,CloseModal,34);
 }

  UICraft.Box(transform,"左遮挡",(-540+left)*.5f,0,left+540,2600,new Color(0,0,0,.35f),false,true);UICraft.Box(transform,"右遮挡",(540+right)*.5f,0,540-right,2600,new Color(0,0,0,.35f),false,true);UICraft.Box(transform,"上遮挡",x,(1300+top)*.5f,252,1300-top,new Color(0,0,0,.35f),false,true);UICraft.Box(transform,"下遮挡",x,(-1300+bottom)*.5f,252,bottom+1300,new Color(0,0,0,.35f),false,true);

  for(int i=0;i<4;i++)UICraft.Box(transform,"金色选中引导",x+(i==0?-126:i==1?126:0),y+(i==2?-126:i==3?126:0),i<2?6:252,i<2?252:6,new Color(1,.82f,.3f));

 }

}
