// Fit only the right badge between the native capsule and the real top vine.
// The child owns the squash animation; this stable parent owns safe positioning.
public sealed class BattleMovesPlacement:MonoBehaviour {
 RectTransform rect,visual;UnityEngine.UI.Image plate;TMP_Text number,caption;Vector2 origin;GameSession game;int generation;readonly Vector3[] corners=new Vector3[4];
 public void Initialize(GameSession g){game=g;generation=g.Generation;rect=(RectTransform)transform;origin=rect.anchoredPosition;}
 public void Bind(RectTransform face,UnityEngine.UI.Image background,TMP_Text value,TMP_Text label){visual=face;plate=background;number=value;caption=label;}
 void Compact(bool compact){if(visual==null)return;var size=new Vector2(170,compact?72:128);rect.sizeDelta=visual.sizeDelta=plate.rectTransform.sizeDelta=size;number.rectTransform.anchoredPosition=compact?new Vector2(26,0):new Vector2(0,-13);number.rectTransform.sizeDelta=compact?new Vector2(112,72):new Vector2(150,76);number.fontSize=compact?72:64;number.textWrappingMode=TextWrappingModes.NoWrap;caption.text=compact?"剩余\n步数":"剩余步数";caption.rectTransform.anchoredPosition=compact?new Vector2(-49,0):new Vector2(0,35);caption.rectTransform.sizeDelta=compact?new Vector2(56,64):new Vector2(160,34);caption.fontSize=compact?22:26;caption.enableAutoSizing=false;caption.lineSpacing=compact?-5:0;}
 void LateUpdate(){if(rect==null||game==null||game.Generation!=generation||Camera.main==null)return;rect.anchoredPosition=origin;rect.localScale=Vector3.one;Compact(false);
  var app=GetComponentInParent<AppUI>();Rect blocked;if(!WeChatLayoutMetrics.TryGetBlockedRect(app!=null?app.MenuAvoidanceViewport:new Vector2Int(Screen.width,Screen.height),out blocked))return;
  var canvas=GetComponentInParent<Canvas>();var camera=canvas.renderMode==RenderMode.ScreenSpaceOverlay?null:canvas.worldCamera;rect.GetWorldCorners(corners);
  var lo=RectTransformUtility.WorldToScreenPoint(camera,corners[0]);var hi=RectTransformUtility.WorldToScreenPoint(camera,corners[2]);if(hi.x<=blocked.xMin||lo.x>=blocked.xMax)return;
  float floor=Camera.main.WorldToScreenPoint(BoardDisplay.Forward(new Vector2(0,JellyWallFrame.TopCenter))).y+6;
  float ceiling=blocked.yMin+16*WeChatLayoutMetrics.LogicalPixelScale(app!=null?app.MenuAvoidanceViewport:new Vector2Int(Screen.width,Screen.height));if(hi.y<=ceiling&&lo.y>=floor)return;
  float available=ceiling-floor;if(available<=0)return;if(available/((hi.y-lo.y)*1.05f)<.9f&&visual!=null){Compact(true);rect.GetWorldCorners(corners);lo=RectTransformUtility.WorldToScreenPoint(camera,corners[0]);hi=RectTransformUtility.WorldToScreenPoint(camera,corners[2]);}float fit=Mathf.Min(1,available/((hi.y-lo.y)*1.05f));rect.localScale=Vector3.one*fit;
  float half=(hi.y-lo.y)*fit*.5f;float cy=Mathf.Clamp((lo.y+hi.y)*.5f,floor+half,ceiling-half);Vector2 a,b;
  RectTransformUtility.ScreenPointToLocalPointInRectangle((RectTransform)rect.parent,(lo+hi)*.5f,camera,out a);RectTransformUtility.ScreenPointToLocalPointInRectangle((RectTransform)rect.parent,new Vector2((lo.x+hi.x)*.5f,cy),camera,out b);rect.anchoredPosition=origin+b-a;
 }
}
