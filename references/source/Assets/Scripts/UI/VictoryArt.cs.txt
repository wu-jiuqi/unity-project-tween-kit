using System;
using System.Collections.Generic;
using UnityEngine;

namespace BounceGarden {
// Shared immutable textures and sprite rectangles. No character assets are loaded.
public static class VictoryArt {
    [Serializable] sealed class Entry { public string name, texture; public int[] rect; }
    [Serializable] sealed class Layout { public Entry[] sprites; }
    static readonly Dictionary<string, Sprite> sprites = new Dictionary<string, Sprite>();
    static readonly Dictionary<string, Texture2D> textures = new Dictionary<string, Texture2D>();
    static bool loaded;
    public static Sprite Get(string name) {
        if (!loaded) Load();
        if (sprites.TryGetValue(name, out var sprite)) return sprite;
        throw new InvalidOperationException("Missing victory sprite: " + name);
    }
    static void Load() {
        var text = Resources.Load<TextAsset>("Art/VictoryV23/atlases");
        if (text == null) throw new InvalidOperationException("VictoryV23 atlas manifest is missing.");
        var layout = JsonUtility.FromJson<Layout>(text.text);
        foreach (var entry in layout.sprites) {
            if (!textures.TryGetValue(entry.texture, out var texture)) {
                texture = Resources.Load<Texture2D>("Art/VictoryV23/" + entry.texture);
                if (texture == null) throw new InvalidOperationException("Missing victory atlas: " + entry.texture);
                textures.Add(entry.texture, texture);
            }
            var r = entry.rect;
            float sourceW=entry.texture=="halo"?3072:1024,sourceH=entry.texture=="halo"?2048:entry.texture=="glyphs"?768:1024;
            float sx=texture.width/sourceW,sy=texture.height/sourceH;
            var sprite = Sprite.Create(texture, new Rect(r[0]*sx, texture.height-(r[1]+r[3])*sy, r[2]*sx, r[3]*sy),
                new Vector2(.5f, .5f), 100, 0, SpriteMeshType.FullRect);
            sprite.name = entry.name;
            sprites.Add(entry.name, sprite);
        }
        loaded = true;
    }
}
}
