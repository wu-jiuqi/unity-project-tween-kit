using System;
using UnityEngine;
using UnityEngine.UI;
using TMPro;
namespace BounceGarden {
// One visible clock, independent of paused simulation and resource-stage changes.
public sealed class LoadingPresentation:MonoBehaviour {
 public const float MinimumDuration=2.6f;
 public float Elapsed{get;private set;}
 public string CurrentPose=>box==null?"":box.CurrentPose;
 
 public BrandGardenHedge Hedge{get;private set;}
 public int Percent=>complete&&shown>=1&&Hedge.GrowthSettled?100:Mathf.Min(99,Mathf.FloorToInt(shown*100));
 public bool CanDismiss=>Elapsed>=MinimumDuration&&Percent==100;
 
 
 public Button ReturnButton{get;private set;}
 RectTransform content,backdrop;LoadingBoxAnimation box;
 TMP_Text label,number;bool running,complete,firstTick;float target,shown;AppUI app;
 public void Build(int level=1,Action onReturn=null){
  app=FindFirstObjectByType<AppUI>();
  var bg=UICraft.Box(transform,"Loading courtyard",0,0,6000,6000,Color.white,false,true);bg.sprite=BrandSplashArt.Get("Garden/courtyard");bg.type=Image.Type.Simple;backdrop=bg.rectTransform;
  content=UICraft.Root(transform,"Cream loading content",0,0,1080,1920);
  LevelLoadingArt.Image(content,"panel",0,5,940);
  Text("准备出发",0,480,800,100,72);
  box=LoadingBoxAnimation.Create(content,145);
  label=Text("正在准备第"+level+"关",0,-285,840,85,47);
  var bed=UICraft.Root(content,"Level growing garden",0,-480,1000,300);bed.localScale=Vector3.one*.83f;Hedge=bed.gameObject.AddComponent<BrandGardenHedge>();Hedge.Build();
  number=Text("0%",0,-585,400,65,46);
  Text("把花园里的小美好，装进口袋。",0,-655,800,70,31);
  ReturnButton=CommonButtonsA.Button(content,"返回花园","返回花园",0,-864,820,128,onReturn,"secondary",53);
  Layout();Render();
 }
 TMP_Text Text(string value,float x,float y,float w,float h,int size){var t=UICraft.Label(content,value,x,y,w,h,size,CSettingsArt.Ink);t.color=CSettingsArt.Ink;t.fontStyle=FontStyles.Bold;return t;}
 
 public void SetProgress(float value){if(!complete)target=Mathf.Max(target,Mathf.Clamp(value,0,.99f));}
 public void Complete(){complete=true;target=1;}
 public void SuspendForError(){running=false;if(content!=null)content.gameObject.SetActive(false);}
 public void Play(){if(running)return;Elapsed=0;running=firstTick=true;Render();}
 void Update(){
  if(!running||content==null)return;
  if(firstTick){firstTick=false;return;}
  float dt=Mathf.Min(Time.unscaledDeltaTime,.05f);Elapsed+=dt;shown=Mathf.MoveTowards(shown,target,dt*.72f);Hedge.SetProgress(shown);number.text=Percent+"%";Render();
 }
 // Compensate the battle parent's separate horizontal/vertical scaling.
 void Layout(){
  if(content==null)return;var canvas=GetComponentInParent<Canvas>().rootCanvas;var cr=(RectTransform)canvas.transform;var parent=(RectTransform)transform;
  var viewport=app!=null?app.MenuAvoidanceViewport:new Vector2Int(Screen.width,Screen.height);
  var area=app!=null?app.LoadingSafeArea:Screen.safeArea;
  float w=cr.rect.width*area.width/viewport.x,h=cr.rect.height*area.height/viewport.y;
  float fit=Mathf.Min(w/1080f,h/1920f);
  Vector3 factors=new Vector3(cr.lossyScale.x/parent.lossyScale.x,cr.lossyScale.y/parent.lossyScale.y,1);
  content.localScale=new Vector3(fit*factors.x,fit*factors.y,1);
  content.position=cr.TransformPoint(new Vector3(cr.rect.xMin+cr.rect.width*area.center.x/viewport.x,cr.rect.yMin+cr.rect.height*area.center.y/viewport.y,0));
  float cover=Mathf.Max(cr.rect.width/backdrop.GetComponent<Image>().sprite.rect.width,cr.rect.height/backdrop.GetComponent<Image>().sprite.rect.height);
  backdrop.position=cr.TransformPoint(Vector3.zero);backdrop.localScale=factors;backdrop.sizeDelta=backdrop.GetComponent<Image>().sprite.rect.size*cover;
 }
 void LateUpdate(){Layout();}
 void Render(){if(box!=null)box.Render(Elapsed);}
 void OnDisable(){running=false;}
}
}
