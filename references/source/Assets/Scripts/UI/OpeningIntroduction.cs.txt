 public static IEnumerable<int> LessonKeys(int level){
  var pieces=OpeningCampaign.Stage(level).pieces;
  if(level==0)yield return 0;
  if(pieces.Any(p=>p.kind=="yarn"))yield return pieces.Any(p=>p.kind=="yarn"&&p.layers>=3)?17:pieces.Any(p=>p.kind=="yarn"&&p.layers>=2)?5:4;
  if(pieces.Any(p=>p.kind=="cookie_box"))yield return 10;
  if(pieces.Any(p=>p.kind=="lucky_cat"))yield return 20;
  if(pieces.Any(p=>p.kind=="copper_coin"))yield return 30;
 }
}
