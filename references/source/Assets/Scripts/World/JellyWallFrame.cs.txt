using UnityEngine;

namespace BounceGarden
{
    // One continuous U-shaped surface. Its shader changes only the artwork;
    // StaticAimWall and the three BoxCollider2D boundaries remain authoritative.
    public sealed class JellyWallFrame : MonoBehaviour
    {
        public const int Capacity = 16;
        public const float Duration = .50f;
        
        public const float HalfWidth = .18f;
        // Stop above the sleeping cat, matching the approved garden painting.
        // This is visual only; collider bounds and the return line are unchanged.
        
        
        public static float TopCenter => FlightGeometry.Top + HalfWidth;

        struct Pulse { public Vector2 point; public int side; public float age,strength; }
        readonly Pulse[] pulses = new Pulse[Capacity];
        readonly Vector4[] shaderPulses = new Vector4[Capacity];
        readonly Vector4[] shaderAges = new Vector4[Capacity];
        GameSession owner;
        Material material;
        Mesh mesh;
        int count;
        
        

        

        // Side: 0 left, 1 ceiling, 2 right. Contacts are passed by the ball after
        // the physics solver, so no second reflection or velocity impulse occurs.
        

        

        
        
        

        

        public void Advance(float deltaTime)
        {
            if (owner == null || owner.IsPaused) return;
            if (!owner.IsPlaying) { Clear(); return; }
            if (count == 0) return;
            for (int i = count - 1; i >= 0; i--)
            {
                pulses[i].age += Mathf.Max(0, deltaTime);
                if (pulses[i].age >= Duration) pulses[i] = pulses[--count];
            }
            Upload();
        }

        void LateUpdate() { Advance(Time.deltaTime); }
        public void Clear() { count = 0; Upload(); }
        void Upload()
        {
            if (material == null) return;
            for (int i = 0; i < Capacity; i++){
                shaderPulses[i] = i < count ? new Vector4(pulses[i].point.x, pulses[i].point.y,pulses[i].side,pulses[i].strength) : Vector4.zero;
                shaderAges[i]=i<count?new Vector4(pulses[i].age,0,0,0):Vector4.zero;
            }
            material.SetInt("_PulseCount", count);
            material.SetVectorArray("_Pulses", shaderPulses);
            material.SetVectorArray("_PulseAges",shaderAges);
        }

        void OnDisable() { Clear(); }
        void OnDestroy()
        {
            if (material != null) Destroy(material);
            if (mesh != null) Destroy(mesh);
        }
    }

    
}
