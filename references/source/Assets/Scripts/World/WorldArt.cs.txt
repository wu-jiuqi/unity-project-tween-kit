    // Runtime fallback illustration. Replaced by Resources/Art sprites when supplied.
    public static class WorldArt
    {
        static readonly Dictionary<string, Sprite> cache = new Dictionary<string, Sprite>();
        public static readonly Color[] Palette = {
            new Color(.58f,.75f,.96f), new Color(1f,.85f,.47f), new Color(.97f,.65f,.67f),
            new Color(.74f,.62f,.90f), new Color(.62f,.88f,.75f), new Color(.96f,.78f,.63f)
        };
        public static Sprite Sprite(string kind)
        {
            if (cache.TryGetValue(kind, out Sprite known)) return known;
            var production=ProductionArt.Get(kind);
            if(production!=null){cache[kind]=production;return production;}
            Sprite supplied;Texture2D texture;
            if(kind!="brick"&&kind!="ring"&&kind!="circle"&&kind!="ball"&&kind!="star"&&kind!="grass"&&kind!="water")throw new System.InvalidOperationException("Unknown artwork: "+kind);
            const int n = 96;
            texture = new Texture2D(n,n,TextureFormat.RGBA32,false) { name = "Fallback_" + kind, filterMode = FilterMode.Bilinear, wrapMode = TextureWrapMode.Clamp };
            var pixels = new Color[n*n];
            for (int y=0;y<n;y++) for(int x=0;x<n;x++)
            {
                float u=(x+.5f)/n*2-1, v=(y+.5f)/n*2-1;
                float edge=1-Mathf.Max(Mathf.Abs(u),Mathf.Abs(v));
                Color c=Color.white;
                if(kind=="ring")
                {float r=Mathf.Sqrt(u*u+v*v);c=new Color(1,1,1,Mathf.Clamp01((.12f-Mathf.Abs(r-.72f))*35));}
                else if(kind=="ball" || kind=="circle")
                {
                    float r=Mathf.Sqrt(u*u+v*v); float a=Mathf.Clamp01((1-r)*n*.5f);
                    float shade=kind=="ball" ? .8f+.2f*v : 1;
                    c=new Color(shade,shade,shade,a);
                    if(kind=="ball") { float gleam=Mathf.Exp(-((u+.3f)*(u+.3f)+(v-.35f)*(v-.35f))*30); c=Color.Lerp(c,Color.white,gleam*.8f); c.a=a; }
                }
                else if(kind=="star")
                {
                    float r=Mathf.Sqrt(u*u+v*v); float a=Mathf.Atan2(v,u);
                    float lim=.54f+.25f*Mathf.Cos(a*5-1.57f); c.a=Mathf.Clamp01((lim-r)*40);
                }
                else if(kind=="grass")
                {
                    float noise=Mathf.PerlinNoise(x*.24f,y*.24f); bool check=((x/16+y/16)%2)==0;
                    c=new Color(.32f+noise*.035f,.67f+noise*.05f,.27f+noise*.04f); if(check)c*=1.035f; c.a=1;
                }
                else if(kind=="water")
                {
                    float wave=Mathf.Sin(x*.19f+y*.5f)*.025f;
                    c=new Color(.13f+wave,.63f+v*.04f+wave,.86f+wave,1);
                    if(v>.8f)c=Color.Lerp(c,Color.white,.5f);
                }
                else
                {
                    float qx=Mathf.Max(Mathf.Abs(u)-.79f,0), qy=Mathf.Max(Mathf.Abs(v)-.79f,0);
                    float a=Mathf.Clamp01((.2f-Mathf.Sqrt(qx*qx+qy*qy))*50);
                    float shade=.86f+.14f*(v*.5f+.5f);
                    if(edge<.12f)shade*=.78f;
                    if(v>.72f && edge>.12f)shade=1;
                    c=new Color(shade,shade,shade,a);
                }
                pixels[y*n+x]=c;
            }
            texture.SetPixels(pixels);texture.Apply(false,true);
            supplied=UnityEngine.Sprite.Create(texture,new Rect(0,0,n,n),new Vector2(.5f,.5f),n);
            cache[kind]=supplied;return supplied;
        }

        public static SpriteRenderer Draw(Transform parent,string name,string sprite,Vector2 position,Vector2 size,Color color,int order)
        {
            var go=new GameObject(name);go.transform.SetParent(parent,false);go.transform.localPosition=position;
            var renderer=go.AddComponent<SpriteRenderer>();renderer.sprite=Sprite(sprite);renderer.color=color;renderer.sortingOrder=order;
            renderer.sharedMaterial=BoardMaterial;
            
            var bounds=renderer.sprite.bounds.size;
            go.transform.localScale=new Vector3(size.x/bounds.x,size.y/bounds.y,1);
            if(PastelArt.Info(renderer.sprite.name)!=null)PastelArt.FitVisible(renderer,position,size);
            return renderer;
        }
        public static TMP_Text Text(Transform parent,string name,string text,Vector2 position,float size,Color color,int order)
        {
            var go=new GameObject(name);go.transform.SetParent(parent,false);go.transform.localPosition=new Vector3(position.x,position.y,-.1f);
            var mesh=go.AddComponent<TextMeshPro>();mesh.font=GameTypography.Font;mesh.text=text;mesh.fontSize=size*60;mesh.alignment=TextAlignmentOptions.Center;mesh.color=color;mesh.rectTransform.sizeDelta=new Vector2(2,1);mesh.overflowMode=TextOverflowModes.Overflow;
            if(worldText==null){worldText=new Material(GameTypography.Font.material);worldText.shader=Resources.Load<Shader>("WorldDigits");}mesh.fontSharedMaterial=worldText;
            mesh.GetComponent<MeshRenderer>().sortingOrder=order;return mesh;
        }
        static Material boardMaterial,worldText;
        public static Material BoardMaterial{get{if(boardMaterial==null)boardMaterial=new Material(Resources.Load<Shader>("HitFlash"));return boardMaterial;}}
    }

    public sealed class WorldFx:MonoBehaviour {
        sealed class Particle {public SpriteRenderer renderer;public Vector2 velocity;public float left,life,size,gravity,spin;public Color color;public string key;}
        readonly List<Particle> live=new List<Particle>();readonly Stack<Particle> pool=new Stack<Particle>();readonly Dictionary<Vector2Int,float> pulse=new Dictionary<Vector2Int,float>();readonly System.Random random=new System.Random(6006);
        JellyPopPool jelly;
        public int JellyActiveCount=>jelly!=null?jelly.ActiveCount:0;
        public void PrepareJelly(int count){jelly=gameObject.AddComponent<JellyPopPool>();jelly.Initialize(Owner,count);}
        public void PlayJelly(BrickBody brick){if(jelly!=null&&brick.UsesJelly)jelly.Play(brick.JellyPosition,brick.RestFaceScale,brick.Face.sortingOrder,brick.PaintFrames);}
        public GameSession Owner;public int Peak{get;private set;}public int Reused{get;private set;}public int RingCount{get;private set;}
        float R(float min,float max)=>(float)(min+random.NextDouble()*(max-min));
        public void Emit(Vector2 position,string cue,Color color,string appearance=null){
            if(!FeedbackProfile.Active.enabledFeedback)return;float cell=Owner!=null?Owner.Board.CellWidth:.73f;
            if(cue=="hit"){var index=Owner.Board.Nearest(position);if(pulse.TryGetValue(index,out float prior)&&Time.time-prior<.05f)return;pulse[index]=Time.time;Add("impact_star",position,.10f,cell*.34f,Color.white,Vector2.zero,0);return;}
            if(cue=="break"&&appearance==JellyBrickArt.SpriteName)return;
            if(cue=="break"){
                Add("impact_star",position,.10f,cell*.38f,Color.white,Vector2.zero,0);Add("shockwave_ring",position,.16f,cell*.68f,new Color(1,.94f,.72f,.65f),Vector2.zero,0);RingCount++;
                string debris=appearance=="blue_star"?"debris_blue":appearance=="orange_star"?"debris_orange":"debris_neutral";Color tint=debris=="debris_neutral"?color:Color.white;string pastelColor=appearance==null?"":appearance.Replace("brick_blank_","").Replace("brick_","");if(pastelColor=="blue"||pastelColor=="mint"||pastelColor=="blush"||pastelColor=="yellow"||pastelColor=="lavender"){debris="pastel/fx_shard_"+pastelColor;tint=Color.white;}
                for(int i=0;i<4;i++){float angle=(i+.25f)*Mathf.PI*.5f;Add(debris,position,.32f,cell*R(.08f,.13f),tint,new Vector2(Mathf.Cos(angle),Mathf.Sin(angle))*cell*R(2.2f,3.2f),cell*8);}
                Add("four_point_spark",position,.17f,cell*.08f,new Color(1,.94f,.7f,.75f),Vector2.up*cell,0);return;
            }
            if(cue=="shoot"){Add("impact_star",position,.07f,cell*.18f,new Color(1,.93f,.6f,.85f),Vector2.zero,0);Add("short_streak",position,.08f,cell*.24f,new Color(1,.95f,.7f,.5f),Vector2.down*cell,0);return;}
            if(cue=="splash")for(int i=0;i<4;i++)Add("four_point_spark",position,.2f,cell*.09f,new Color(.6f,.9f,1,.6f),new Vector2(R(-1,1),R(1,2))*cell,cell*9);
        }
        void Add(string key,Vector2 position,float duration,float size,Color color,Vector2 velocity,float gravity){
            if(live.Count>=64){if(key!="impact_star")return;Recycle(live.Count-1);}Particle p;if(pool.Count>0){p=pool.Pop();Reused++;}else p=new Particle{renderer=WorldArt.Draw(transform,"Pooled Round6 feedback","circle",Vector2.zero,Vector2.one,Color.white,30)};
            p.key=key;p.renderer.sprite=ProductionArt.Get(key.StartsWith("pastel/")?key:"round6/"+key);p.renderer.gameObject.SetActive(true);p.renderer.transform.position=position;p.renderer.transform.rotation=Quaternion.identity;p.renderer.color=color;p.color=color;p.velocity=velocity;p.left=p.life=duration;p.size=size;p.gravity=gravity;p.spin=R(-240,240);SetSize(p,key=="shockwave_ring"?size*.2f/.68f:size);live.Add(p);Peak=Mathf.Max(Peak,live.Count);
        }
        void Recycle(int i){var p=live[i];p.renderer.gameObject.SetActive(false);p.renderer.color=Color.white;p.renderer.transform.localScale=Vector3.one;p.renderer.transform.localRotation=Quaternion.identity;p.velocity=Vector2.zero;pool.Push(p);live.RemoveAt(i);}
        void Update(){if(Owner!=null&&Owner.IsPaused)return;float dt=Owner!=null?Owner.PresentationDelta(Time.deltaTime):Time.deltaTime;for(int i=live.Count-1;i>=0;i--){var p=live[i];p.left-=dt;if(p.left<=0){Recycle(i);continue;}p.velocity.y-=p.gravity*dt;p.renderer.transform.position+=(Vector3)(p.velocity*dt);p.renderer.transform.Rotate(0,0,p.spin*dt);float age=1-p.left/p.life;float alpha=p.key.StartsWith("debris")?Mathf.Clamp01((1-age)/.4f):1-age;p.renderer.color=new Color(p.color.r,p.color.g,p.color.b,p.color.a*alpha);SetSize(p,p.size*(p.key=="shockwave_ring"?Mathf.Lerp(.2f/.68f,1,age):p.key.StartsWith("debris")?Mathf.Lerp(1,.6f,age):1));}}
        static void SetSize(Particle p,float size){var b=p.renderer.sprite.bounds.size;var info=PastelArt.Info(p.renderer.sprite.name);float pad=info==null?1:Mathf.Max(info.size[0],info.size[1])/(float)Mathf.Max(info.visible[2],info.visible[3]);float scale=size*pad/Mathf.Max(b.x,b.y);float y=p.renderer.transform.position.y;float slope=(BoardDisplay.Forward(new Vector2(0,y+.001f)).y-BoardDisplay.Forward(new Vector2(0,y)).y)/.001f;p.renderer.transform.localScale=new Vector3(scale,scale/Mathf.Max(.2f,slope),1);}
    }
