using System;
using System.Collections.Generic;
using UnityEngine;
namespace BounceGarden {
// Gold is painted into these atlases; ordinary pair colors only tint their halo.
public sealed class IngotSpecialFx:MonoBehaviour {
 static readonly Dictionary<string,Sprite[]> sheets=new Dictionary<string,Sprite[]>();
 static readonly Dictionary<Texture,Material> materials=new Dictionary<Texture,Material>();
 public static readonly string[] Keys={"laser","bomb","flower","merge"};
 static readonly Dictionary<GameSession,int> counts=new Dictionary<GameSession,int>();
 public static int Alive(GameSession g)=>counts.TryGetValue(g,out int n)?n:0;
 GameSession owner;int generation;SpriteRenderer face;string key;float age,duration,size,width;Vector2 from,to;bool beam;
 public string Key=>key;
 public static Sprite Frame(string key,int index){if(!sheets.TryGetValue(key,out var frames)){var t=Resources.Load<Texture2D>("Art/IngotSpecialFXV1/"+key);if(t==null)throw new InvalidOperationException("Missing ingot skill animation: "+key);frames=new Sprite[16];float w=t.width/4f,h=t.height/4f;for(int i=0;i<16;i++){frames[i]=Sprite.Create(t,new Rect(i%4*w,(3-i/4)*h,w,h),Vector2.one*.5f,100,0,SpriteMeshType.FullRect);frames[i].name="ingot-fx-"+key+"-"+i;}sheets[key]=frames;}return frames[Mathf.Clamp(index,0,15)];}
 public static IngotSpecialFx Play(GameSession game,Transform root,string key,Vector2 p,float size,int accent=-1,float duration=.76f){var f=new GameObject("Ingot skill sequence · "+key).AddComponent<IngotSpecialFx>();f.transform.SetParent(root,false);f.transform.position=p;f.owner=game;counts[game]=Alive(game)+1;f.generation=game.Generation;f.key=key;f.size=size;f.duration=duration;f.face=f.gameObject.AddComponent<SpriteRenderer>();f.face.sprite=Frame(key,0);f.face.sortingOrder=80;var t=f.face.sprite.texture;if(!materials.TryGetValue(t,out var m)){m=new Material(Resources.Load<Shader>("IngotFx")){mainTexture=t};materials[t]=m;}f.face.sharedMaterial=m;var block=new MaterialPropertyBlock();block.SetColor("_Accent",accent<0?Color.white:SevenSpecialArt.Palette[Mathf.Clamp(accent,0,4)]);block.SetFloat("_Mix",accent<0?0:.24f);f.face.SetPropertyBlock(block);f.Pose();return f;}
 public static void Skill(GameSession g,Transform root,int kind,Vector2 p,float size,int accent=-1){if(kind==3)Play(g,root,"flower",p,size*1.8f,accent);else if(kind==2)Play(g,root,"bomb",p,size*1.7f,accent);}
 public static void Beam(GameSession g,Transform root,Vector2 from,Vector2 to,float width,int accent=-1,float life=.66f){if((to-from).sqrMagnitude<.0001f)return;var f=Play(g,root,"laser",(from+to)*.5f,1,accent,life);f.beam=true;f.from=from;f.to=to;f.width=width;f.Pose();}
 void OnDestroy(){if(!ReferenceEquals(owner,null)&&counts.TryGetValue(owner,out int n)){if(n<=1)counts.Remove(owner);else counts[owner]=n-1;}}
 void Pose(){var b=face.sprite.bounds.size;if(beam){var delta=to-from;transform.position=(from+to)*.5f;transform.rotation=Quaternion.Euler(0,0,Mathf.Atan2(delta.y,delta.x)*Mathf.Rad2Deg);transform.localScale=new Vector3(delta.magnitude/(b.x*.89f),width/(b.y*.48f),1);}else transform.localScale=Vector3.one*(size/b.x);}
 void Update(){if(BalanceClock.Active)return;Advance(Time.unscaledDeltaTime);}
 public void Advance(float dt){if(owner==null||owner.Generation!=generation){Destroy(gameObject);return;}if(owner.IsPaused)return;age+=owner.PresentationDelta(dt);if(age>=duration){if(BalanceClock.Active){OnDestroy();owner=null;gameObject.SetActive(false);}Destroy(gameObject);return;}face.sprite=Frame(key,Mathf.FloorToInt(age/duration*16));Pose();if(key=="flower")transform.rotation=Quaternion.Euler(0,0,-200*age/duration);}
 }
}
