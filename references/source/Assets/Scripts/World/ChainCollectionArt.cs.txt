using UnityEngine;
namespace BounceGarden {
public static class ChainCollectionArt {
 static GameSession Owner(Transform t)=>t.GetComponentInParent<GameSession>();
 public static Transform Loot(Transform parent,int kind,Vector2 p,float size,int order){var root=new GameObject(kind==1?"Fish biscuit loot":"Plush star loot").transform;root.SetParent(parent,false);root.localPosition=p;root.localScale=Vector3.one*size;var owner=Owner(parent);var icon=ChainHudArt.Frame(kind==1?4:5);var rendered=ChainVisualArt.Draw(root,"Painted loot",icon,1f/Mathf.Max(.01f,icon.bounds.size.x),order);return root;}
 public static void Container(BrickBody b){int kind=b.Definition.juiceGroup;if(kind<=0)return;var root=new GameObject("Container shell").transform;root.SetParent(b.transform,false);b.Face.enabled=false;root.gameObject.AddComponent<ChainCubeView>().Initialize(b,Owner(b.transform),kind);var lid=new GameObject("Lid").transform;lid.SetParent(root,false);lid.localPosition=new Vector2(0,.29f);}
 public static void Damage(BrickBody b){SpecialDamage(b);var root=b.transform.Find("Container shell");if(root==null)return;root.GetComponent<ChainCubeView>().Refresh();if(b.Definition.juiceGroup==2){root.Find("Lid").localPosition=new Vector2(.1f,.48f);}}
 public static void Special(BrickBody b){if(ChainPowerRules.IsNest(b.Definition)){b.Face.enabled=false;var nestRoot=new GameObject("Large square shell").transform;nestRoot.SetParent(b.transform,false);nestRoot.gameObject.AddComponent<ChainCubeView>().Initialize(b,Owner(b.transform),3);return;}var kind=b.Definition.kind;var owner=Owner(b.transform);if(kind==BrickKind.Stone){b.gameObject.AddComponent<ChainStoneView>().Initialize(owner,b);return;}if(kind==BrickKind.Armor){var shell=new GameObject("Armor shell").transform;shell.SetParent(b.transform,false);return;}if(kind!=BrickKind.RearWeak)return;var root=new GameObject("Rear guard").transform;root.SetParent(b.transform,false);b.Face.enabled=false;root.gameObject.AddComponent<ChainBrickView>().Initialize(b,owner,3);WorldArt.Text(root,"Rear HP",b.CurrentHealth.ToString(),Vector2.zero,.07f,Color.white,69);}
 public static void SpecialDamage(BrickBody b){var cube=b.GetComponentInChildren<ChainCubeView>();if(cube!=null)cube.Refresh();if(ChainPowerRules.IsNest(b.Definition)){b.GetComponentInChildren<ChainCubeView>().Refresh();return;}var shell=b.transform.Find("Armor shell");if(shell!=null&&!b.ShellIntact&&shell.gameObject.activeSelf){shell.gameObject.SetActive(false);ChainVisualArt.Effect(Owner(b.transform),3,1,b.transform.position,1.05f);}var root=b.transform.Find("Rear guard");if(root!=null){root.Find("Rear HP").GetComponent<TMPro.TMP_Text>().text=b.CurrentHealth.ToString();root.GetComponent<ChainBrickView>().Hit(true);}}
 public static void Block(BrickBody b,Vector2 point){var stone=b.GetComponent<ChainStoneView>();if(stone!=null){stone.Hit();return;}var view=b.GetComponentInChildren<ChainBrickView>();if(view!=null)view.Block(point);}
 public static bool Broken(BrickBody b){var owner=Owner(b.transform);if(ChainPowerRules.IsNest(b.Definition))return true;if(b.Definition.juiceGroup>0){ChainCubeArt.Burst(owner,b.transform.position,1.05f);return true;}if(b.Definition.kind==BrickKind.RearWeak){ChainVisualArt.Effect(owner,3,2,b.transform.position,1.05f);return true;}return false;}
 public static void LegacyPreview(Transform parent,BrickKind kind,float size){var go=new GameObject("Legacy "+kind+" preview");go.transform.SetParent(parent,false);var renderer=go.AddComponent<SpriteRenderer>();renderer.sprite=AbilityArt.Brick(kind);renderer.sortingOrder=56;renderer.sharedMaterial=WorldArt.BoardMaterial;go.transform.localScale=Vector3.one*(size/1.84f);}
 public static Transform Tool(Transform parent,int kind,Vector2 p)=>ChainCubeArt.Tool(parent,kind,p);
 public static Transform Mini(Transform parent,Vector2 p){var root=new GameObject("Drum yarn mini ball").transform;root.SetParent(parent,false);root.localPosition=p;ChainVisualArt.Draw(root,"Painted yarn",ChainVisualArt.Frame(2,2,2),.2f,70);return root;}
}
}
