        public void Create(GameSession session, PhysicsMaterial2D material)

            owner=session;gameObject.layer=9;

        public void Initialize(int id,int generation,Vector2 position,Vector2 velocity,float gravity)
        {
            ChainBreaks=ChainBankBreaks=0;ChainBank=ChainGenerated=false;IncomingDirection=velocity.normalized;impact=0;Id=id;SessionGeneration=generation;Active=true;enteredBoard=false;previousPosition=position;previousY=position.y;lastContactTime=-1;

            int yarn=owner.YarnVariant;

            trail.ResetTrail(position);Body.position=position;Body.rotation=0;Body.gravityScale=0f;Body.linearDamping=0;Body.angularDamping=0;Body.linearVelocity=flightDirection*flightSpeed;Body.angularVelocity=0;Body.WakeUp();
        }
        public void Deactivate(bool fade=false)
        {
            if(circle!=null)foreach(var c in ignored)if(c!=null)Physics2D.IgnoreCollision(circle,c,false);ignored.Clear();
            Active=false;if(trail!=null)trail.Stop(fade);
            if(Body!=null){Body.linearVelocity=Vector2.zero;Body.angularVelocity=0;Body.simulated=false;}
            gameObject.SetActive(false);
        }
        void FixedUpdate()
        {
            if(!Active||owner==null||!owner.IsPlaying||owner.IsPaused||owner.RefillFreezeBalls)return;
            for(int i=ignored.Count-1;i>=0;i--){var c=ignored[i];if(c==null){ignored.RemoveAt(i);continue;}var bounds=c.bounds;bounds.Expand(FlightGeometry.Radius*2+.03f);if(!c.enabled||!bounds.Contains(new Vector3(Body.position.x,Body.position.y,c.bounds.center.z))){Physics2D.IgnoreCollision(circle,c,false);ignored.RemoveAt(i);}}
            if(restoreStreakPierce){Body.linearVelocity=flightDirection*flightSpeed;restoreStreakPierce=false;}
            RestoreSpeed();
            SweepStreakPierce();
            Vector2 p=Body.position;
            float returnY=owner.ReturnLineY;
            if(p.y>(owner.IsChainCollection?returnY+.1f:-5.0f))enteredBoard=true;
            if((enteredBoard&&previousY>returnY&&p.y<=returnY) || p.y < -10.65f || Mathf.Abs(p.x)>6.2f || p.y>10.5f)
            { var exit=enteredBoard&&previousY>returnY&&p.y<=returnY?Vector2.Lerp(previousPosition,p,(previousY-returnY)/(previousY-p.y)):p;trail.Sample(exit);owner.LoseBall(this,exit);return; }
            owner.CheckChainTools(this,previousPosition,p);
            trail.Sample(p);previousPosition=p;previousY=p.y;
        }
        void SweepStreakPierce(){
            return;
        }
        void RestoreSpeed(Collision2D collision=null)
        {
            if(collision==null&&(Body.linearVelocity-flightDirection*flightSpeed).sqrMagnitude<.00000001f)return;
            Vector2 direction=flightDirection;
            // The solver supplies the bounce. Only recover a stopped/inward result at a contact.
            if(collision!=null)for(int i=0;i<collision.contactCount;i++){
                Vector2 normal=collision.GetContact(i).normal;
                if(Vector2.Dot(direction,normal)<-.0001f)direction=Vector2.Reflect(direction,normal);
            }
            flightDirection=direction.normalized;Body.linearVelocity=flightDirection*flightSpeed;
        }
        void OnCollisionStay2D(Collision2D collision)
        {
            if(Active&&owner!=null&&owner.IsPlaying&&!owner.IsPaused&&SessionGeneration==owner.Generation){if(StreakPower==StreakBooster.Pierce&&ignored.Contains(collision.collider))return;RestoreSpeed(collision);}
        }
        void OnTriggerEnter2D(Collider2D other)
        {
            if(Active&&enteredBoard&&Body.position.y<=owner.ReturnLineY&&Body.linearVelocity.y<0&&other.GetComponent<WaterVolume>()!=null)
                {owner.LoseBall(this,Body.position);}
        }
        
        public void QATick(){FixedUpdate();}
        
        public void PauseTrail(bool pause){Body.gravityScale=0;}
        void Update(){if(!Active||owner==null||owner.IsPaused)return;impact=Mathf.Max(0,impact-Time.deltaTime);float k=FeedbackProfile.Active.enabledFeedback?Mathf.Sin(impact/.10f*Mathf.PI)*.07f:0;visual.transform.localScale=Vector3.Scale(StreakPower==StreakBooster.None?visualBaseScale:Vector3.one*StreakBallArt.BallScale(StreakPower),new Vector3(1+k,1-k,1)*(1));if(StreakPower!=StreakBooster.None){streakArtAge+=Time.deltaTime;visual.sprite=StreakBallArt.Frame(StreakPower,4+(int)(streakArtAge*12)%4);}}

        
        
        
        


        void OnCollisionEnter2D(Collision2D collision)
        {
            if(!Active||owner==null||!owner.IsPlaying||owner.IsPaused||SessionGeneration!=owner.Generation)return;
            PlayerHaptics71.Pulse();
            var abilityBrick=collision.collider.GetComponentInParent<BrickBody>();
            IncomingDirection=flightDirection;RestoreSpeed(collision);
            impact=.10f;if(collision.contactCount>0){var c=collision.GetContact(0);trail.Sample(c.point+c.normal*FlightGeometry.Radius);}
            var brick=collision.collider.GetComponentInParent<BrickBody>();
            Vector2 point=collision.contactCount>0?collision.GetContact(0).point:Body.position;
            var jellyWall=collision.collider.GetComponent<BoardWallContact>();
            if(jellyWall!=null){float normalSpeed=Mathf.Abs(jellyWall.Side==1?IncomingDirection.y:IncomingDirection.x)*collision.relativeVelocity.magnitude;jellyWall.Hit(point,normalSpeed);owner.OnChainWall(this,jellyWall.Side);}
            if(brick!=null){{owner.HitBrick(brick,this,point);owner.TriggerStreakImpact(this,point);}return;}
            if(jellyWall!=null)owner.TriggerStreakImpact(this,point);
            var ball=collision.collider.GetComponent<BallBody>();
            if(ball!=null&&Id>ball.Id)return;
            if(Time.time-lastContactTime>.06f && collision.relativeVelocity.sqrMagnitude>1)
            {lastContactTime=Time.time;owner.EmitFeedback(point,"wall",new Color(.92f,1,.94f));}
        }
    }

        public void Remove(bool countRefill=true)
        {
            if(!Alive)return;if(countRefill&&owner!=null)owner.RecordRefillRemoval(this);Alive=false;hitCollider.enabled=false;gameObject.SetActive(false);
        }
        bool visualSettled;
        void Update()
        {
            if(!Alive||owner==null||owner.IsPaused)return;
            // ChainTileVisibility owns chain colliders. Never briefly re-enable
            // hidden supply tiles or rebuild unchanged broadphase proxies here.
            if(!owner.IsChainCollection){bool enabled=age>=delay+.28f;if(hitCollider.enabled!=enabled)hitCollider.enabled=enabled;}
            if(visualSettled&&pulse<=0)return;
            age+=Time.deltaTime;
            float t=owner.IsChainCollection?1:Mathf.Clamp01((age-delay)/.28f),s=t-1;
            float entrance=1+2.70158f*s*s*s+1.70158f*s*s;
            pulse=Mathf.Max(0,pulse-Time.deltaTime);
            float hit=pulse>0&&FeedbackProfile.Active.enabledFeedback?Mathf.Sin((1-pulse/FeedbackProfile.Active.hitDuration)*Mathf.PI)*FeedbackProfile.Active.hitScale:0;
            if(!owner.IsChainCollection&&hitCollider.enabled!=(t>=1))hitCollider.enabled=t>=1;
            float scale=Mathf.Max(0,entrance);
            visualRoot.localScale=new Vector3(scale*(1+hit),scale*(1-hit),1);
            if(round5)visualRoot.localPosition=new Vector3(0,(visualRoot.localScale.y-1)*owner.Board.CellHeight*.5f,0);
            var profile=FeedbackProfile.Active;
            float hitAge=profile.hitDuration-pulse;
            if(flash==null)flash=new MaterialPropertyBlock();
            flash.SetFloat("_Flash",pulse>0&&profile.enabledFeedback?.48f*(1-Mathf.Clamp01(hitAge/profile.flashDuration)):0);
            face.SetPropertyBlock(flash);
            foreach(var image in hitFaces)if(image!=null&&image!=face&&image.sharedMaterial!=null&&image.sharedMaterial.HasProperty("_Flash"))image.SetPropertyBlock(flash);
            if(number!=null)number.transform.localScale=Vector3.one*(1+hit*3);
            visualSettled=t>=1&&pulse<=0;
            if(visualSettled){face.SetPropertyBlock(null);enabled=false;}
        }
