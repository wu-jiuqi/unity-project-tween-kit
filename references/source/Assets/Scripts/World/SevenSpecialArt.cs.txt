public sealed class SevenSpecialIdle:MonoBehaviour {
 GameSession owner;ChainTool tool;SpriteRenderer face;float age,born,waitAt;bool waiting;int generation;
 public void Initialize(GameSession g,ChainTool t){owner=g;tool=t;generation=g.Generation;face=new GameObject("Complete baked special jelly").AddComponent<SpriteRenderer>();face.transform.SetParent(transform,false);face.sortingOrder=65;Render();}
 void Render(){if(tool.Ingot){OpeningThirdArt.Fit(face,"ingot",OpeningThirdArt.Variant(tool.Kind),owner.Board.CellWidth);face.color=Color.white;return;}face.sprite=SevenSpecialArt.Special(tool.Kind,tool.ColorIndex,(int)(age*5)%4);face.color=Color.white;SevenSpecialArt.FitBaked(face,owner.Board.CellWidth);}
 public void Wait(){waiting=true;waitAt=age;}
 public void Resume(){waiting=false;transform.localScale=Vector3.one;}
 void Update(){if(owner==null||owner.Generation!=generation){Destroy(gameObject);return;}if(owner.IsPaused)return;age+=owner.PresentationDelta(Time.deltaTime);born=Mathf.Min(1,born+owner.PresentationDelta(Time.deltaTime)/.2f);if(waiting){float u=Mathf.Clamp01((age-waitAt)/.16f);transform.localScale=Vector3.one*Mathf.Lerp(1,1.14f,1-Mathf.Pow(1-u,3));return;}Render();float grow=1-Mathf.Pow(1-born,3);transform.localScale=Vector3.one*(grow*(1+Mathf.Sin(age*3)*.015f));}
}
public sealed class SevenSpecialFx:MonoBehaviour {
 public Vector2 Origin {get;private set;}
 
 
 static readonly Dictionary<GameSession,int> counts=new Dictionary<GameSession,int>();
 GameSession owner;SpriteRenderer face;float age,duration=.66f,baseSize;int generation,kind,color,colorB,first=7,last=15;string sheet;MaterialPropertyBlock block;
 public static int Alive(GameSession g)=>counts.TryGetValue(g,out int n)?n:0;

  var go=new GameObject("Generated combination sequence "+key);go.transform.SetParent(root,false);go.transform.position=p;go.transform.rotation=Quaternion.Euler(0,0,angle);var f=go.AddComponent<SevenSpecialFx>();f.owner=g;f.block=new MaterialPropertyBlock();f.Origin=p;f.generation=g.Generation;f.sheet=key;f.color=colorA;f.colorB=colorB;f.first=0;f.last=15;f.duration=.76f;f.baseSize=size;f.face=new GameObject("Anchored effect artwork").AddComponent<SpriteRenderer>();f.face.transform.SetParent(go.transform,false);f.face.sortingOrder=79;f.face.sprite=SevenSpecialArt.Frame(key,0);SevenSpecialArt.Fit(f.face,size);f.face.color=Color.white;SevenSpecialArt.EffectColors(f.face,colorA,colorB,f.block);counts[g]=Alive(g)+1;
 }
 void Update(){if(BalanceClock.Active)return;Advance(Time.deltaTime);}
 public void Advance(float dt){if(owner==null||owner.Generation!=generation){Destroy(gameObject);return;}if(owner.IsPaused)return;age+=owner.PresentationDelta(dt);if(age>=duration){if(BalanceClock.Active){OnDestroy();owner=null;gameObject.SetActive(false);}Destroy(gameObject);return;}int frame=first+Mathf.Min(last-first,(int)(age/duration*(last-first+1)));face.sprite=sheet!=null?SevenSpecialArt.Frame(sheet,frame):SevenSpecialArt.Special(kind,color,frame);float u=age/duration;if(sheet=="flower-vortex")transform.rotation=Quaternion.Euler(0,0,-260*u);if(sheet!=null)SevenSpecialArt.Fit(face,baseSize*(1+.24f*u));else SevenSpecialArt.FitBaked(face,baseSize*(1+.24f*u));SevenSpecialArt.EffectColors(face,color,colorB,block);var c=face.color;c.a=Mathf.Clamp01((1-u)*4);face.color=c;}
 void OnDestroy(){if(!ReferenceEquals(owner,null)&&counts.TryGetValue(owner,out int n)){if(n<=1)counts.Remove(owner);else counts[owner]=n-1;}}
}
