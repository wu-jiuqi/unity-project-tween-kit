using UnityEngine;
namespace BounceGarden {
// Origin is the top-left CELL EDGE. U and V are physical world steps, not PNG size.
// Legacy authored levels have non-square pitches. Preserve those explicitly until
// a separate physics/layout calibration is authorized; never hide it by moving roots.
public sealed class BoardSpace {
 public readonly Vector2 Origin,U,V;public readonly int Columns,Rows,Phase;public readonly string Status;
 public float CellWidth=>U.magnitude;public float CellHeight=>V.magnitude;
 
 public BoardSpace(Vector2 firstCenter,Vector2 u,Vector2 v,int columns,int rows,int phase,string status){U=u;V=v;Origin=firstCenter-(u+v)*.5f;Columns=columns;Rows=rows;Phase=phase;Status=status;}
 public Vector2 CellToLocal(float col,float row)=>Origin+(col+.5f)*U+(row+.5f)*V;
 public Vector2 LocalToCell(Vector2 p)=>new Vector2((p.x-Origin.x)/U.x-.5f,(p.y-Origin.y)/V.y-.5f);
 public Vector2Int Nearest(Vector2 p){var a=LocalToCell(p);return new Vector2Int(Mathf.RoundToInt(a.x),Mathf.RoundToInt(a.y));}
 
 
 
 
 public static BoardSpace ForLevel(int index,LevelData fitted){return new BoardSpace(CollectionLevels.Cell(0,0),Vector2.right*CollectionLevels.Pitch,Vector2.down*CollectionLevels.Pitch,9,9,0,"Opening 45-stage lattice");}}
}
