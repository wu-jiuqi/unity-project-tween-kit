using System.Collections.Generic;
using UnityEngine;
namespace BounceGarden {
public static class PaintingArt {
 const string Root="Art/PaintingTrial/";
 static readonly string[] Colors={"orange","ear_rust","cream","ink"};
 static readonly Dictionary<string,Sprite> images=new Dictionary<string,Sprite>();
 static readonly Sprite[][] animations=new Sprite[4][];
 static readonly Dictionary<int,Sprite> stamps=new Dictionary<int,Sprite>();
 public static readonly Color[] Palette={new Color32(242,139,69,255),new Color32(172,73,61,255),new Color32(255,240,207,255),new Color32(73,54,77,255)};
 public static Sprite Image(string key){
  if(images.TryGetValue(key,out var sprite))return sprite;
  var texture=Resources.Load<Texture2D>(Root+key);
  if(texture==null)throw new System.InvalidOperationException("Painting art missing: "+key);
  sprite=Sprite.Create(texture,new Rect(0,0,texture.width,texture.height),new Vector2(.5f,.5f),100,0,SpriteMeshType.FullRect);sprite.name=key;images.Add(key,sprite);return sprite;
 }
 public static Sprite[] Frames(int color){
  color=Mathf.Clamp(color,0,3);if(animations[color]!=null)return animations[color];
  var texture=Resources.Load<Texture2D>(Root+Colors[color]+"_jelly_pop");
  if(texture==null)throw new System.InvalidOperationException("Painting jelly missing");
  var frames=new Sprite[16];for(int i=0;i<16;i++){frames[i]=Sprite.Create(texture,new Rect(i%4*256,(3-i/4)*256,256,256),new Vector2(.5f,.5f),100,0,SpriteMeshType.FullRect);frames[i].name=i==0?JellyBrickArt.SpriteName:"painting_pop_"+color+"_"+i;}
  animations[color]=frames;return frames;
 }
 public static Sprite Stamp(int row,int col){
  int key=row*12+col;if(stamps.TryGetValue(key,out var sprite))return sprite;
  int index=0;for(int r=0;r<12;r++)for(int c=0;c<12;c++)if(PaintingLevel.Rows[r][c]!='.'){
   if(r==row&&c==col){var t=Resources.Load<Texture2D>(Root+"paint_atlas");sprite=Sprite.Create(t,new Rect(index%10*128,(9-index/10)*128,128,128),new Vector2(.5f,.5f),100,0,SpriteMeshType.FullRect);sprite.name="paint_"+row+"_"+col;stamps.Add(key,sprite);return sprite;}index++;
  }
  throw new System.ArgumentException("No painted cell");
 }
 [System.Serializable]public class StampData{public StampOffset[] stamps;}
 [System.Serializable]public class StampOffset{public int row,col;public float x,y;}
 static Dictionary<int,Vector2> offsets;
 public static Vector2 Offset(int row,int col){if(offsets==null){offsets=new Dictionary<int,Vector2>();var data=JsonUtility.FromJson<StampData>(Resources.Load<TextAsset>(Root+"paint_offsets").text);foreach(var e in data.stamps)offsets[e.row*12+e.col]=new Vector2(e.x,-e.y)*PaintingLevel.Pitch/60;}return offsets[row*12+col];}
}
}
