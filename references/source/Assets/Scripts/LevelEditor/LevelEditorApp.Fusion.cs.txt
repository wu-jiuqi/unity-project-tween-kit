        internal static List<string> RecentFusionProgress(GameSession game)
        {
            var lines = new List<string>(); var categories = new HashSet<string>();
            for (int i = game.FusionEvents.Count - 1; i >= 0 && lines.Count < 3; i--)
            {
                var record = game.FusionEvents[i]; string category = record.Kind, label = null;
                game.fusionPieces.TryGetValue(record.TargetId ?? "", out var target);
                switch (record.Kind)
                {
                    case "ArmorRemoved": case "IceRemoved": category = "protection"; label = "保护已解除，后续命中可推进本体。"; break;
                    case "PieceRemoved": if (target?.Definition.kind == "chest") label = "箱子已打开，继续收集展开的目标。"; break;
                    case "ContentActivated": category = "output"; label = "产物已上场，仍需击中收集。"; break;
                    case "ProducerBatchReleased": category = "output"; label = "一批补给已展开，查看剩余目标。"; break;
                    case "GateOpened": label = "关联门已打开，可尝试新的入口。"; break;
                    case "CannonFired": label = "机关炮已发射，剩余弹药见炮身。"; break;
                    case "AssistantJarOpened": case "AssistantResolved": category = "assistant"; label = record.Kind == "AssistantJarOpened" ? "助手已出发，等待它推进目标。" : "助手已命中，查看剩余目标。"; break;
                    case "MilestoneClaimed": label = "收集里程碑已兑现一次助力。"; break;
                    case "BatteryDischarged": case "ToolEnabled": category = "wake"; label = "预置工具已唤醒，可继续触发接力。"; break;
                    case "GoalCredited": category = "goal"; label = record.TargetId != null && record.TargetId.Contains(":stock:") ? "出货已计入收集，无需重复击中。" : "目标已收集，完成数已更新。"; break;
                }
                // Group rapid identical credits so they do not bury the mechanism that produced them.
                if (label != null && categories.Add(category)) lines.Add("第" + Math.Max(1, record.VolleyId) + "组 · " + label);
            }
            return lines;
        }
        static string FusionNextAction(GameSession game)
        {
            if (!string.IsNullOrEmpty(game.FusionRuleError)) return "配置问题：" + game.FusionRuleError;
            if (game.FusionGoalsCredited == game.FusionGoalsTotal) return "全部目标已完成。";
            var chest = game.FusionPieces.FirstOrDefault(p => p.Alive && p.Definition.kind == "chest");
            if (chest != null) return "目标箱还在场：先解除保护或击中箱体，打开后再收集产物。";
            var cannon = game.FusionPieces.FirstOrDefault(p => p.Alive && p.Definition.kind == "cannon" && p.Stock > 0);
            if (cannon != null) return "机关炮 " + cannon.Charge + "/" + cannon.Definition.threshold + " · 余弹 " + cannon.Stock + "。" + (cannon.Definition.activationMode == "neighbor" ? "清除炮旁砖可充能。" : cannon.Definition.activationMode == "direct" ? "命中炮身可充能。" : "命中炮身或清除旁砖可充能。");
            if (game.FusionPieces.Any(p => p.Alive && (p.Ice > 0 || p.Brick != null && p.Brick.ShellIntact))) return "甲/冰会先挡住一次攻击；解除后，让后续球或独立工具继续推进。";
            if (game.fusionGoals.Values.Any(g => g.State == FusionGoalState.Reserved)) return "还有目标待展开；继续推进对应容器或生产器。";
            return "观察剩余目标与可触发工具，再选择下一次发射入口。";
        }

        void LoadFusionInspector()
        {
            if (CurrentLevel == null || !CurrentLevel.IsFusion) return;
            fusionLoadedLevel = CurrentLevel.EditorLevelId; fusionLoadedObject = selectedPiece;
            fusionOptions = null; fusionEdit = null; fusionOptionText.Clear(); fusionPieceText.Clear(); fusionContentRows.Clear(); fusionMilestoneRows.Clear();
            try
            {
                fusionOptions = CurrentLevel.FusionDefinition;
                LoadNumbers(fusionOptions, FusionOptionNumbers, fusionOptionText);
                fusionOptionText["requiredMechanisms"] = Join(fusionOptions.requiredMechanisms);
                fusionOptionText["introducedMechanisms"] = Join(fusionOptions.introducedMechanisms);
                foreach (var m in fusionOptions.milestones ?? Array.Empty<FusionMilestoneDefinition>()) if (m != null)
                    fusionMilestoneRows.Add(new FusionMilestoneRow { id = m.id, kind = m.kind, threshold = m.threshold.ToString(), count = m.count.ToString(), links = Join(m.links) });
                if (selectedPiece == null || !CurrentLevel.Pieces.Any(p => p.Id == selectedPiece)) return;
                fusionEdit = CurrentLevel.GetFusionPiece(selectedPiece);
                foreach (string key in new[] { "c", "r", "size", "hp", "ice" }) fusionPieceText[key] = ((int)typeof(FusionPieceDefinition).GetField(key).GetValue(fusionEdit) + (key == "c" || key == "r" ? 1 : 0)).ToString();
                LoadNumbers(fusionEdit, FusionPieceNumbers, fusionPieceText);
                fusionPieceText["group"] = fusionEdit.group ?? ""; fusionPieceText["network"] = fusionEdit.network ?? "";
                fusionPieceText["links"] = Join(fusionEdit.links); fusionPieceText["eligibleGoals"] = Join(fusionEdit.eligibleGoals);
                fusionPieceText["rewards"] = Join(fusionEdit.rewardSequence);
                fusionPieceText["spawn"] = CellText(fusionEdit.spawnCells); fusionPieceText["fallback"] = CellText(fusionEdit.fallbackCells);
                foreach (var c in fusionEdit.contents ?? Array.Empty<FusionContentDefinition>()) if (c != null)
                    fusionContentRows.Add(new FusionContentRow { id = c.id, kind = c.kind, goal = c.goal, hp = c.hp.ToString() });
            }
            catch (Exception ex) { SetStatus("融合草稿配置有误，可先查看检查结果：" + ex.Message, true); }
        }
        static void LoadNumbers(object value, string[,] fields, Dictionary<string, string> output)
        {
            for (int i = 0; i < fields.GetLength(0); i++) output[fields[i, 0]] = Convert.ToString(value.GetType().GetField(fields[i, 0]).GetValue(value), CultureInfo.InvariantCulture);
        }
        static void ApplyNumbers(object value, string[,] fields, Dictionary<string, string> input)
        {
            for (int i = 0; i < fields.GetLength(0); i++) value.GetType().GetField(fields[i, 0]).SetValue(value, Number(input[fields[i, 0]], fields[i, 1]));
        }
        static int Number(string text, string label)
        {
            if (!int.TryParse(text, NumberStyles.Integer, CultureInfo.InvariantCulture, out int result)) throw new ArgumentException(label + "必须填写整数。");
            return result;
        }
        static string Join(IEnumerable<string> items) => string.Join(",", items ?? Array.Empty<string>());
        static string[] SequenceTokens(string value) => (value ?? "").Split(new[] { ',', '，', ';', '；', ' ', '\n', '\r' }, StringSplitOptions.RemoveEmptyEntries).Select(t => t.Trim()).ToArray();
        static string[] Tokens(string value) => SequenceTokens(value).Distinct().ToArray();
        static string CellText(Vector2Int[] cells) => string.Join("; ", (cells ?? Array.Empty<Vector2Int>()).Select(c => (c.x + 1) + "," + (c.y + 1)));
        static Vector2Int[] ParseCells(string value)
        {
            if (string.IsNullOrWhiteSpace(value)) return Array.Empty<Vector2Int>();
            return value.Split(new[] { ';', '；', '\n' }, StringSplitOptions.RemoveEmptyEntries).Select(t =>
            {
                var xy = t.Trim().Split(new[] { ',', '，' }, StringSplitOptions.RemoveEmptyEntries);
                if (xy.Length != 2) throw new ArgumentException("出生位置按 列,行;列,行 填写，范围1–9。");
                int x = Number(xy[0], "列"), y = Number(xy[1], "行");
                if (x < 1 || x > 9 || y < 1 || y > 9) throw new ArgumentException("出生位置范围为1–9。");
                return new Vector2Int(x - 1, y - 1);
            }).ToArray();
        }

        static bool FusionNumberRelevant(string kind, string field)
        {
            if (field == "layers" || field == "hitsPerRing") return kind == "snowball" || kind == "snowdrift";
            if (field == "ripenNeeds") return kind == "strawberry";
            if (field == "hitsPerItem" || field == "bonusServings") return kind == "basket";
            if (field == "batchSize") return kind == "producer";
            if (field == "range" || field == "damage") return kind == "cannon" || kind == "vine" || kind == "assistant" || kind == "reward";
            if (field == "stock") return new[] { "producer", "cannon", "assistant", "reward", "basket" }.Contains(kind);
            return new[] { "producer", "cannon", "vine", "battery", "assistant", "reward" }.Contains(kind);
        }

        void DrawFusionOverlays()
        {
            var definition = FusionPreview(); if (definition?.pieces == null) return;
            var pieces = definition.pieces.Where(p => p != null && p.size >= 1 && p.size <= 9 && p.c >= 0 && p.r >= 0 && p.c <= 9 - p.size && p.r <= 9 - p.size).ToArray();
            foreach (var source in pieces)
            {
                foreach (var id in source.links ?? Array.Empty<string>())
                {
                    var target = pieces.FirstOrDefault(p => p.id == id); if (target == null) continue;
                    DrawFusionLine(CellRect(source.c, source.r, source.size).center, CellRect(target.c, target.r, target.size).center, source.id == selectedPiece ? Accent : new Color(.2f, .55f, .7f, .65f), source.id == selectedPiece ? 3 : 1.5f);
                }
            }
            foreach (var group in pieces.Where(p => !string.IsNullOrWhiteSpace(p.group)).GroupBy(p => p.group))
            {
                var nodes = group.OrderBy(p => p.r).ThenBy(p => p.c).ToArray();
                for (int i = 1; i < nodes.Length; i++) DrawFusionLine(CellRect(nodes[i - 1].c, nodes[i - 1].r, nodes[i - 1].size).center, CellRect(nodes[i].c, nodes[i].r, nodes[i].size).center, new Color(.3f, .8f, .35f, .7f), 2);
            }
            var selected = pieces.FirstOrDefault(p => p.id == selectedPiece); if (selected == null) return;
            foreach (var cell in selected.spawnCells ?? Array.Empty<Vector2Int>()) if (cell.x >= 0 && cell.x < 9 && cell.y >= 0 && cell.y < 9) Border(CellRect(cell.x, cell.y, 1), Accent, 3);
            foreach (var cell in selected.fallbackCells ?? Array.Empty<Vector2Int>()) if (cell.x >= 0 && cell.x < 9 && cell.y >= 0 && cell.y < 9) Border(CellRect(cell.x, cell.y, 1), Amber, 2);
            if (selected.kind == "cannon" || selected.kind == "soda" || selected.kind == "line_h" || selected.kind == "line_v" || selected.kind == "cross" || selected.kind == "bomb")
            {
                foreach (var cell in FusionEffectCells(selected, pieces, definition)) Border(CellRect(cell.x, cell.y, 1), new Color(1f, .7f, .1f, .9f), 2);
            }
        }
        static IEnumerable<Vector2Int> FusionEffectCells(FusionPieceDefinition source, FusionPieceDefinition[] pieces, FusionLevelDefinition definition)
        {
            if (source.kind == "bomb")
            {
                for (int y = source.r - 1; y <= source.r + 1; y++) for (int x = source.c - 1; x <= source.c + 1; x++) if (x >= 0 && x < 9 && y >= 0 && y < 9) yield return new Vector2Int(x, y);
                yield break;
            }
            Vector2Int[] directions = source.kind == "line_h" ? new[] { Vector2Int.left, Vector2Int.right } : source.kind == "line_v" ? new[] { Vector2Int.up, Vector2Int.down } : source.kind == "cross" ? new[] { Vector2Int.left, Vector2Int.right, Vector2Int.up, Vector2Int.down } : new[] { source.direction == "left" ? Vector2Int.left : source.direction == "right" ? Vector2Int.right : source.direction == "up" ? Vector2Int.down : Vector2Int.up };
            foreach (var direction in directions) for (int distance = 1; distance <= (source.kind == "cannon" ? source.range : 9); distance++)
            {
                var cell = new Vector2Int(source.c, source.r) + direction * distance;
                if (cell.x < 0 || cell.x >= 9 || cell.y < 0 || cell.y >= 9) break;
                yield return cell;
                var blocker = pieces.FirstOrDefault(p => cell.x >= p.c && cell.x < p.c + p.size && cell.y >= p.r && cell.y < p.r + p.size);
                if (blocker != null)
                {
                    bool waitingTool = pieces.Any(p => (p.kind == "battery" || p.kind == "reward") && (p.links ?? Array.Empty<string>()).Contains(blocker.id))
                        || (definition.milestones ?? Array.Empty<FusionMilestoneDefinition>()).Any(m => m != null && m.kind == "activate" && (m.links ?? Array.Empty<string>()).Contains(blocker.id));
                    bool barrier = waitingTool || blocker.armor || blocker.ice > 0 || new[] { "stone", "gate", "armor", "ice", "snowball", "snowdrift", "strawberry" }.Contains(blocker.kind)
                        || source.kind == "cannon" && (blocker.kind == "chest" || blocker.kind == "basket");
                    barrier |= blocker.kind == "rear" && direction != Vector2Int.up;
                    if (barrier) break;
                }
            }
        }
        static void DrawFusionLine(Vector2 from, Vector2 to, Color color, float thickness)
        {
            if (Event.current.type != EventType.Repaint) return;
            var matrix = GUI.matrix; var old = GUI.color; Vector2 delta = to - from;
            GUIUtility.RotateAroundPivot(Mathf.Atan2(delta.y, delta.x) * Mathf.Rad2Deg, from); GUI.color = color;
            GUI.DrawTexture(new Rect(from.x, from.y - thickness * .5f, delta.magnitude, thickness), Texture2D.whiteTexture);
            GUI.DrawTexture(new Rect(from.x + delta.magnitude - 7, from.y - 4, 7, 8), Texture2D.whiteTexture);
            GUI.matrix = matrix; GUI.color = old;
        }
