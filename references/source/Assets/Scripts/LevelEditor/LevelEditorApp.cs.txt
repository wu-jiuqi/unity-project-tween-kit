        public static bool HasArgument(string value) => Array.IndexOf(Environment.GetCommandLineArgs(), value) >= 0;
        public static string Argument(string key)
        {
            var args = Environment.GetCommandLineArgs(); int i = Array.IndexOf(args, key);
            return i >= 0 && i + 1 < args.Length ? args[i + 1] : null;
        }

        void Awake()
        {
            Application.targetFrameRate = 60;
            Application.runInBackground = true;
            qaMode = HasArgument("--level-editor-qa");
            rootDirectory = Path.Combine(Environment.GetFolderPath(Environment.SpecialFolder.MyDocuments), "BounceBustersLevelEditor");
            if (qaMode) rootDirectory = Argument("--level-editor-qa-out") ?? Path.Combine(Path.GetTempPath(), "BounceBustersLevelEditorQA");
            workspaceDirectory = Argument("--level-editor-workspace") ?? Path.Combine(rootDirectory, "Projects", "MyLevels");
            if (!qaMode && Argument("--level-editor-workspace") == null)
            {
                try { var path = Path.Combine(rootDirectory, "authoring-settings.json"); if (File.Exists(path)) { var settings = JsonUtility.FromJson<WorkspaceSettings>(File.ReadAllText(path)); if (!string.IsNullOrWhiteSpace(settings.workspace)) workspaceDirectory = settings.workspace; } }
                catch (Exception ex) { Debug.LogWarning("Level editor workspace preference: " + ex.Message); }
            }
            exportParent = Path.Combine(rootDirectory, "Exports");
            Playtest = gameObject.AddComponent<LevelEditorPlaytest>();
            Playtest.Completed += OnCompleted;
            try
            {
                Directory.CreateDirectory(rootDirectory);
                bool anyCampaign = Directory.Exists(workspaceDirectory) && Directory.GetFiles(workspaceDirectory, "campaign.json", SearchOption.AllDirectories).Length > 0;
                Workspace = anyCampaign ? LevelEditorWorkspace.LoadDirectory(workspaceDirectory) : LevelEditorWorkspace.CreateEmpty();
                if (anyCampaign && string.IsNullOrEmpty(Workspace.LastSaveDirectory)) workspaceDirectory = Path.Combine(rootDirectory, "Imports", DateTime.Now.ToString("yyyyMMdd-HHmmss-fff"));
                if (!anyCampaign || string.IsNullOrEmpty(Workspace.LastSaveDirectory))
                {
                    var saved = Workspace.SaveWorkspace(workspaceDirectory);
                    if (!saved.Success) throw new IOException(saved.Message);
                }
                loaded = true; SelectLevel(0);
                RememberWorkspace();
                SetStatus(Workspace.Levels.Count == 0 ? "新项目已就绪 · 选择规则模板，点击「＋ 新建关卡」从空棋盘开始" : "已载入 " + Workspace.Levels.Count + " 关 · 草稿每 15 秒自动保存 · " + workspaceDirectory);
            }
            catch (Exception ex) { SetStatus("工作区载入失败：" + ex.Message, true); }
            nextSave = Time.unscaledTime + 15;
            Application.wantsToQuit += CanQuit;
        }

        void OnDestroy()
        {
            FinishBoardGesture();
            Application.wantsToQuit -= CanQuit;
            foreach (var texture in styleTextures) if (texture != null) Destroy(texture);
        }

        bool CanQuit()
        {
            FinishBoardGesture();
            if (qaMode || !loaded || closeRequested) return true;
            if (IsPlaytesting) WriteRecord(false, false);
            if (!Workspace.IsDirty) return true;
            if (SaveWorkspace()) return true;
            SetStatus("退出前保存失败。请先解决工作区冲突，或导出到其他目录后再退出。", true);
            return false;
        }

        void Update()
        {
            // A release outside the native window may not deliver a GUI MouseUp.
            if (gestureLevel != null && !Input.GetMouseButton(0) && !Input.GetMouseButtonUp(0)) FinishBoardGesture();
            if (loaded && !qaMode && !IsPlaytesting && gestureLevel == null && Time.unscaledTime >= nextSave)
            {
                nextSave = Time.unscaledTime + 15;
                if (Workspace.IsDirty) SaveWorkspace();
            }
            if (IsPlaytesting && Camera.main != null)
            {
                var rect = previewRect;
                Camera.main.rect = new Rect(rect.x * scale / Screen.width, 1 - (rect.y + rect.height) * scale / Screen.height,
                    rect.width * scale / Screen.width, rect.height * scale / Screen.height);
            }
        }

        public void SelectLevel(int index)
        {
            if (Workspace == null || IsPlaytesting) return;
            FinishBoardGesture();
            selectedLevel = index < 0 || Workspace.Levels.Count == 0 ? -1 : Mathf.Clamp(index, 0, Workspace.Levels.Count - 1);
            if (CurrentLevel != null) selectedChapter = Array.IndexOf(Chapters, CurrentLevel.Chapter);
            selectedPiece = null; paintKind = CurrentLevel != null && CurrentLevel.IsFusion ? "normal" : "N"; fusionLoadedLevel = null; copiedPiece = null; toolMode = 0;
            editName = CurrentLevel?.Name ?? ""; movesText = (CurrentLevel?.Moves ?? 20).ToString(); inspectorScroll = Vector2.zero; RefreshValidation();
        }

        void SelectChapter(int chapter)
        {
            FinishBoardGesture(); selectedChapter = chapter;
            var first = Workspace.Levels.FirstOrDefault(l => l.Chapter == Chapters[chapter]);
            SelectLevel(first == null ? -1 : first.Index); levelScroll = Vector2.zero;
        }

        public bool CreateLevel()
        {
            if (Workspace == null || IsPlaytesting) return false;
            FinishBoardGesture();
            try
            {
                var level = Workspace.CreateLevel(Chapters[selectedChapter], "新关卡 " + (Workspace.Levels.Count + 1));
                search = ""; SelectLevel(level.Index); toolMode = 1; levelScroll = new Vector2(0, float.MaxValue);
                SetStatus("已新建空白关卡 · 拖动画笔布置砖块，或先补满普通砖 · Ctrl S 保存"); return true;
            }
            catch (Exception ex) { SetStatus("新建失败：" + ex.Message, true); return false; }
        }

        void DuplicateLevel()
        {
            if (CurrentLevel == null) return; FinishBoardGesture();
            try { var level = Workspace.DuplicateLevel(CurrentLevel); search = ""; SelectLevel(level.Index); SetStatus("已创建独立副本，可以分别编辑。"); }
            catch (Exception ex) { SetStatus("复制失败：" + ex.Message, true); }
        }

        public bool CreateProject(string directory)
        {
            FinishBoardGesture();
            if (loaded && !SaveWorkspace()) return false;
            try
            {
                if (Directory.Exists(directory) && Directory.GetFileSystemEntries(directory).Length > 0) throw new IOException("新项目目录须为空，请另选目录。");
                var next = LevelEditorWorkspace.CreateEmpty(); var result = next.SaveWorkspace(directory);
                if (!result.Success) { SetStatus(result.Message, true); return false; }
                Workspace = next; workspaceDirectory = directory; loaded = true; selectedChapter = 0; search = ""; showIssues = false; confirmAction = null; SelectLevel(-1); RememberWorkspace();
                SetStatus("已创建空白项目 · 选择规则模板并新建关卡 · " + directory); return true;
            }
            catch (Exception ex) { SetStatus("新建项目失败：" + ex.Message, true); return false; }
        }

        public bool SaveWorkspace()
        {
            if (Workspace == null) return false;
            FinishBoardGesture();
            try { var result = Workspace.SaveWorkspace(workspaceDirectory); SetStatus(result.Message, !result.Success); if (result.Success) SetStatus("草稿已保存 · " + DateTime.Now.ToString("HH:mm:ss") + " · " + workspaceDirectory); return result.Success; }
            catch (Exception ex) { SetStatus("保存失败：" + ex.Message, true); return false; }
        }

        public bool ExportTo(string directory)
        {
            if (Workspace == null) return false;
            FinishBoardGesture();
            try
            {
                var result = Workspace.Export(directory); SetStatus(result.Message, !result.Success);
                if (result.Success) { LastExportDirectory = directory; SetStatus("导出完成：" + directory); }
                else { issues = result.Issues; issueScroll = Vector2.zero; showIssues = true; }
                return result.Success;
            }
            catch (Exception ex) { SetStatus("导出失败：" + ex.Message, true); return false; }
        }

        public bool BeginPlaytest()
        {
            if (CurrentLevel == null || IsPlaytesting) return false;
            FinishBoardGesture();
            RefreshValidation();
            if (issues.Any(p => p.IsError)) { SetStatus("请先修复当前关卡的校验错误，再开始试玩。", true); return false; }
            if (!SaveWorkspace()) return false;
            string snapshot = CurrentLevel.SnapshotJson();
            if (!Playtest.StartDraft(CurrentLevel, Scenarios[scenario])) { SetStatus(Playtest.LastError, true); return false; }
            IsPlaytesting = true; aiming = false; runStart = Time.unscaledTime; recordWritten = false;
            fusionPlayScroll = Vector2.zero;
            shotRecords.Clear(); runScenario = Scenarios[scenario];
            using (var hash = SHA256.Create()) runHash = BitConverter.ToString(hash.ComputeHash(Encoding.UTF8.GetBytes(snapshot))).Replace("-", "").ToLowerInvariant();
            Playtest.Session.VolleyStarted -= OnVolley; Playtest.Session.VolleyStarted += OnVolley;
            if (Camera.main != null) Camera.main.enabled = true;
            SetStatus("真实试玩 · 按住棋盘瞄准，松开发射 · 道具需手动选择 · Esc 返回编辑");
            return true;
        }

        void RestartPlaytest()
        {
            WriteRecord(false, false); ReturnToEditor(); BeginPlaytest();
        }

        public void ReturnToEditor()
        {
            if (!IsPlaytesting) return;
            WriteRecord(false, false);
            Playtest.Session.VolleyStarted -= OnVolley;
            Playtest.Stop(); IsPlaytesting = false; aiming = false;
            if (Camera.main != null) Camera.main.rect = new Rect(0, 0, 1, 1);
            SetStatus("已返回编辑 · 棋盘修改保留，试玩状态已清理");
        }

        void OnVolley(StreakBooster booster)
        {
            shotRecords.Add(new ShotRecord { elapsedSeconds = Time.unscaledTime - runStart, booster = booster.ToString(),
                remainingShots = Playtest.Session.Shots, hits = Playtest.Session.HitCount, breaks = Playtest.Session.BreakCount,
                collected = (int[])Playtest.Session.Collected.Clone(), jellyCollected = (int[])Playtest.Session.JellyCollected.Clone() });
        }

        void OnCompleted(LevelEditorPlaytestResult result)
        {
            WriteRecord(true, result.Won);
            SetStatus(result.Won ? "试玩通关 · 结果已保存，可返回继续编辑" : "试玩结束 · 结果已保存，可重开比较");
        }

        void WriteRecord(bool completed, bool won)
        {
            if (recordWritten || !IsPlaytesting || Playtest.Session == null) return;
            try
            {
                var g = Playtest.Session;
                var report = new RunRecord { level = selectedLevel + 1, editorLevelId = CurrentLevel.EditorLevelId, chapter = CurrentLevel.Chapter, runtimeIndex = CurrentLevel.RuntimeIndex, draftSha256 = runHash, scenario = runScenario, rulesProfile = CurrentLevel.RulesProfile, fusionRuleVersion = CurrentLevel.IsFusion ? CurrentLevel.FusionDefinition.ruleVersion : 0,
                    utc = DateTime.UtcNow.ToString("o"), unityVersion = Application.unityVersion, gameVersion = GameVersion.Name, seed = Playtest.Seed, completed = completed, won = won,
                    elapsedSeconds = Playtest.ElapsedSeconds, shotsUsed = g.CurrentLevel.initialShots - g.Shots,
                    remainingShots = g.Shots, score = g.Score, hits = g.HitCount, breaks = g.BreakCount,
                    specialActivations = g.ChainActivationCount, boosterExplosions = g.StreakExplosions, boosterLasers = g.StreakLasers,
                    volleys = shotRecords.ToArray(), fusion = g.FusionActive ? g.FusionMetrics : null, fusionEvents = g.FusionActive ? g.FusionEvents.ToArray() : null };
                string directory = Path.Combine(rootDirectory, "Playtests"); Directory.CreateDirectory(directory);
                File.WriteAllText(Path.Combine(directory, DateTime.Now.ToString("yyyyMMdd-HHmmss-fff") + "-L" + (selectedLevel + 1) + ".json"), JsonUtility.ToJson(report, true), new UTF8Encoding(false));
                recordWritten = true;
            }
            catch (Exception ex) { SetStatus("试玩记录保存失败：" + ex.Message, true); }
        }

        [Serializable] sealed class ShotRecord { public float elapsedSeconds; public string booster; public int remainingShots, hits, breaks; public int[] collected, jellyCollected; }
        [Serializable] sealed class RunRecord
        {
            public int level, runtimeIndex, shotsUsed, remainingShots, score, hits, breaks, specialActivations, boosterExplosions, boosterLasers, seed;
            public string editorLevelId, chapter, draftSha256, scenario, utc, unityVersion, gameVersion, rulesProfile; public int fusionRuleVersion; public bool completed, won; public float elapsedSeconds; public ShotRecord[] volleys;
            public FusionMetrics fusion; public FusionEventRecord[] fusionEvents;
        }

        [Serializable] sealed class WorkspaceSettings { public string workspace; }
        void RememberWorkspace()
        {
            if (qaMode || Argument("--level-editor-workspace") != null) return;
            try { File.WriteAllText(Path.Combine(rootDirectory, "authoring-settings.json"), JsonUtility.ToJson(new WorkspaceSettings { workspace = workspaceDirectory }, true), new UTF8Encoding(false)); }
            catch (Exception ex) { Debug.LogWarning("Could not remember workspace: " + ex.Message); }
        }

        void OnGUI()
        {
            EnsureStyles();
            scale = Mathf.Min(Screen.width / 1440f, Screen.height / 900f);
            scale = Mathf.Max(.3f, scale); width = Screen.width / scale; height = Screen.height / scale;
            GUI.matrix = Matrix4x4.TRS(Vector3.zero, Quaternion.identity, Vector3.one * scale);
            boardControlId = GUIUtility.GetControlID(0x424253, FocusType.Passive);
            HandleBoardGesture();
            // Modal input never reaches the canvas or controls behind the dialog.
            if ((folderMode != null || showIssues || confirmAction != null || showFusionSamples) && Event.current.type != EventType.Repaint && Event.current.type != EventType.Layout)
            {
                if (showFusionSamples) DrawFusionSamples(); else if (folderMode != null) DrawFolderDialog(); else if (confirmAction != null) DrawConfirmDialog(); else DrawIssuesDialog();
                return;
            }
            if (!IsPlaytesting) Fill(new Rect(0, 0, width, height), Background);
            else
            {
                float ph = height - 173, pw = ph * 1080 / 2340;
                previewRect = new Rect((width - pw) * .5f, 108, pw, ph);
                Fill(new Rect(0, 0, previewRect.x, height), Background);
                Fill(new Rect(previewRect.xMax, 0, width - previewRect.xMax, height), Background);
                Fill(new Rect(previewRect.x, 0, previewRect.width, previewRect.y), Background);
                Fill(new Rect(previewRect.x, previewRect.yMax, previewRect.width, height - previewRect.yMax), Background);
            }
            DrawHeader();
            if (!loaded)
            {
                GUI.Label(new Rect(36, 160, width - 72, 100), status, heading);

            }
            else if (IsPlaytesting) DrawPlaytest();
            else { DrawSidebar(); if (CurrentLevel != null) { DrawBoard(); DrawInspector(); DrawFooter(); } else DrawEmptyProject(); }
            Fill(new Rect(0, height - 37, width, 37), Hex(0x0B1420));
            GUI.Label(new Rect(18, height - 31, width - 36, 26), status, statusError ? ErrorStyle() : small);
            if (showFusionSamples) DrawFusionSamples();
            else if (folderMode != null) DrawFolderDialog();
            else if (confirmAction != null) DrawConfirmDialog();
            else if (showIssues) DrawIssuesDialog();
            else HandleShortcuts();
        }

                if (Tab(new Rect((i % 2) * 92, i / 2 * 39, 88, 34), KindName(kind), toolMode == 1 && paintKind == kind)) { FinishBoardGesture(); paintKind = kind; toolMode = 1; copiedPiece = null; GUI.FocusControl(null); }

        void DrawBoard()
        {
            float left = Sidebar + 44, right = width - InspectorWidth - 30, centerWidth = right - left;
            GUI.Label(new Rect(left, 105, centerWidth - 145, 32), "第 " + (selectedLevel + 1).ToString("00") + " 关  ·  " + (CurrentLevel.IsFusion ? "融合规则" : ChapterNames[selectedChapter]), heading);

            if (Btn(new Rect(right - 68, 103, 66, 33), "重做")) ApplyResult(CurrentLevel.Redo());

        void BeginBoardGesture(Vector2 pointer, int col, int row)
        {
            gestureLevel = CurrentLevel; gestureMode = toolMode; gestureStart = pointer;
            gestureScreenWidth = Screen.width; gestureScreenHeight = Screen.height;
            gestureControlId = boardControlId; GUIUtility.hotControl = gestureControlId;
            strokeVisited.Clear(); previousStrokePoint = null;
            if (gestureMode == 1 || gestureMode == 2)
            {
                gestureLevel.BeginEditGroup(); ContinueBoardStroke(pointer);
            }
            else
            {
                var p = gestureLevel.PieceAt(col, row); movingPiece = selectedPiece = p?.Id;
                moveOffset = p == null ? Vector2Int.zero : new Vector2Int(col - p.Col, row - p.Row); LoadInspector();
            }
        }

        void HandleBoardGesture()
        {
            if (gestureLevel == null) return;
            if (IsPlaytesting || folderMode != null || showIssues || confirmAction != null || showFusionSamples || GUIUtility.hotControl != gestureControlId
                || Screen.width != gestureScreenWidth || Screen.height != gestureScreenHeight)
            { FinishBoardGesture(); return; }
            var e = Event.current;
            if (e.button != 0) return;
            if (e.type == EventType.MouseDrag)
            {
                if (gestureMode == 1 || gestureMode == 2) ContinueBoardStroke(e.mousePosition);
                e.Use();
            }
            else if (e.type == EventType.MouseUp)
            {
                if (gestureMode == 1 || gestureMode == 2) ContinueBoardStroke(e.mousePosition);
                else if (movingPiece != null && boardRect.Contains(e.mousePosition) && (e.mousePosition - gestureStart).sqrMagnitude * scale * scale > 16)
                {
                    var cell = GridPoint(e.mousePosition);
                    int col = Mathf.FloorToInt(cell.x) - moveOffset.x, row = Mathf.FloorToInt(cell.y) - moveOffset.y;
                    ApplyResult(gestureLevel.TryMove(movingPiece, col, row));
                }
                FinishBoardGesture(); e.Use();
            }
        }

        Vector2 GridPoint(Vector2 pointer) => (pointer - boardRect.position) / (boardRect.width / 9);
        void ContinueBoardStroke(Vector2 pointer)
        {
            bool inside = boardRect.Contains(pointer);
            Vector2 current = GridPoint(pointer);
            if (previousStrokePoint.HasValue || inside)
                foreach (var cell in LevelEditorStrokePath.Cells(previousStrokePoint ?? current, current))
                {
                    if (!strokeVisited.Add(cell)) continue;
                    if (gestureMode == 1) Paint(cell.x, cell.y);
                    else
                    {
                        var p = gestureLevel.PieceAt(cell.x, cell.y);
                        if (p != null) { ApplyResult(gestureLevel.TryDelete(p.Id)); selectedPiece = null; }
                    }
                }
            // Resume at the re-entry point, never join a route taken outside the board.
            previousStrokePoint = inside ? current : (Vector2?)null;
        }

        void FinishBoardGesture()
        {
            if (gestureLevel == null) return;
            var level = gestureLevel; gestureLevel = null;
            if (level.IsEditGroupActive) level.EndEditGroup();
            if (GUIUtility.hotControl == gestureControlId) GUIUtility.hotControl = 0;
            gestureControlId = 0; movingPiece = null; previousStrokePoint = null; strokeVisited.Clear();
            RefreshValidation();
        }

        void OnApplicationFocus(bool focused) { if (!focused) FinishBoardGesture(); }
        void OnApplicationPause(bool paused) { if (paused) FinishBoardGesture(); }
        void OnDisable() { FinishBoardGesture(); }

        void HandlePlayInput()
        {
            if (folderMode != null || !Playtest.Session.IsPlaying || Playtest.Session.IsPaused) return;
            var e = Event.current;
            if (e.type == EventType.MouseDown && e.button == 0 && previewRect.Contains(e.mousePosition)) { aiming = true; pointerStart = e.mousePosition * scale; e.Use(); }
            if (aiming && (e.type == EventType.MouseDrag || e.type == EventType.MouseDown || e.type == EventType.Repaint))
            {
                Playtest.SetAim(AimDirection(e.mousePosition));
            }
            if (aiming && e.type == EventType.MouseUp)
            {
                aiming = false;
                if (previewRect.Contains(e.mousePosition) && (e.mousePosition * scale - pointerStart).sqrMagnitude > 36)
                {
                    Playtest.Shoot(AimDirection(e.mousePosition));
                }
                Playtest.SetAim(null); e.Use();
            }
        }

        Vector2 AimDirection(Vector2 pointer)
        {
            var p = new Vector3(pointer.x * scale, Screen.height - pointer.y * scale, 10);
            var d = BoardDisplay.Inverse(Camera.main.ScreenToWorldPoint(p)) - Playtest.Session.LauncherPosition;
            if (d.y < .2f) d.y = .2f;
            return d.sqrMagnitude < .01f ? Vector2.up : d.normalized;
        }

            {
                var values = contentsText.Split(new[] { ',', '，', ' ' }, StringSplitOptions.RemoveEmptyEntries); var contents = new List<int>();
                foreach (string v in values) { if (!int.TryParse(v, out int n) || n < 0 || n > 2) { SetStatus("箱内内容填写4个0/1/2，以逗号分隔。", true); return; } contents.Add(n); }
                if (contents.Count != 4) { SetStatus("箱内必须恰好4项。", true); return; } edit.Contents = contents.ToArray(); edit.Mode = chestMode;
            }
            if (selectedChapter == 2)
            {
                if (piece.Kind == "O")
                {
                    if (!int.TryParse(sizeText, out int s) || !int.TryParse(ringText, out int ring)) { SetStatus("雪球参数需填写整数。", true); return; } edit.Size = s; edit.HitsPerRing = ring;
                }
                else if (piece.Kind == "S") { if (!int.TryParse(layersText, out int l)) { SetStatus("雪层需填写整数。", true); return; } edit.Layers = l; }
                else if (piece.Kind != "X") { if (!int.TryParse(iceText, out int ice)) { SetStatus("冰层需填写整数。", true); return; } edit.Ice = ice; }
            }
            if (selectedChapter == 1 && piece.Kind == "D") edit.Direction = direction;
            if (selectedChapter == 1 && piece.Kind == "T") { if (!int.TryParse(ripenText, out int n)) { SetStatus("成熟条件需填写整数。", true); return; } edit.RipenNeeds = n; }
            ApplyResult(CurrentLevel.TryUpdate(piece.Id, edit));
        }

        LevelEditorPiece Selected() => CurrentLevel?.Pieces.FirstOrDefault(p => p.Id == selectedPiece);
        void LoadInspector()
        {
            if (CurrentLevel != null && CurrentLevel.IsFusion) { LoadFusionInspector(); return; }
            var p = Selected(); if (p == null) return;
            hpText = p.Hp.ToString(); colText = (p.Col + 1).ToString(); rowText = (p.Row + 1).ToString(); sizeText = p.Size.ToString();
            armor = p.Armor; iceText = p.Ice.ToString(); layersText = p.Layers.ToString(); ringText = p.HitsPerRing.ToString(); ripenText = p.RipenNeeds.ToString();
            chestMode = string.IsNullOrEmpty(p.Mode) ? "plain" : p.Mode; direction = string.IsNullOrEmpty(p.Direction) ? "up" : p.Direction;
            contentsText = p.Contents == null ? "1,0,2,0" : string.Join(",", p.Contents);
        }

        void ApplyResult(OperationResult result)
        {
            if (result == null) return; SetStatus(result.Message, !result.Success);
            if (result.Success) { if (!string.IsNullOrEmpty(result.PieceId)) selectedPiece = result.PieceId; if (CurrentLevel != null) { editName = CurrentLevel.Name; movesText = CurrentLevel.Moves.ToString(); } RefreshValidation(); LoadInspector(); }
        }
        void RefreshValidation()
        {
            var current = CurrentLevel == null ? new List<ValidationIssue>() : CurrentLevel.Validate().ToList();
            if (Workspace != null && Workspace.NeedsRulesReview) current.Add(new ValidationIssue { Severity = ValidationSeverity.Warning, Code = "RULES_VERSION", LevelNumber = 0,
                Message = "项目规则 " + Workspace.SavedRulesVersion + " → 当前 " + LevelEditorRuleCatalog.RulesVersion + "，请重新试玩验证。" });
            issues = current.AsReadOnly();
        }
        void SetStatus(string value, bool error = false) { status = value ?? ""; statusError = error; }

        void HandleShortcuts()
        {
            var e = Event.current; if (e.type != EventType.KeyDown) return;
            if (e.keyCode == KeyCode.Escape || e.keyCode == KeyCode.Delete || e.control) FinishBoardGesture();
            if (e.keyCode == KeyCode.Escape) { if (IsPlaytesting) ReturnToEditor(); else { toolMode = 0; copiedPiece = null; GUI.FocusControl(null); } e.Use(); return; }
            if (IsPlaytesting) return;
            if (e.control && e.keyCode == KeyCode.S) { SaveWorkspace(); e.Use(); return; }
            if (CurrentLevel == null) return;
            bool textEditing = !string.IsNullOrEmpty(GUI.GetNameOfFocusedControl());
            if (e.control && e.keyCode == KeyCode.S) { SaveWorkspace(); e.Use(); }
            else if (!textEditing && e.control && e.keyCode == KeyCode.Z) { ApplyResult(e.shift ? CurrentLevel.Redo() : CurrentLevel.Undo()); e.Use(); }
            else if (!textEditing && e.control && e.keyCode == KeyCode.Y) { ApplyResult(CurrentLevel.Redo()); e.Use(); }
            else if (!textEditing && e.control && e.keyCode == KeyCode.C) { CopySelection(); e.Use(); }
            else if (!textEditing && e.control && e.keyCode == KeyCode.V && copiedPiece != null && copiedChapter == CurrentLevel.Chapter) { toolMode = 1; paintKind = copiedPiece.Kind; e.Use(); }
            else if (!textEditing && e.keyCode == KeyCode.Delete && Selected() != null) { ApplyResult(CurrentLevel.TryDelete(selectedPiece)); selectedPiece = null; e.Use(); }
        }

                catch (Exception ex) { folderError = ex.Message; }
