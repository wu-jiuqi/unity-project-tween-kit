namespace BounceGarden {
// Predictable authored waves: learn, build pressure, breathe, then a higher plateau.
public static class ChainCollectionBalance {
 
 public const int JarHealth=7,BoxHealth=6,ArmorHealth=8,RearHealth=1;
 public static bool IsBreather(int round)=>round>=6&&round%4==2;
 public static int Tier(int round)=>round<=2||IsBreather(round)?0:round<=5?1:2;
 
 
 
 
 public static int OrdinaryHealth(int round,int col)=>Tier(round)==0?4:Tier(round)==1?((col+round)%4==0?6:5):((col+round)%2==0?7:6);
 
}
}
