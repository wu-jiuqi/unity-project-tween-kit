public static class OpeningCampaign {
 static OpeningData data;
 public const int Count=45;
 public static OpeningTarget[] Targets(OpeningStage s){if(s.goal.kind=="multi")return s.goal.targets??Array.Empty<OpeningTarget>();var list=new List<OpeningTarget>();if(s.goal.kind=="ordinary_color")list.Add(new OpeningTarget{kind="ordinary_color",color=s.goal.color,count=s.goal.count});else{if(s.goal.count>0)list.Add(new OpeningTarget{kind="yarn",count=s.goal.count});if(s.goal.cookies>0)list.Add(new OpeningTarget{kind="cookie",count=s.goal.cookies});}return list.ToArray();}
 public static bool Contains(int index)=>index>=0&&index<Count;
 public static OpeningData Data {get {if(data==null){var asset=Resources.Load<TextAsset>("Opening/levels");if(asset==null)throw new InvalidOperationException("Missing opening campaign");data=JsonUtility.FromJson<OpeningData>(asset.text);if(data.levels==null||data.levels.Length!=Count)throw new InvalidOperationException("Invalid opening campaign");}return data;}}
 public static OpeningStage Stage(int index)=>Data.levels[index];
 public static int Color(string code)=>code=="P"?0:code=="Y"?1:code=="M"?2:code=="B"?3:code=="L"?4:throw new InvalidOperationException("Unknown five-colour code "+code);
 public static AlphabetStage Metadata(int index){var s=Stage(index);return new AlphabetStage{letter=(index+1).ToString(),name=s.name,lesson=s.focus,moves=s.moves,pressure=index<4?1:1.3f,glyph=Enumerable.Repeat("00000",7).ToArray(),cells=Array.Empty<AlphabetCell>(),chests=Array.Empty<AlphabetChest>()};}
 public static bool Allows(int index,int kind)=>Contains(index)&&SevenSpecialRules.IsSpecial(kind);
 public static int ToolKind(string kind){switch(kind){case "bomb":return SevenSpecialRules.Bomb;case "horizontal":return SevenSpecialRules.Horizontal;case "vertical":return SevenSpecialRules.Vertical;case "cross":return SevenSpecialRules.Cross;case "rainbow":return SevenSpecialRules.Rainbow;default:throw new InvalidOperationException("Unknown opening special "+kind);}}
 public static void LoadBalanceData(string json){if(!RuntimeQA.IsIsolated)throw new InvalidOperationException("Balance data requires isolation");var previous=data;try{data=JsonUtility.FromJson<OpeningData>(json);if(data?.levels==null||data.levels.Length!=Count)throw new InvalidOperationException("Need 45 stages");Validate();}catch{data=previous;throw;}}
 public static void ValidateHealth(OpeningStage s){
  if(s.health!=null&&(s.health.ordinaryHp<1||s.health.ordinaryHp>99||s.health.refillHp<1||s.health.refillHp>99))throw new InvalidOperationException("Opening HP must be 1-99");
  var cells=new HashSet<int>();foreach(var p in s.hpOverrides??Array.Empty<OpeningHpOverride>()){if(p.c<0||p.c>8||p.r<0||p.r>8||p.hp<1||p.hp>99||!cells.Add(p.r*9+p.c)||s.rows[p.r][p.c]=='.'||s.pieces.Any(v=>v.kind!="ingot"&&p.c>=v.c&&p.c<v.c+(v.kind=="cookie_box"?2:1)&&p.r>=v.r&&p.r<v.r+(v.kind=="cookie_box"?2:1)))throw new InvalidOperationException("Invalid ordinary HP override");}
 }
 public static int InitialHp(OpeningStage s,int c,int r){var p=s.hpOverrides?.FirstOrDefault(v=>v.c==c&&v.r==r);return p?.hp??s.health?.ordinaryHp??1;}
 public static int Validate(){
  int checks=0;for(int i=0;i<Count;i++){
   var s=Stage(i);if(s.id!=i+1||s.moves<=0||s.rows.Length!=9||s.rows.Any(r=>r.Length!=9||r.Any(c=>"PYMBL.".IndexOf(c)<0)))throw new InvalidOperationException("Opening grid "+i);
   ValidateHealth(s);var positions=new HashSet<int>();foreach(var p in s.pieces){
    int size=p.kind=="cookie_box"?2:1;
    if(p.c<0||p.c+size>9||p.r<0||p.r+size>9)throw new InvalidOperationException("Opening piece bounds");
    for(int r=p.r;r<p.r+size;r++)for(int c=p.c;c<p.c+size;c++)if(s.rows[r][c]=='.'||!positions.Add(r*9+c))throw new InvalidOperationException("Opening overlap");
    if(p.kind=="yarn"){if(p.layers<1||p.layers>3)throw new InvalidOperationException("Yarn layers");}
    else if(p.kind=="cookie_box"){if(p.size!=2||p.capacity<1||p.capacity>8)throw new InvalidOperationException("Cookie box size or capacity");}
    else if(p.kind=="lucky_cat"){if(p.charge!=0)throw new InvalidOperationException("Cat starts charged");}
    else if(p.kind=="ingot"){if(s.rows[p.r][p.c]!='Y')throw new InvalidOperationException("Ingot palette");}
    else if(p.kind=="copper_coin"){if(i<30||p.hp<1||p.hp>OpeningCopperArt.MaxHealth)throw new InvalidOperationException("Copper HP or group");}
    else if(!Allows(i,ToolKind(p.kind)))throw new InvalidOperationException("Opening special");checks++;
   }
   var goals=Targets(s);if(goals.Length==0)throw new InvalidOperationException("Missing goals");var keys=new HashSet<string>();foreach(var t in goals){if(t.count<=0||!keys.Add(t.kind+"/"+t.color))throw new InvalidOperationException("Invalid goal count or duplicate");if(t.kind=="ordinary_color")Color(t.color);else if(t.kind!="yarn"&&t.kind!="cookie"&&t.kind!="ingot"&&t.kind!="copper_coin")ToolKind(t.kind);}checks++;
  }return checks;
 }
}
