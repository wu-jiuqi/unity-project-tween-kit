 internal float FusionRayLimit(Vector2 origin,Vector2 direction,int range){
  float limit=range*CollectionLevels.Pitch,half=CollectionLevels.Pitch*.5f;
  if(direction.x>.001f)limit=Mathf.Min(limit,(ChainGridRules.Center(8,0).x+half-origin.x)/direction.x);
  if(direction.x<-.001f)limit=Mathf.Min(limit,(ChainGridRules.Center(0,0).x-half-origin.x)/direction.x);
  if(direction.y>.001f)limit=Mathf.Min(limit,(CollectionLevels.Top+half-origin.y)/direction.y);
  if(direction.y<-.001f)limit=Mathf.Min(limit,(ChainGridRules.Center(0,8).y-half-origin.y)/direction.y);
  return Mathf.Max(0,limit);
 }
 static bool FusionSnapshotRayEntry(Vector2 center,int size,Vector2 origin,Vector2 direction,float limit,out float distance,out Vector2 point){
  float half=(CollectionLevels.Pitch*size-.06f)*.5f,enter=0,leave=limit;Vector2 offset=origin-center;
  for(int axis=0;axis<2;axis++){
   if(Mathf.Abs(direction[axis])<.00001f){if(Mathf.Abs(offset[axis])>half){distance=0;point=origin;return false;}continue;}
   float a=(-half-offset[axis])/direction[axis],b=(half-offset[axis])/direction[axis];enter=Mathf.Max(enter,Mathf.Min(a,b));leave=Mathf.Min(leave,Mathf.Max(a,b));if(enter>leave){distance=0;point=origin;return false;}
  }distance=enter;point=origin+direction*enter;return true;
 }
