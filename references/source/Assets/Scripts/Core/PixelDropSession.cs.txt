using System.Collections.Generic;
using UnityEngine;
namespace BounceGarden {
public sealed partial class GameSession {
 public bool IsPixelDrop=>CurrentLevel!=null&&CurrentLevel.pixelDrop;
 public readonly List<PixelDropBody> DropBricks=new List<PixelDropBody>();
 public bool DropSettling {get{if(!IsPixelDrop)return false;foreach(var b in DropBricks)if(b!=null&&b.Brick.Alive&&b.Moving)return true;return false;}}
 void TickPixelDrop(){if(!IsPixelDrop)return;foreach(var b in DropBricks)if(b!=null&&b.Brick.Alive)b.Tick(Time.fixedDeltaTime);}
 void HitPixelDrop(BrickBody brick,BallBody ball,Vector2 contact){var b=brick.GetComponent<PixelDropBody>();if(b==null||!b.BallHit(ball,contact,simulationClock))return;HitCount++;noDamageTime=0;changed=true;EmitFeedback(contact,"hit",b.Color);}
 public void RecyclePixelDrop(PixelDropBody b){if(!b.Brick.Alive)return;rules.DamageTarget(b.Brick.Id,int.MaxValue,out bool removed);if(!removed)return;BreakCount++;changed=true;EmitFeedback(new Vector2(Mathf.Sign(b.Body.position.x)*FlightGeometry.Right,b.Body.position.y),"splash",b.Color);b.Brick.Remove();}
}
public sealed class PixelDropBody:MonoBehaviour {
 public Rigidbody2D Body{get;private set;} public BrickBody Brick{get;private set;}
 public Color Color{get;private set;} public bool Detached{get;private set;}
 public float Stability{get;private set;}
 public bool Moving=>Body.linearVelocity.sqrMagnitude>.015f||Mathf.Abs(Body.angularVelocity)>3||(!Detached&&Vector2.Distance(Body.position,anchor)>.015f);
 static Material jellyMaterial;
 GameSession owner;Collider2D box;Vector2 anchor;Vector3 faceScale;Transform visual;float wobble=10,axis;int grade;
 readonly Dictionary<int,float> lastBallHit=new Dictionary<int,float>();readonly List<SpriteRenderer> marks=new List<SpriteRenderer>();
 public void Initialize(GameSession session,BrickBody brick){owner=session;Brick=brick;grade=brick.Definition.hp;Stability=grade;anchor=transform.position;
 ColorUtility.TryParseHtmlString(brick.Definition.paintColor,out var tint);Color=tint;
 var original=GetComponent<BoxCollider2D>();original.enabled=false;
 int shape=brick.Definition.axis;var size=brick.Definition.size;
 var polygon=gameObject.AddComponent<PolygonCollider2D>();polygon.points=PixelPieceArt.Points(shape,size);box=polygon;box.enabled=true;
 foreach(var wall in session.GetComponentsInChildren<StaticAimWall>())if(wall.name=="LeftWall"||wall.name=="RightWall")Physics2D.IgnoreCollision(box,wall.GetComponent<Collider2D>());
 var material=new PhysicsMaterial2D("Tabletop jelly"){friction=.28f,bounciness=.72f};box.sharedMaterial=material;
 Body=gameObject.AddComponent<Rigidbody2D>();Body.mass=PixelDropLevels.Mass(grade);Body.gravityScale=0;Body.linearDamping=.7f;Body.angularDamping=1.8f;Body.collisionDetectionMode=CollisionDetectionMode2D.Continuous;Body.interpolation=RigidbodyInterpolation2D.Interpolate;
 visual=brick.Face.transform.parent;visual.localScale=Vector3.one;visual.localPosition=Vector3.zero;
 if(jellyMaterial==null)jellyMaterial=new Material(Resources.Load<Shader>("PixelJelly"));brick.Face.sprite=PixelPieceArt.Sprite(shape);brick.Face.sharedMaterial=jellyMaterial;brick.Face.color=tint;brick.Face.transform.localScale=new Vector3(size.x,size.y,1);faceScale=brick.Face.transform.localScale;
 var shadow=transform.Find("Ground contact shadow");if(shadow!=null){shadow.localPosition=new Vector3(0,-size.y*.4f,0);shadow.localScale*=.9f;shadow.GetComponent<SpriteRenderer>().color=new Color(0.28f,.20f,.23f,.12f+.035f*grade);}
 for(int i=0;i<grade;i++)marks.Add(WorldArt.Draw(visual,"Stability tick "+i,"brick",new Vector2((i-(grade-1)*.5f)*.075f,-size.y*.30f),new Vector2(.045f,.035f),new Color(.23f,.22f,.27f,.7f),45));
 }
 public bool BallHit(BallBody ball,Vector2 contact,float clock){if(lastBallHit.TryGetValue(ball.Id,out float last)&&clock-last<.16f)return false;lastBallHit[ball.Id]=clock;
 Vector2 dir=(Body.position-ball.Body.position).normalized;if(dir.sqrMagnitude<.1f)dir=Vector2.up;
 // Solver transfers linear momentum. A small point impulse makes grazing corner torque legible.
 Body.AddForceAtPosition(dir*1.4f,contact,ForceMode2D.Impulse);Weaken(1,contact,dir);return true;}
 public void Weaken(float amount,Vector2 point,Vector2 direction){if(!Brick.Alive)return;wobble=0;axis=Mathf.Atan2(direction.y,direction.x)*Mathf.Rad2Deg;
 if(!Detached){Stability=Mathf.Max(0,Stability-amount);if(Stability<=0)Detach();}
 UpdateMarks();}
 void UpdateMarks(){for(int i=0;i<marks.Count;i++)marks[i].color=i<Mathf.CeilToInt(Stability)?new Color(.23f,.22f,.27f,.7f):new Color(.23f,.22f,.27f,.12f);}
 public void Detach(){if(Detached)return;Detached=true;Stability=0;Body.gravityScale=0;Body.linearDamping=0;Body.linearVelocity+=Vector2.right*(Mathf.Abs(Body.position.x)>.1f?Mathf.Sign(Body.position.x):(Brick.Id%2==0?1:-1))*5.2f;Body.angularDamping=.35f;wobble=0;UpdateMarks();}
 public void Tick(float dt){if(!owner.IsPlaying||owner.IsPaused||!Brick.Alive)return;
 wobble+=dt;
 if(box.bounds.min.x>FlightGeometry.Right||box.bounds.max.x<FlightGeometry.Left){owner.RecyclePixelDrop(this);return;}
 if(!Detached){Vector2 delta=anchor-Body.position;float softness=.7f+.3f*Stability/grade;
 Body.AddForce((delta*(30f*softness)-Body.linearVelocity*7f)*Body.mass);
 Body.AddTorque((-Mathf.DeltaAngle(0,Body.rotation)*.12f-Body.angularVelocity*.065f)*Body.mass);
 // Large displacement/rotation can tear a weakened tether before its nominal count reaches zero.
 if(delta.magnitude>.18f+.085f*Stability||Mathf.Abs(Mathf.DeltaAngle(0,Body.rotation))>30+Stability*9)Detach();
 }else{
 // Table friction opposes motion in both axes; no vertical force or automatic drift.
 var v=Body.linearVelocity;Body.linearVelocity=Vector2.MoveTowards(v,Vector2.zero,(.65f+.12f*Body.mass)*dt);
 Body.angularVelocity=Mathf.MoveTowards(Body.angularVelocity,0,45*dt);
 }
 if(Body.position.y>FlightGeometry.Top-.4f){Body.position=new Vector2(Body.position.x,FlightGeometry.Top-.4f);Body.linearVelocity=new Vector2(Body.linearVelocity.x,-Mathf.Abs(Body.linearVelocity.y)*.8f);}
 if(Body.position.y<FlightGeometry.ReturnY+.4f){Body.position=new Vector2(Body.position.x,FlightGeometry.ReturnY+.4f);Body.linearVelocity=new Vector2(Body.linearVelocity.x,Mathf.Abs(Body.linearVelocity.y)*.8f);}

 }
 void OnCollisionEnter2D(Collision2D c){if(!owner.IsPlaying||owner.IsPaused)return;var other=c.collider.GetComponent<PixelDropBody>();if(other==null)return;float speed=c.relativeVelocity.magnitude;if(speed<.6f)return;
 Vector2 point=c.contactCount>0?c.GetContact(0).point:Body.position;
 Weaken(Mathf.Clamp(speed*other.Body.mass*.11f,0,2.5f),point,c.relativeVelocity);owner.EmitFeedback(point,"hit",Color);}
 void LateUpdate(){if(owner==null||owner.IsPaused||!Brick.Alive)return;
 float f=grade<=2?20:15;float pulse=Mathf.Exp(-wobble*5)*Mathf.Cos(wobble*f)*(.22f/Mathf.Sqrt(Body.mass));
 // Rotate the visual deformation axes; the collider retains its rigid square shape.
 visual.localRotation=Quaternion.Euler(0,0,axis);Brick.Face.transform.localRotation=Quaternion.Euler(0,0,-axis);
 visual.localScale=new Vector3(1-pulse,1+pulse,1);Brick.Face.transform.localScale=faceScale;
 }
 void OnDestroy(){if(box!=null&&box.sharedMaterial!=null)Destroy(box.sharedMaterial);}
}
}
