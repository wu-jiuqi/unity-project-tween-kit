        public static Vector2 NormalizeAimDirection(Vector2 direction)

        }

        internal static ChainTool NearestReadyAimTool(IReadOnlyList<ChainTool> tools,float clock,Vector2 origin,Vector2 direction,float limit,out float distance){

        public void PredictAim(Vector2 direction,bool render=false)
        {
            AimPredictionCount++;Vector2 d=NormalizeAimDirection(direction),p=SpawnPosition(d);AimReflections=0;ChainTool aimedPower=null;float budget=32f,nextDot=0;int index=0;Collider2D previousWall=null,cornerWall=null;
            bool finitePierceAim=false;int passes=int.MaxValue;
            var filter=new ContactFilter2D();filter.SetLayerMask(1<<8);filter.useTriggers=false;
            for(int segment=0;segment<3&&budget>0;segment++){
                int count=Physics2D.CircleCast(p,FlightGeometry.Radius,d,filter,aimHits,budget);
                RaycastHit2D hit=default;bool found=false;float nearest=budget;
                for(int i=0;i<count;i++){
                    var h=aimHits[i];
                    if((UsesShopBoosters)&&SelectedStreakBooster==StreakBooster.Pierce&&h.collider!=null){var target=h.collider.GetComponentInParent<BrickBody>();if(target!=null&&target.Definition.kind!=BrickKind.Stone&&passes>0){passes--;continue;}}
                    if(h.distance<=.0001f&&(h.collider==previousWall||h.collider==cornerWall))continue;
                    if(h.collider==null||h.collider.attachedRigidbody!=null&&h.collider.attachedRigidbody.bodyType!=RigidbodyType2D.Static&&!(IsChainCollection&&h.collider.GetComponent<BrickBody>()!=null))continue;
                    if(h.distance<nearest){hit=h;nearest=h.distance;found=true;}
                }
                if(IsChainCollection&&render){var tool=NearestReadyAimTool(ChainTools,simulationClock,p,d,nearest,out float toolDistance);if(tool!=null){nearest=toolDistance;found=false;aimedPower=tool;}}
                Vector2 end=p+d*nearest;
                if(render)while(nextDot<=nearest&&index<aimDots.Count){var dot=aimDots[index++];dot.transform.position=p+d*nextDot;dot.color=new Color(.35f,.63f,.93f,.85f);AimVisible(dot.gameObject,true);nextDot+=.34f;}
                nextDot-=nearest;AimEndpoint=end;budget-=nearest;
                if(!found||hit.collider.GetComponent<StaticAimWall>()==null||AimReflections>=2||hit.distance<=.0001f)break;
                previousWall=hit.collider;cornerWall=null;Vector2 offset=hit.normal;d=Vector2.Reflect(d,hit.normal).normalized;AimReflections++;
                // Simultaneous corner contacts consume both reflections and leave both surfaces.
                for(int i=0;i<count;i++){var other=aimHits[i];if(other.collider==hit.collider||other.collider==null||other.collider.GetComponent<StaticAimWall>()==null||Mathf.Abs(other.distance-nearest)>.002f||Vector2.Dot(d,other.normal)>=0)continue;if(AimReflections>=2){budget=0;break;}cornerWall=other.collider;d=Vector2.Reflect(d,other.normal).normalized;offset+=other.normal;AimReflections++;}
                p=end+offset*.001f;
