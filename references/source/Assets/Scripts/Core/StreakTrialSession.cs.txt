namespace BounceGarden {
public enum StreakBooster {None,Pierce,Bomb,Super}


public static class StreakTrials {
 
 
 
 public static string Name(StreakBooster k)=>k==StreakBooster.Pierce?"穿透毛线球":k==StreakBooster.Bomb?"爆炸毛线球":k==StreakBooster.Super?"超级光球":"普通毛线球";
 public static string Rule(StreakBooster k)=>k==StreakBooster.Pierce?"穿过砖块，直接伤害不变，石墩阻挡":k==StreakBooster.Bomb?"碰砖或碰墙反弹，触发3×3爆炸，范围伤害2点":k==StreakBooster.Super?"碰砖或碰墙时，触发十字激光，造成2点伤害":"选择一种道具，强化下一组全部5颗球";
 
}
public sealed partial class GameSession {
 public int StreakTrialIndex {get;private set;}=-1;
 
 readonly int[] streakTools=new int[4];
 public StreakBooster SelectedStreakBooster {get;private set;}
 public int StreakExplosions {get;private set;}public int StreakLasers {get;private set;}
 StreakBooster preparedShopBooster;
 // Fusion editor sessions exercise every booster with their isolated memory profile.
 // Production and legacy editor sessions retain their actual campaign unlock index.
 int CampaignBoosterIndex=>LevelIndex;
 bool PrepareShopBooster(){preparedShopBooster=StreakBooster.None;if(!UsesShopBoosters||SelectedStreakBooster==StreakBooster.None)return true;if(!BoosterProfile.ConsumeCampaignBooster(SelectedStreakBooster,CampaignBoosterIndex)){changed=true;Feedback?.Invoke(LauncherPosition,"booster-save-failed");return false;}preparedShopBooster=SelectedStreakBooster;return true;}
 public LocalProfile BoosterProfile {get;set;}
 public bool UsesShopBoosters=>IsChainCollection&&!IsOpening&&BoosterProfile!=null;
 public int StreakStock(StreakBooster k)=>UsesShopBoosters?BoosterProfile.BoosterStock(k)+BoosterProfile.StreakBonusStock(k,CampaignBoosterIndex)+(k==StreakBooster.Pierce&&BoosterProfile.HasRescue(CampaignBoosterIndex)?1:0):(int)k>0&&(int)k<streakTools.Length?streakTools[(int)k]:0;
 public bool ApplyCampaignStreakBonus(){if(!UsesShopBoosters||!BoosterProfile.PrepareCampaignStreakRound(CampaignBoosterIndex))return false;NotifyBoosterStock();for(int i=1;i<=3;i++){var k=(StreakBooster)i;if(BoosterProfile.StreakBonusStock(k,CampaignBoosterIndex)>0&&BoosterAvailable(k))return true;}return false;}
 public bool BoosterAvailable(StreakBooster k)=>UsesShopBoosters&&BoosterProfile.BoosterUnlocked(k,CampaignBoosterIndex);
 public void NotifyBoosterStock(){changed=true;StateChanged?.Invoke();}
 public bool CanSelectStreakBooster=>(UsesShopBoosters)&&ChainCanLaunch&&!bufferedChainShot.HasValue&&!CollectionVictoryPending;
 public bool SelectStreakBooster(StreakBooster kind){if(!CanSelectStreakBooster||(int)kind<0||(int)kind>3||kind!=StreakBooster.None&&(!BoosterAvailable(kind)||StreakStock(kind)==0))return false;SelectedStreakBooster=SelectedStreakBooster==kind?StreakBooster.None:kind;SoundService.Play(SelectedStreakBooster==StreakBooster.None?"booster_cancel":"booster_select");RefreshLoadedStreakBall();SetAim(null);changed=true;StateChanged?.Invoke();return true;}
 void ResetStreakTrialTools(){Array.Clear(streakTools,0,streakTools.Length);SelectedStreakBooster=StreakBooster.None;StreakExplosions=StreakLasers=0;}
 StreakBooster ConsumeStreakSelection(){var k=SelectedStreakBooster;if(UsesShopBoosters){k=preparedShopBooster;preparedShopBooster=StreakBooster.None;}else return StreakBooster.None;SelectedStreakBooster=StreakBooster.None;if(k!=StreakBooster.None){SoundService.Play("booster_apply");}RefreshLoadedStreakBall();return k;}
 
 public void TriggerStreakImpact(BallBody ball,Vector2 contact){
  if(!(UsesShopBoosters)||!IsPlaying||IsPaused||CollectionVictoryPending||ball==null||!ball.Active||ball.SessionGeneration!=Generation||ball.StreakProcStep==PhysicsStep)return;
  if(ball.StreakPower!=StreakBooster.Bomb&&ball.StreakPower!=StreakBooster.Super)return;
  // One event per ball per physics step coalesces simultaneous corner contacts;
  // ordinary subsequent bounces retain the requested repeated activation.
  ball.StreakProcStep=PhysicsStep;using(StrategyCause("booster",ball.ChainBurstId))ReleaseStreakImpact(ball,contact);
 }
 void ReleaseStreakImpact(BallBody ball,Vector2 contact){
  int c=Mathf.Clamp(Mathf.RoundToInt(contact.x/CollectionLevels.Pitch)+4,0,8),r=Mathf.Clamp(Mathf.RoundToInt((CollectionLevels.Top-contact.y)/CollectionLevels.Pitch),0,8);
  Vector2 center=ChainGridRules.Center(c,r);long previous=powerAttack;powerAttack=++powerAttackSerial;
  long openingBatch=BeginOpeningElimination();try{
   if(ball.StreakPower==StreakBooster.Super){PlayStreakImpact(StreakBooster.Super,center);StreakLasers++;SoundService.Play("cross_laser");foreach(var d in new[]{Vector2.up,Vector2.right,Vector2.down,Vector2.left})FireCrossRay(center,d,1,false,StreakBooster.Super);}
   else {
    StreakExplosions++;SoundService.Play("bomb");
    PlayStreakImpact(StreakBooster.Bomb,center);
    float radius=CollectionLevels.Pitch*1.5f;
    foreach(var b in CollectionBricks.ToArray())if(b.Alive&&!b.GetComponent<ChainTileVisibility>().Hidden){
     var body=b.GetComponent<Rigidbody2D>();Vector2 p=body!=null?body.position:(Vector2)b.transform.position;var half=b.GetComponent<BoxCollider2D>().size*.5f;
     if(Mathf.Abs(p.x-center.x)>=radius+half.x-.005f||Mathf.Abs(p.y-center.y)>=radius+half.y-.005f)continue;
     Vector2 d=p-center;if(d.sqrMagnitude<.001f)d=ball.IncomingDirection;d=d.normalized;
     ChainRayEntry(b,center,d,24,out _,out Vector2 point);ApplyChainImpact(b,2,d,point,false);
    }
    foreach(var tool in ChainTools.ToArray())if(Mathf.Abs(tool.Position.x-center.x)<=radius&&Mathf.Abs(tool.Position.y-center.y)<=radius)ActivateChainTool(tool,tool.Position-center,1);
   }
  }finally{powerAttack=previous;EndOpeningElimination(openingBatch);}
  changed=true;
 }
}
