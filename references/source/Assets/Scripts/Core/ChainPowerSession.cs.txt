using System.Collections.Generic;
using System.Linq;
using UnityEngine;

namespace BounceGarden {
public static class ChainPowerRules {
 
 public static bool IsNest(BrickSpawn d)=>d.reveal=="chain-nest";
 
}
public sealed class ChainBond {
 public readonly List<BrickBody> Members=new List<BrickBody>();
}
public sealed partial class GameSession {
 public readonly List<ChainBond> ChainBonds=new List<ChainBond>();
 public int ChainNestSplits {get;private set;}
 int chainBondSerial;
 void ClearChainPowers(){ClearChainPowerPreview();ChainBonds.Clear();ChainNestSplits=chainBondSerial=0;}
 
 
 void PowerLine(Vector2 a,Vector2 b,Color color,float width=.10f,float life=.32f,int art=0){
  if(art==3){DrawJellyPowerBeam(3,a,b,width,life);return;}
  Vector2 delta=b-a;if(delta.sqrMagnitude<.00001f)return;
  var view=WorldArt.Draw(world,"Chain power trail","brick",(a+b)*.5f,new Vector2(delta.magnitude,width),color,72).transform;
  view.rotation=Quaternion.Euler(0,0,Mathf.Atan2(delta.y,delta.x)*Mathf.Rad2Deg);chainBeams.Add(new ChainBeam{View=view,Life=life,Duration=life,StartAlpha=color.a});
 }
 
 
 void FireCrossRay(Vector2 p,Vector2 d,int depth,bool soda=false,StreakBooster skin=StreakBooster.None){
  float limit=d.x!=0?((d.x>0?FlightGeometry.Right:FlightGeometry.Left)-p.x)/d.x:((d.y>0?FlightGeometry.Top:ReturnLineY)-p.y)/d.y;limit=Mathf.Max(0,limit);
  var hits=new List<ChainRayTarget>();foreach(var b in CollectionBricks)if(b.Alive&&!b.GetComponent<ChainTileVisibility>().Hidden&&ChainRayEntry(b,p,d,limit,out float entry,out Vector2 point))hits.Add(new ChainRayTarget{Brick=b,Distance=entry,Point=point});
  foreach(var hit in hits.OrderBy(h=>h.Distance)){
   var b=hit.Brick;if(!b.Alive)continue;
   bool barrier=b.Definition.kind==BrickKind.Stone||b.Definition.kind==BrickKind.RearWeak&&!ChainRearEntry(b,d,hit.Point);
   ApplyChainImpact(b,skin==StreakBooster.Super?2:ChainPowerRules.IsNest(b.Definition)?8:int.MaxValue,d,hit.Point,false);
   if(barrier||b.Alive&&(b.Definition.kind==BrickKind.Armor||ChainPowerRules.IsNest(b.Definition))){limit=hit.Distance;break;}
  }
  if(skin==StreakBooster.Super)DrawStreakBeam(p,p+d*limit);else if(soda){PowerLine(p,p+d*limit,new Color(.7f,.92f,.4f,.85f),.23f);PowerLine(p,p+d*limit,new Color(.95f,1,.85f,.95f),.045f);}else DrawJellyPowerBeam(1,p,p+d*limit);
  foreach(var other in ChainTools.ToArray())if(!float.IsPositiveInfinity(Along(other.Position,p,d,limit,.39f)))ActivateChainTool(other,d,depth+1);
 }
 
 
}
}
