  if(CollectionGoalEarned){FinishEarnedCollectionSupply();return;}
  if(SevenFreezeBoard)return;
  if(chainReplacementRewards.Count==0||!IsPlaying&&!VictorySettlementActive||IsPaused||CollectionWon&&!SevenSpecialActive)return;
  var candidates=CollectionBricks.Where(b=>b.Alive&&!RefillProtected(b)&&b.Definition.kind==BrickKind.Normal&&b.Definition.juiceGroup==0&&!OpeningCopperBrick(b)&&!OpeningCatBrick(b)&&!OpeningIngotBrick(b)&&!OpeningYarnBrick(b)&&!OpeningCookieBrick(b)&&(!IsJellyTarget(b)||IsOpening)&&!ChainPowerRules.IsNest(b.Definition)&&!b.GetComponent<ChainTileVisibility>().Hidden&&!ChainBonds.Any(g=>g.Members.Contains(b))).ToList();

   var preferred=candidates.Where(b=>!IsJellyTarget(b)).ToList();if(preferred.Count==0)preferred=candidates;var old=preferred[chainRewardRandom.Next(preferred.Count)];candidates.Remove(old);

   jellyTargets.Remove(old.Id);rules.DamageTarget(old.Id,int.MaxValue,out _);old.Remove(false);
   ChainTool tool;int origin=strategyRewardOrigins.Count>0?strategyRewardOrigins.Dequeue():0;
   using(StrategyCause("reward-place",origin))tool=SpawnChainTool(chainReplacementRewards.Dequeue(),position);var piece=tool.View.GetComponent<ChainGridPiece>();
   piece.Col=previous.Col;piece.Row=previous.Row;piece.Size=previous.Size;piece.FallAt=previous.FallAt;piece.MotionVersion=previous.MotionVersion;
   // Replacing a waiting or moving tile keeps its existing deadline and path.
   foreach(var move in chainMotions)if(move.View==old.transform)move.View=tool.View;
   CollectionBricks.Remove(old);Destroy(old.gameObject);changed=true;
  }
  // A stable board made entirely of mechanisms/specials cannot accept a rating
  // replacement. During victory those rewards must not wait for the next
  // phase's special casts to create a candidate: that phase waits for us.
  if(VictorySettlementActive&&chainReplacementRewards.Count>0&&ChainGridReady&&!ChainEffectsPending&&!SevenAnimationsPending){
   while(chainReplacementRewards.Count>0)victoryUnplacedRewards.Enqueue(chainReplacementRewards.Dequeue());
   changed=true;
  }
 }
