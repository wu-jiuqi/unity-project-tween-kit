public sealed partial class GameSession {
 public bool IsCollection=>IsChainCollection;
 public readonly List<BrickBody> CollectionBricks=new List<BrickBody>();
 public readonly List<CollectionTreat> CollectionTreats=new List<CollectionTreat>();
 public readonly int[] Collected=new int[2],CollectionGoal=new int[2];
 public int CollectionEnergy {get;private set;}
 public int CollectionRound {get;private set;}
 public int CollectionTool {get;private set;}
 public bool CollectionBusy=>collectionPhase!=0||collectionRoundPending||IsChainCollection&&!ChainGridReady;
 public bool CollectionReady=>IsChainCollection?ChainCanLaunch:IsCollection&&IsPlaying&&!IsPaused&&!CollectionBusy&&ActiveBalls==0&&PendingBurstBalls==0&&!queuedShot.HasValue;
 public bool CollectionWon=>IsOpening?OpeningGoalsComplete:Collected[0]>=CollectionGoal[0]&&Collected[1]>=CollectionGoal[1]&&JellyCollectionComplete;
 public const float CollectionVictoryDelay=1f;
 public bool CollectionVictoryPending {get;private set;}
 float collectionVictoryAt;
 
 public int[] NextCollectionRow {get;private set;}=new int[9];
 int collectionPhase;bool collectionFailed,collectionRoundPending;
 
 
 
 void ClearCollection(){CollectionVictoryPending=false;collectionVictoryAt=0;ClearJellyCollection();ClearChainCollection();CollectionBricks.Clear();collectionFailed=false;collectionRoundPending=false;CollectionTool=0;CollectionEnergy=0;CollectionRound=1;CollectionGoal[0]=CollectionGoal[1]=0;Collected[0]=Collected[1]=0;} void StartCollection(){if(IsChainCollection)StartChainCollection();}
 
 
 
 void AttachCollectionBrick(BrickBody b){CollectionBricks.Add(b);RegisterChainGrid(b.transform,OpeningCookieBrick(b)?2:1);} 
 
 
 
 
 
 void TickCollection(float dt){if(!IsCollection||CollectionWon||CollectionVictoryPending)return;TickChainCollection(dt);} bool ResolveCollection(out bool won){
  won=false;
  if(IsChainCollection){
   FinishEarnedCollectionSupply();
   if(CollectionWon){CollectionVictoryPending=true;collectionVictoryAt=simulationClock;collectionRoundPending=false;queuedShot=null;ClearBufferedChainShot();SetAim(null);won=true;rules.CommitOutcome(true);return true;}
   // Finish work already in flight, never wait for a new refill or an impossible
   // rating replacement after all collection goals have been earned.
   if(CollectionVictoryPending||CollectionWon||EarnedChainGoal()){if(SevenSpecialActive&&(SevenSettlementPending||!ChainGridReady||ChainEffectsPending||ActiveBalls>0||PendingBurstBalls>0||chainReplacementRewards.Count>0))return false;return ResolveCollectionVictory(out won);}
   // Failure still waits for all real and reserved balls, including split balls.
   if(ActiveBalls>0||PendingBurstBalls>0||ChainMinis.Count>0||queuedShot.HasValue||SevenSpecialActive&&(SevenSettlementPending||!ChainGridReady||ChainEffectsPending))return false;
   if(Shots>0&&!collectionFailed&&!CollectionWon&&!EarnedChainGoal())return false;
   CompleteEarnedChainLoot();won=CollectionWon;
   if(won)return ResolveCollectionVictory(out won);
   if(!won&&Shots>0&&!collectionFailed)return false;
   collectionRoundPending=false;ClearBufferedChainShot();rules.CommitOutcome(won);return true;
  }
  won=CollectionWon;if(!won&&!collectionFailed)return false;rules.CommitOutcome(won);return true;
 }
 bool ResolveCollectionVictory(out bool won){
  won=false;
  if(!CollectionVictoryPending){CollectionVictoryPending=true;collectionVictoryAt=simulationClock+CollectionVictoryDelay;queuedShot=null;ClearBufferedChainShot();SetAim(null);changed=true;}
  if(simulationClock+.00001f<collectionVictoryAt)return false;
  CompleteEarnedChainLoot();collectionRoundPending=false;won=true;rules.CommitOutcome(true);return true;
 }
}
