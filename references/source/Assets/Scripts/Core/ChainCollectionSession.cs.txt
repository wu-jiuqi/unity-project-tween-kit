 static float Along(Vector2 p,Vector2 start,Vector2 direction,float distance,float radius){
  float t=Vector2.Dot(p-start,direction);if(t<0||t>distance)return float.PositiveInfinity;
  return ((start+direction*t)-p).sqrMagnitude<=radius*radius?t:float.PositiveInfinity;
 }
 sealed class ChainRayTarget {public BrickBody Brick;public float Distance;public Vector2 Point;}
 static bool ChainRayEntry(BrickBody brick,Vector2 origin,Vector2 direction,float limit,out float entry,out Vector2 point){
  var box=brick.GetComponent<BoxCollider2D>();var body=brick.GetComponent<Rigidbody2D>();Vector2 center=body!=null?body.position+box.offset:(Vector2)brick.transform.TransformPoint(box.offset);
  Vector2 half=Vector2.Scale(box.size,(Vector2)brick.transform.lossyScale)*.5f,min=center-half,max=center+half;
  float near=0,far=limit;
  for(int axis=0;axis<2;axis++){
   float v=direction[axis],o=origin[axis];
   if(Mathf.Abs(v)<.00001f){if(o<min[axis]||o>max[axis]){entry=0;point=origin;return false;}continue;}
   float a=(min[axis]-o)/v,b=(max[axis]-o)/v;if(a>b){float swap=a;a=b;b=swap;}
   near=Mathf.Max(near,a);far=Mathf.Min(far,b);if(far<near){entry=0;point=origin;return false;}
  }
  entry=near;point=origin+direction*near;return near<=limit;
 }

 void TickChainMinis(float dt){
  foreach(var mini in ChainMinis.ToArray()){
   mini.Life-=dt;if(mini.Life<=0||mini.Position.y<-6.1f){Destroy(mini.View.gameObject);ChainMinis.Remove(mini);continue;}
   float remaining=11*dt;
   // Swept collisions prevent tunneling. Only layer 8 contains board colliders.
   for(int bounceIndex=0;remaining>.0001f&&bounceIndex<5;bounceIndex++){
    var hits=Physics2D.CircleCastAll(mini.Position,.09f,mini.Direction,remaining,1<<8);
    var hit=hits.Where(h=>h.collider!=null&&!h.collider.isTrigger).OrderBy(h=>h.distance).FirstOrDefault();
    float travel=hit.collider!=null?hit.distance:remaining;
    var tool=ChainTools.Where(t=>simulationClock>=t.ReadyAt).OrderBy(t=>Along(t.Position,mini.Position,mini.Direction,travel,.4f)).FirstOrDefault();
    if(tool!=null&&!float.IsPositiveInfinity(Along(tool.Position,mini.Position,mini.Direction,travel,.4f)))ActivateChainTool(tool,mini.Direction,mini.Depth);
    mini.Position+=mini.Direction*travel;remaining-=travel;
    if(hit.collider==null)break;
    PlayerHaptics71.Pulse();
    var brick=hit.collider.GetComponentInParent<BrickBody>();if(brick!=null)ApplyChainImpact(brick,1,mini.Direction,hit.point,false);
    mini.Direction=Vector2.Reflect(mini.Direction,hit.normal).normalized;mini.Position+=hit.normal*.02f;remaining=Mathf.Max(0,remaining-.02f);
   }
   mini.View.position=mini.Position;mini.View.localRotation=Quaternion.Euler(0,0,simulationClock*240);
  }
 }
