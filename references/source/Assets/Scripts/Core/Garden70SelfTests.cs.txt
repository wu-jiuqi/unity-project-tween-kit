using System;using UnityEngine;

public static class Garden70SelfTests {
 public static string RunAll(){int checks=0;Action<bool,string> check=(ok,text)=>{checks++;if(!ok)throw new Exception("Garden70: "+text);};var store=new LeagueSelfTests.Store();var p=LocalProfile.Load(store);p.InitializeChapterGates();
  for(int n=0;n<10;n++){int before=p.CampaignWinStreak;check(p.CommitAlphabetWin("streak70-"+n,n,100),"win saves "+n);check(p.CampaignWinStreak==before+1,"new progression win increments once "+n);p.CommitAlphabetWin("streak70-"+n,n,100);check(p.CampaignWinStreak==before+1,"duplicate result does not increment "+n);
   p.PrepareCampaignStreakRound(n+1);for(int k=1;k<=3;k++)check(p.StreakBonusStock((StreakBooster)k,n+1)==(n+1>=(k==1?3:k==2?5:10)?1:0),"real streak loadout "+(n+1)+"/"+k);
  }
  int own=p.BoosterStock(StreakBooster.Super);check(p.ConsumeCampaignBooster(StreakBooster.Super,10)&&p.BoosterStock(StreakBooster.Super)==own&&p.StreakBonusStock(StreakBooster.Super,10)==0,"streak bonus consumed before purchased stock");p.PrepareCampaignStreakRound(10);check(p.StreakBonusStock(StreakBooster.Super,10)==0,"restart cannot refill spent bonus");p=LocalProfile.Load(store);check(p.CampaignWinStreak==10&&p.StreakBonusStock(StreakBooster.Super,10)==0&&p.StreakBonusStock(StreakBooster.Bomb,10)==1,"streak and remaining loadout survive restart");

  var legacy=new LocalProfile();legacy.CommitAlphabetWin("legacy58",57,100);legacy.InitializeChapterGates();check(legacy.ChapterOpened(1)&&legacy.ChapterOpened(2)&&legacy.PendingChapter<0,"legacy player already inside winter keeps access");
  var prior=new LocalProfile();prior.InitializeStreak70(5);check(prior.CampaignWinStreak==5,"legacy home-feedback streak is adopted once");prior.InitializeStreak70(20);check(prior.CampaignWinStreak==5,"legacy counter cannot overwrite new gameplay counter");return "GARDEN70_MODELS_OK checks="+checks;
 }
