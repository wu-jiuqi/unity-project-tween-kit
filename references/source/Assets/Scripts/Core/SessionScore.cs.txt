using UnityEngine;
namespace BounceGarden {
public static class ScoreRules {
 public const int Damage=100, Break=500, Objective=1000, SpareShot=3000, Cap=1000000000;
 public static int WinCoins(int score)=>40+Mathf.Min(30,Mathf.Max(0,score)/10000*5);
}
public sealed partial class GameSession {
 public int Score {get;private set;}
 public int CompletionBonus {get;private set;}
 public bool ScoreFinalized {get;private set;}
 // Observational counters only. Other score paths are reported as the residual.
 public int DamageScore {get;private set;}
 public int BreakScore {get;private set;}
 public int ObjectiveScore {get;private set;}
 public int CreditedCompletionBonus {get;private set;}
 public int VictoryClearScore {get;private set;}int victoryClearScoreFrom;
 // Optional future goal. Existing levels leave this zero and keep their collection rules.
 public int RequiredScore {get;private set;}
 public bool ScoreGoalMet=>RequiredScore<=0||Score>=RequiredScore;
 public void SetScoreGoal(int value){RequiredScore=Mathf.Clamp(value,0,ScoreRules.Cap);changed=true;}
 void ResetScore(){Score=CompletionBonus=RequiredScore=DamageScore=BreakScore=ObjectiveScore=CreditedCompletionBonus=VictoryClearScore=victoryClearScoreFrom=0;ScoreFinalized=false;}
 void AddScore(int value){if(ScoreFinalized||!IsPlaying&&!VictorySettlementActive||IsPaused||value<=0)return;int before=Score;Score=(int)System.Math.Min(ScoreRules.Cap,(long)Score+value);StrategyEmit(new StrategyEvent{kind="score",amount=Score-before});changed=true;}
 void ScoreImpact(BrickBody brick,int hp,bool broken){
  int damage=Mathf.Clamp(hp,0,10000)*ScoreRules.Damage,clear=broken?ScoreRules.Break:0,goal=broken&&(brick.Definition.juiceGroup>0||IsJellyTarget(brick))?ScoreRules.Objective:0;
  int before=Score;AddScore(damage+clear+goal);int credited=Score-before;
  int d=Mathf.Min(credited,damage);DamageScore+=d;credited-=d;
  int b=Mathf.Min(credited,clear);BreakScore+=b;credited-=b;ObjectiveScore+=credited;
 }
 public int FinalizeScore(bool win){if(ScoreFinalized)return Score;if(win){int before=Score;CompletionBonus=Mathf.Max(0,Shots)*ScoreRules.SpareShot;Score=(int)System.Math.Min(ScoreRules.Cap,(long)Score+CompletionBonus);CreditedCompletionBonus=Score-before;}ScoreFinalized=true;return Score;}
}
}
