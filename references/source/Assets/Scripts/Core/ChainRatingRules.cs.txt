namespace BounceGarden {
public static class ChainRatingRules {
 public static readonly string[] Names={"NICE","GREAT","AMAZING","INCREDIBLE"};
 public static int Tier(int breaks){int tier=-1;for(int i=0;i<InGameFeedbackState.Thresholds.Length;i++)if(breaks>=InGameFeedbackState.Thresholds[i])tier=i;return tier;}
 public static int[] Rewards(int tier){switch(tier){case 0:return new[]{1};case 1:return new[]{2};case 2:return new[]{1,2};case 3:return new[]{1,2,3};default:return new int[0];}}
}
}
