 internal bool FusionSpawnSafe(Vector2 p,int size){
  float half=CollectionLevels.Pitch*size*.5f+FlightGeometry.Radius+.04f;
  foreach(var ball in active.Values.Where(b=>b.Active)){
   Vector2 start=ball.Body.position-p,delta=ball.Body.linearVelocity*Time.fixedDeltaTime;float enter=0,leave=1;bool intersects=true;
   for(int axis=0;axis<2;axis++){
    if(Mathf.Abs(delta[axis])<.00001f){if(Mathf.Abs(start[axis])>half){intersects=false;break;}continue;}
    float a=(-half-start[axis])/delta[axis],b=(half-start[axis])/delta[axis];enter=Mathf.Max(enter,Mathf.Min(a,b));leave=Mathf.Min(leave,Mathf.Max(a,b));if(enter>leave){intersects=false;break;}
   }
   if(intersects)return false;
  }return true;
 }

 internal void FusionHit(FusionRuntimePiece p,int damage,Vector2 incoming,Vector2 point,bool direct=false,bool clear=false){
  if(p==null||!p.Alive||!p.Enabled||damage<1||!FusionActive||!IsPlaying||IsPaused)return;
  if(fusionAttack==null){FusionRunAttack("external",chainBurstSerial,()=>FusionHit(p,damage,incoming,point,direct,clear));return;}
  if(!fusionAttack.Snapshot.TryGetValue(p.Definition.id,out var pre)||!pre.Enabled)return;
  if(p.Definition.kind=="rear"&&!ChainRearEntry(p.Brick,incoming,point)){
   ChainBlocks++;FusionRecord("RearBlocked",p.Definition.id);return;
  }
  if(fusionEarliestTool.TryGetValue(p.Definition.id,out int earliest)&&FusionCurrentVolley<earliest)return;
  var hits=direct?fusionAttack.DirectHits:fusionAttack.EffectHits;if(!hits.Add(p.Definition.id))return;
  if(pre.Protected){
   if(!fusionAttack.GateHits.Add(p.Definition.id))return;
   if(p.Ice>0){p.Ice--;SnowThaws++;FusionProgress(p,"IceRemoved");SoundService.Play("ice_break");}
   else if(p.Brick!=null&&p.Brick.ShellIntact){p.Brick.BreakShell();ChainArmorBreaks++;FusionProgress(p,"ArmorRemoved");SoundService.Play("armor_break");}
   FusionPulse(point,"保护解除",new Color(1,.8f,.4f));return;
  }
  if(pre.Unripe||p.Definition.kind=="stone"){ChainBlocks++;FusionRecord("Blocked",p.Definition.id);return;}
  if(p.Tool!=null){QueueFusionTool(p);return;}
  bool gate=FusionRules.IsDevice(p.Definition.kind)||p.Definition.kind=="basket"||p.Definition.kind=="snowball"||p.Definition.kind=="snowdrift";
  if(gate&&!fusionAttack.GateHits.Add(p.Definition.id))return;
  if(FusionRules.IsDevice(p.Definition.kind)){HandleFusionDeviceHit(p,damage,incoming,point,direct);return;}
  if(p.Definition.kind=="basket"){
   p.Progress++;FusionProgress(p,"BasketProgress");if(p.Progress>=p.Definition.hitsPerItem){p.Progress=0;FusionServe(p,1);}return;
  }
  if(p.Definition.kind=="snowball"||p.Definition.kind=="snowdrift"){FusionSnowHit(p,false);return;}
  if(p.Definition.kind=="cookie"){StartFusionCookie(p,direct);return;}
  FusionDamageBody(p,clear?int.MaxValue:damage,direct);
 }

 }

 void StartFusionCookie(FusionRuntimePiece trigger,bool direct){
  if(fusionPendingRemoval.Contains(trigger.Definition.id))return;
  var members=fusionPieces.Values.Where(p=>p.Alive&&p.Definition.kind=="cookie"&&!FusionProtected(p)).ToArray();var todo=new Queue<Tuple<FusionRuntimePiece,int>>();var seen=new HashSet<string>();todo.Enqueue(Tuple.Create(trigger,0));
  long chain=++fusionCascadeSerial;
  while(todo.Count>0){var next=todo.Dequeue();var p=next.Item1;if(!seen.Add(p.Definition.id))continue;fusionPendingRemoval.Add(p.Definition.id);int distance=next.Item2;
   Action remove=()=>{fusionPendingRemoval.Remove(p.Definition.id);if(!p.Alive)return;FusionRemove(p,true,true,direct&&p==trigger);PicnicCookieBreaks++;foreach(var basket in fusionPieces.Values.Where(b=>b.Alive&&b.Definition.kind=="basket"&&FusionAdjacent(b,p)).ToArray())if(basket.SeenRemovals.Add("cookie:"+chain))FusionServe(basket,basket.Definition.bonusServings);};
   if(p==trigger)remove();else FusionQueue(()=>FusionRunAttack("cookie:"+chain,FusionCurrentVolley,remove),distance*.07f);
   foreach(var candidate in members)if(!seen.Contains(candidate.Definition.id)&&FusionAdjacent(candidate,p))todo.Enqueue(Tuple.Create(candidate,distance+1));
  }

 void TickFusion(float dt){
  var ready=fusionQueue.Where(q=>q.At<=simulationClock).OrderBy(q=>q.At).ThenBy(q=>q.Order).ToArray();
  foreach(var q in ready){fusionQueue.Remove(q);if(q.Generation!=Generation||CollectionVictoryPending)continue;var previous=fusionAttack;fusionAttack=new FusionAttackContext{Cascade=q.Cascade,Parent=q.Parent,Volley=q.Volley,Network=q.Network,Source="queued",Depth=q.Depth-1};try{q.Action();}finally{fusionAttack=previous;}}
  TickFusionDevices(dt);TickFusionPhysics();TickChainGravity(dt);TickFusionPresentation(dt);
  FusionMetrics.NoProgressVolleys=FusionMetrics.Volleys.Count(v=>v.VolleyId>0&&!chainBursts.ContainsKey(v.VolleyId)&&v.ObjectiveCredits==0&&v.Transitions==0);
 }
