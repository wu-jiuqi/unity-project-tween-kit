using System;
using System.IO;
using UnityEditor;
using UnityEditor.Build;
using UnityEditor.Build.Reporting;
using UnityEngine;
public static class CombatFeedback80Build {
 public static void Windows(){
  if(BounceGarden.GameVersion.Name!="0.1.80"||BounceGarden.GameVersion.Code!=80)throw new Exception("This release builder requires 0.1.80");
  OptimizationV76Build.Validate();Debug.Log(BounceGarden.LeagueRoundSelfTests.RunAll(Path.GetFullPath("../../BounceGarden/Evidence/CombatFeedback-v80/LeagueModel")));
  BuildEntry.Configure();PlayerSettings.SetScriptingBackend(NamedBuildTarget.Standalone,ScriptingImplementation.Mono2x);
  string requested=Environment.GetEnvironmentVariable("BOUNCE_COMBAT80_OUTPUT");string output=string.IsNullOrEmpty(requested)?Path.GetFullPath("../../Builds/CombatFeedback-v80/Windows/BounceGarden.exe"):Path.GetFullPath(requested);Directory.CreateDirectory(Path.GetDirectoryName(output));
  var r=BuildPipeline.BuildPlayer(new BuildPlayerOptions{scenes=new[]{"Assets/Scenes/Main.unity"},locationPathName=output,target=BuildTarget.StandaloneWindows64,options=BuildOptions.None});
  Debug.Log("COMBAT_FEEDBACK_80_BUILD "+r.summary.result+" version="+PlayerSettings.bundleVersion+" errors="+r.summary.totalErrors+" warnings="+r.summary.totalWarnings);
  if(r.summary.result!=BuildResult.Succeeded)throw new Exception("Combat feedback 0.1.80 build failed");
 }
}
