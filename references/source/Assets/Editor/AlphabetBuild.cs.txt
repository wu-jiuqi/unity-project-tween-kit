using System;
using System.IO;
using UnityEditor;
using UnityEditor.Build;
using UnityEditor.Build.Reporting;
using UnityEngine;
using BounceGarden;
public static class AlphabetBuild {
 public static void Compile(){BuildEntry.Configure();AssetDatabase.Refresh();Debug.Log("ALPHABET_DATA_CHECKS "+AlphabetCampaign.Validate());Debug.Log(RuleSelfTests.RunAll());Debug.Log(LocalProfile.RunSelfTests());Debug.Log("ALPHABET_PROFILE_CHECKS "+AlphabetProfileTests.Run());Debug.Log("ALPHABET_COMPILE_OK");}
 public static void Windows(){Compile();ValidateIntroductionAssets();PlayerSettings.SetScriptingBackend(NamedBuildTarget.Standalone,ScriptingImplementation.Mono2x);
  string path=Environment.GetEnvironmentVariable("BOUNCE_ALPHABET_WINDOWS")??Path.GetFullPath(Path.Combine(Application.dataPath,"../../../Builds/AlphabetCampaign-v"+GameVersion.Code+"/Windows/BounceGarden.exe"));Directory.CreateDirectory(Path.GetDirectoryName(path));
  var report=BuildPipeline.BuildPlayer(new BuildPlayerOptions{scenes=new[]{"Assets/Scenes/Main.unity"},target=BuildTarget.StandaloneWindows64,locationPathName=path,options=BuildOptions.CompressWithLz4HC,extraScriptingDefines=new[]{"BOUNCE_TEST_TOOLS"}});
  string result="ALPHABET_BUILD "+report.summary.result+" errors="+report.summary.totalErrors+" warnings="+report.summary.totalWarnings;File.WriteAllText(path+".build.txt",result);Debug.Log(result);if(report.summary.result!=BuildResult.Succeeded)throw new Exception(result);
 }
 public static void DevelopmentWindows(){Compile();PlayerSettings.SetScriptingBackend(NamedBuildTarget.Standalone,ScriptingImplementation.Mono2x);string path=Path.GetFullPath(Path.Combine(Application.dataPath,"../../../Builds/AlphabetCampaign-v38/QA/BounceGarden.exe"));Directory.CreateDirectory(Path.GetDirectoryName(path));var report=BuildPipeline.BuildPlayer(new BuildPlayerOptions{scenes=new[]{"Assets/Scenes/Main.unity"},target=BuildTarget.StandaloneWindows64,locationPathName=path,options=BuildOptions.CompressWithLz4HC,extraScriptingDefines=new[]{"BOUNCE_TEST_TOOLS"}});Debug.Log("ALPHABET_DEV_BUILD "+report.summary.result);if(report.summary.result!=BuildResult.Succeeded)throw new Exception("Development build failed");}
 public static void ValidateIntroductionAssets(){int n=0;for(int id=1;id<=12;id++)foreach(string layer in new[]{"title","hero1","hero2","hero3","hero4","arrow"}){var s=Resources.Load<Sprite>("Art/AlphabetV38/Introductions/"+id.ToString("00")+"/"+layer);if(s==null)throw new Exception("Missing independent introduction layer "+id+"/"+layer);n++;}Debug.Log("ALPHABET_INTRO_LAYERS "+n);}
}
public sealed class AlphabetAssetImporter:AssetPostprocessor {
 void OnPreprocessTexture(){if(!assetPath.Contains("/Art/AlphabetV38/"))return;var t=(TextureImporter)assetImporter;t.textureType=TextureImporterType.Sprite;t.spriteImportMode=SpriteImportMode.Single;t.alphaIsTransparency=true;t.mipmapEnabled=false;t.filterMode=FilterMode.Bilinear;t.wrapMode=TextureWrapMode.Clamp;t.maxTextureSize=2048;t.textureCompression=TextureImporterCompression.Uncompressed;t.spritePixelsPerUnit=100;}
}
