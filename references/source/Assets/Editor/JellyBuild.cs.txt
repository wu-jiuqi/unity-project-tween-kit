using System;
using System.IO;
using UnityEditor;
using UnityEditor.Build.Reporting;
using UnityEngine;

public sealed class JellyAtlasImporter : AssetPostprocessor
{
    void OnPreprocessTexture()
    {
        if (assetPath != "Assets/Resources/Art/Jelly/blue_jelly_pop.png") return;
        var t = (TextureImporter)assetImporter;
        t.textureType = TextureImporterType.Default;
        t.alphaSource = TextureImporterAlphaSource.FromInput;
        t.alphaIsTransparency = true;
        t.mipmapEnabled = false;
        t.isReadable = false;
        t.npotScale = TextureImporterNPOTScale.None;
        t.maxTextureSize = 1024;
        t.wrapMode = TextureWrapMode.Clamp;
        t.filterMode = FilterMode.Bilinear;
        t.textureCompression = TextureImporterCompression.Uncompressed;
        t.SetPlatformTextureSettings(new TextureImporterPlatformSettings {
            name = "Android", overridden = true, maxTextureSize = 1024,
            format = TextureImporterFormat.ETC2_RGBA8, compressionQuality = 100
        });
    }
}

public static class JellyBuild
{
    public static void Windows()
    {
        BuildEntry.Configure(); BuildEntry.Rules();
        Debug.Log("PROFILE_TESTS " + BounceGarden.LocalProfile.RunSelfTests());
        PlayerSettings.bundleVersion = BounceGarden.GameVersion.Name;
        PlayerSettings.SetScriptingBackend(UnityEditor.Build.NamedBuildTarget.Standalone, ScriptingImplementation.Mono2x);
        string path = Environment.GetEnvironmentVariable("BOUNCE_JELLY_OUTPUT");
        if (string.IsNullOrEmpty(path)) throw new Exception("BOUNCE_JELLY_OUTPUT required");
        Build(BuildTarget.StandaloneWindows64, path);
    }
    public static void Android()
    {
        BuildEntry.Configure(); BuildEntry.Rules();
        PlayerSettings.bundleVersion = BounceGarden.GameVersion.Name;
        PlayerSettings.Android.bundleVersionCode = BounceGarden.GameVersion.Code;
        string path = Environment.GetEnvironmentVariable("BOUNCE_APK_OUTPUT");
        if (string.IsNullOrEmpty(path)) throw new Exception("BOUNCE_APK_OUTPUT required");
        Build(BuildTarget.Android, path);
    }
    static void Build(BuildTarget target, string path)
    {
        AssetDatabase.ImportAsset("Assets/Resources/Art/Jelly/blue_jelly_pop.png", ImportAssetOptions.ForceUpdate);
        Directory.CreateDirectory(Path.GetDirectoryName(path));
        var report = BuildPipeline.BuildPlayer(new BuildPlayerOptions {
            scenes = new[] { "Assets/Scenes/Main.unity" }, target = target,
            locationPathName = path, options = BuildOptions.CompressWithLz4HC
        });
        var s = report.summary;
        string summary = "JELLY_BUILD " + s.result + " errors=" + s.totalErrors + " warnings=" + s.totalWarnings + " bytes=" + s.totalSize;
        Debug.Log(summary); File.WriteAllText(path + ".build.txt", summary);
        if (s.result != BuildResult.Succeeded) throw new Exception(summary);
    }
}
