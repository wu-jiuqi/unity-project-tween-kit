using System;
using System.IO;
using UnityEditor;
using UnityEditor.Build;
using UnityEditor.Build.Reporting;
using UnityEngine;
public static class MobileV53Build {
 public static void Windows(){
  BuildEntry.Configure();PlayerSettings.SetScriptingBackend(NamedBuildTarget.Standalone,ScriptingImplementation.Mono2x);
  string path=Path.GetFullPath(Path.Combine(Application.dataPath,"../../../Builds/Mobile-v53/Windows/BounceGarden.exe"));
  Directory.CreateDirectory(Path.GetDirectoryName(path));
  var report=BuildPipeline.BuildPlayer(new BuildPlayerOptions{scenes=new[]{"Assets/Scenes/Main.unity"},locationPathName=path,target=BuildTarget.StandaloneWindows64,options=BuildOptions.None});
  Debug.Log("MOBILE_V53_BUILD "+report.summary.result+" errors="+report.summary.totalErrors+" warnings="+report.summary.totalWarnings);
  if(report.summary.result!=BuildResult.Succeeded)throw new Exception("Mobile v53 validation build failed");
 }
}
