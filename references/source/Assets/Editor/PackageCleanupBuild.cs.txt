using System;
using System.IO;
using System.Linq;
using System.Reflection;
using System.Text.RegularExpressions;
using BounceGarden;
using UnityEditor;
using UnityEditor.Build;
using UnityEditor.Build.Reporting;
using UnityEngine;

public static class PackageCleanupBuild {
 public static void Validate(){
  if(AlphabetCampaign.Count!=45||LevelCatalog.Count!=45)throw new Exception("Campaign must end at level 45");
  int checks=OpeningCampaign.Validate();
  var flags=BindingFlags.Instance|BindingFlags.NonPublic;
  foreach(int count in new[]{45,52,78}){
   var profile=new LocalProfile();
   string json=(string)typeof(LocalProfile).GetMethod("Serialize",flags).Invoke(profile,null);
   json=Regex.Replace(json,"\"bestScores\":\\[[^\\]]*\\]","\"bestScores\":["+string.Join(",",Enumerable.Repeat("100",count))+"]");
   json=Regex.Replace(json,"\"coins\":\\d+","\"coins\":1234");
   json=Regex.Replace(json,"\"alphabetUnlocked\":\\d+","\"alphabetUnlocked\":"+count);
   json=Regex.Replace(json,"\"streakRoundLevel\":-?\\d+","\"streakRoundLevel\":"+(count-1));
   json=Regex.Replace(json,"\"rescueLevel\":-?\\d+","\"rescueLevel\":"+(count-1));
   typeof(LocalProfile).GetMethod("Restore",flags).Invoke(profile,new object[]{json});
   if(profile.Coins!=1234||profile.BestScoreTotal!=4500||profile.AlphabetUnlocked!=45||!profile.StorageHealthy)throw new Exception("Legacy save migration failed: "+count);
   string roundTrip=(string)typeof(LocalProfile).GetMethod("Serialize",flags).Invoke(profile,null);
   typeof(LocalProfile).GetMethod("Restore",flags).Invoke(new LocalProfile(),new object[]{roundTrip});checks+=5;
  }
  foreach(string name in new[]{"Picnic","PicnicJellyV62","SnowGardenV68","YarnObstacleV2","ShopV30"})
   if(Directory.Exists("Assets/Resources/Art/"+name))throw new Exception("Retired resource remains: "+name);
  OnboardingBuild.Validate();VictoryFinaleBuild.Validate();
  Debug.Log("PACKAGE_CLEANUP_VALIDATED checks="+checks);
 }
 public static void Windows(){
  Validate();BuildEntry.Configure();PlayerSettings.SetScriptingBackend(NamedBuildTarget.Standalone,ScriptingImplementation.Mono2x);
  string path=Path.GetFullPath("../../Builds/Cleanup-v87/Windows/BounceGarden.exe");Directory.CreateDirectory(Path.GetDirectoryName(path));
  var result=BuildPipeline.BuildPlayer(new BuildPlayerOptions{scenes=new[]{"Assets/Scenes/Main.unity"},locationPathName=path,target=BuildTarget.StandaloneWindows64,options=BuildOptions.None});
  Debug.Log("PACKAGE_CLEANUP_BUILD "+result.summary.result+" errors="+result.summary.totalErrors+" bytes="+result.summary.totalSize);
  if(result.summary.result!=BuildResult.Succeeded)throw new Exception("Cleanup build failed");
 }
}
