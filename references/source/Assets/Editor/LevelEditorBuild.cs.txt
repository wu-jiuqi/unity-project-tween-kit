using System;
using System.Collections.Generic;
using System.IO;
using System.Linq;
using System.Security.Cryptography;
using UnityEditor;
using UnityEditor.Build;
using UnityEditor.Build.Reporting;
using UnityEditor.SceneManagement;
using UnityEngine;

/// <summary>Dedicated desktop tool build. Never invokes the game's release configurator.</summary>
public static class LevelEditorBuild {
    [Serializable] sealed class FileStamp { public string path, sha256; public long bytes; }
    [Serializable] sealed class Report {
        public string status, unityVersion, executable, builtAtUtc, productName, dataLayout,balancePolicyVersion;
        public int errors, warnings, payloadFiles;
        public long payloadBytes;
        public bool settingsRestored, sceneSetupRestored, temporarySceneRemoved;
        public string[] defines, cleanupErrors;
        public FileStamp[] campaigns, binaries;
    }
    sealed class SettingsSnapshot {
        public string company = PlayerSettings.companyName, product = PlayerSettings.productName;
        public int width = PlayerSettings.defaultScreenWidth, height = PlayerSettings.defaultScreenHeight;
        public bool resizable = PlayerSettings.resizableWindow, background = PlayerSettings.runInBackground;
        public FullScreenMode mode = PlayerSettings.fullScreenMode;
        public UIOrientation orientation = PlayerSettings.defaultInterfaceOrientation;
        public ScriptingImplementation backend = PlayerSettings.GetScriptingBackend(NamedBuildTarget.Standalone);
        public void Restore() {
            PlayerSettings.companyName = company; PlayerSettings.productName = product;
            PlayerSettings.defaultScreenWidth = width; PlayerSettings.defaultScreenHeight = height;
            PlayerSettings.resizableWindow = resizable; PlayerSettings.runInBackground = background;
            PlayerSettings.fullScreenMode = mode; PlayerSettings.defaultInterfaceOrientation = orientation;
            PlayerSettings.SetScriptingBackend(NamedBuildTarget.Standalone, backend);
        }
        public bool Matches() => company == PlayerSettings.companyName && product == PlayerSettings.productName
            && width == PlayerSettings.defaultScreenWidth && height == PlayerSettings.defaultScreenHeight
            && resizable == PlayerSettings.resizableWindow && background == PlayerSettings.runInBackground
            && mode == PlayerSettings.fullScreenMode && orientation == PlayerSettings.defaultInterfaceOrientation
            && backend == PlayerSettings.GetScriptingBackend(NamedBuildTarget.Standalone);
    }
    static string Workspace() {
        var directory = new DirectoryInfo(Application.dataPath);
        while (directory != null && !File.Exists(Path.Combine(directory.FullName, "version.json"))) directory = directory.Parent;
        if (directory == null) throw new InvalidOperationException("Cannot find the BounceBusters workspace.");
        return directory.FullName;
    }
    static FileStamp Stamp(string path) {
        using (var hash = SHA256.Create()) using (var input = File.OpenRead(path))
            return new FileStamp { path = Path.GetFullPath(path), bytes = input.Length,
                sha256 = BitConverter.ToString(hash.ComputeHash(input)).Replace("-", "").ToLowerInvariant() };
    }
    public static void Windows() {
        if (!Application.unityVersion.StartsWith("6000.0.26f1", StringComparison.Ordinal))
            throw new InvalidOperationException("Level editor expects Unity 6000.0.26f1; got " + Application.unityVersion);
        string root = Workspace();
        string executable = Environment.GetEnvironmentVariable("BOUNCE_LEVEL_EDITOR_OUTPUT");
        if (string.IsNullOrWhiteSpace(executable)) executable = Path.Combine(root, "Builds/LevelEditor/Windows/BounceBustersLevelEditor.exe");
        executable = Path.GetFullPath(executable);
        if (!string.Equals(Path.GetExtension(executable), ".exe", StringComparison.OrdinalIgnoreCase))
            throw new InvalidOperationException("BOUNCE_LEVEL_EDITOR_OUTPUT must name the executable.");
        string folder = Path.GetDirectoryName(executable);
        Directory.CreateDirectory(folder);
        var sourceFiles = new List<FileStamp>();
        foreach (string chapter in new[] { "Opening" }) {
            string assetPath = "Assets/Resources/" + chapter + "/levels.json";
            if (AssetDatabase.LoadAssetAtPath<TextAsset>(assetPath) == null)
                throw new InvalidOperationException("Missing bundled campaign: " + assetPath);
            sourceFiles.Add(Stamp(assetPath));
        }
        var original = new SettingsSnapshot();
        var sceneSetup = EditorSceneManager.GetSceneManagerSetup();
        var report = new Report { status = "failed", unityVersion = Application.unityVersion,
            executable = executable, builtAtUtc = DateTime.UtcNow.ToString("o"),
            productName = "Bounce Busters Level Editor",balancePolicyVersion=BounceGarden.LevelEditing.LevelBalanceRunner.PolicyVersion, campaigns = sourceFiles.ToArray(),
            defines = new[] { "BOUNCE_LEVEL_EDITOR", "BOUNCE_TEST_TOOLS" } };
        string tempFolder = "Assets/__LevelEditorBuild_" + Guid.NewGuid().ToString("N");
        try {
            Directory.CreateDirectory(tempFolder); AssetDatabase.Refresh();
            var scene = EditorSceneManager.NewScene(NewSceneSetup.EmptyScene, NewSceneMode.Single);
            string scenePath = tempFolder + "/LevelEditorBootstrap.unity";
            if (!EditorSceneManager.SaveScene(scene, scenePath)) throw new IOException("Cannot save temporary tool scene.");
            PlayerSettings.companyName = "BounceBustersTools";
            PlayerSettings.productName = report.productName;
            PlayerSettings.defaultScreenWidth = 1440; PlayerSettings.defaultScreenHeight = 900;
            PlayerSettings.resizableWindow = true; PlayerSettings.fullScreenMode = FullScreenMode.Windowed;
            PlayerSettings.runInBackground = true; PlayerSettings.defaultInterfaceOrientation = UIOrientation.LandscapeLeft;
            PlayerSettings.SetScriptingBackend(NamedBuildTarget.Standalone, ScriptingImplementation.Mono2x);
            var result = BuildPipeline.BuildPlayer(new BuildPlayerOptions {
                scenes = new[] { scenePath }, target = BuildTarget.StandaloneWindows64,
                locationPathName = executable, options = BuildOptions.CompressWithLz4HC,
                extraScriptingDefines = report.defines
            });
            report.errors = (int)result.summary.totalErrors; report.warnings = (int)result.summary.totalWarnings;
            if (result.summary.result != BuildResult.Succeeded) throw new InvalidOperationException("Level editor build failed: " + result.summary.result);
            string data = Path.Combine(folder, Path.GetFileNameWithoutExtension(executable) + "_Data");
            var binaries = new List<string> { executable, Path.Combine(folder, "UnityPlayer.dll"),
                Path.Combine(data, "app.info"), Path.Combine(data, "Managed/Assembly-CSharp.dll") };
            string archive = Path.Combine(data, "data.unity3d");
            if (File.Exists(archive)) {
                // CompressWithLz4HC packages the serialized scenes and resources
                // into this archive instead of loose globalgamemanagers/assets.
                binaries.Add(archive); report.dataLayout = "compressed-data.unity3d";
            } else {
                binaries.Add(Path.Combine(data, "globalgamemanagers"));
                binaries.Add(Path.Combine(data, "resources.assets"));
                report.dataLayout = "uncompressed-assets";
            }
            foreach (string file in binaries)
                if (!File.Exists(file) || new FileInfo(file).Length == 0) throw new IOException("Incomplete desktop tool payload: " + file);
            var identity = File.ReadAllLines(Path.Combine(data, "app.info"));
            if (identity.Length < 2 || identity[0] != "BounceBustersTools" || identity[1] != report.productName)
                throw new IOException("Desktop tool payload has unexpected company/product identity.");
            report.binaries = binaries.Select(Stamp).ToArray();
            var payload = Directory.GetFiles(folder, "*", SearchOption.AllDirectories);
            report.payloadFiles = payload.Length; report.payloadBytes = payload.Sum(p => new FileInfo(p).Length);
            foreach (var source in report.campaigns)
                if (Stamp(source.path).sha256 != source.sha256) throw new InvalidOperationException("Build modified campaign source: " + source.path);
            report.status = "built";
        } finally {
            var cleanupErrors = new List<string>();
            try { original.Restore(); }
            catch (Exception error) { cleanupErrors.Add("PlayerSettings restore: " + error.Message); }
            try {
                // A fresh -batchmode process has no loaded scene to restore.
                // Close the build scene before deleting its temporary asset folder.
                if (sceneSetup != null && sceneSetup.Any(s => s.isLoaded)) EditorSceneManager.RestoreSceneManagerSetup(sceneSetup);
                else EditorSceneManager.NewScene(NewSceneSetup.EmptyScene, NewSceneMode.Single);
                report.sceneSetupRestored = true;
            } catch (Exception error) {
                cleanupErrors.Add("Scene setup restore: " + error.Message);
                try { EditorSceneManager.NewScene(NewSceneSetup.EmptyScene, NewSceneMode.Single); }
                catch (Exception fallback) { cleanupErrors.Add("Close temporary scene: " + fallback.Message); }
            } finally {
                try {
                    if (AssetDatabase.IsValidFolder(tempFolder)) AssetDatabase.DeleteAsset(tempFolder);
                } catch (Exception error) { cleanupErrors.Add("Temporary scene removal: " + error.Message); }
                finally {
                    try { AssetDatabase.SaveAssets(); }
                    catch (Exception error) { cleanupErrors.Add("Save restored settings: " + error.Message); }
                }
            }
            try { report.settingsRestored = original.Matches(); }
            catch (Exception error) { cleanupErrors.Add("Verify settings restore: " + error.Message); }
            report.temporarySceneRemoved = !Directory.Exists(tempFolder);
            report.cleanupErrors = cleanupErrors.ToArray();
            File.WriteAllText(Path.Combine(folder, "build-summary.json"), JsonUtility.ToJson(report, true));
            Debug.Log("LEVEL_EDITOR_BUILD " + report.status + " errors=" + report.errors + " warnings=" + report.warnings
                + " settingsRestored=" + report.settingsRestored + " temporarySceneRemoved=" + report.temporarySceneRemoved
                + " cleanupErrors=" + string.Join(" | ", report.cleanupErrors));
        }
        if (!report.settingsRestored || !report.sceneSetupRestored || !report.temporarySceneRemoved || report.cleanupErrors.Length > 0)
            throw new InvalidOperationException("Tool build cleanup did not complete: " + string.Join(" | ", report.cleanupErrors));
        Debug.Log("LEVEL_EDITOR_BUILD_OK " + executable);
    }
}
