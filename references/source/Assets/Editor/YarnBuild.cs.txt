using System;
using System.IO;
using UnityEditor;
using UnityEditor.Build.Reporting;
using UnityEngine;

public sealed class YarnTextureImporter:AssetPostprocessor {
 void OnPreprocessTexture(){
  if(!assetPath.StartsWith("Assets/Resources/Art/YarnV19/",StringComparison.Ordinal))return;
  var t=(TextureImporter)assetImporter;t.textureType=TextureImporterType.Default;
  t.alphaSource=TextureImporterAlphaSource.FromInput;t.alphaIsTransparency=true;
  t.mipmapEnabled=false;t.isReadable=false;t.sRGBTexture=true;t.npotScale=TextureImporterNPOTScale.None;
  t.wrapMode=TextureWrapMode.Clamp;t.filterMode=FilterMode.Bilinear;t.maxTextureSize=assetPath.Contains("atlas")?2048:256;
  t.textureCompression=TextureImporterCompression.Uncompressed;
  t.SetPlatformTextureSettings(new TextureImporterPlatformSettings{name="Android",overridden=true,maxTextureSize=t.maxTextureSize,format=TextureImporterFormat.ETC2_RGBA8,compressionQuality=100});
 }
}
public static class YarnBuild {
 public static void Windows(){Build(BuildTarget.StandaloneWindows64);}
 public static void Android(){Build(BuildTarget.Android);}
 public static void Import(){
  foreach(var p in Directory.GetFiles("Assets/Resources/Art/YarnV19","*.png"))AssetDatabase.ImportAsset(p.Replace('\\','/'),ImportAssetOptions.ForceUpdate|ImportAssetOptions.ForceSynchronousImport);
 }
 static void Build(BuildTarget target){
  Import();BuildEntry.Configure();BuildEntry.Rules();Debug.Log("PROFILE_TESTS "+BounceGarden.LocalProfile.RunSelfTests());
  PlayerSettings.bundleVersion = BounceGarden.GameVersion.Name;PlayerSettings.Android.bundleVersionCode = BounceGarden.GameVersion.Code;
  PlayerSettings.SetScriptingBackend(UnityEditor.Build.NamedBuildTarget.Standalone,ScriptingImplementation.Mono2x);
  string path=Environment.GetEnvironmentVariable("BOUNCE_YARN_OUTPUT");if(string.IsNullOrEmpty(path))throw new Exception("BOUNCE_YARN_OUTPUT required");
  Directory.CreateDirectory(Path.GetDirectoryName(path));
  var report=BuildPipeline.BuildPlayer(new BuildPlayerOptions{scenes=new[]{"Assets/Scenes/Main.unity"},target=target,locationPathName=path,options=BuildOptions.CompressWithLz4HC,extraScriptingDefines=new[]{"BOUNCE_TEST_TOOLS"}});
  string result="FLIGHT_V22_BUILD "+report.summary.result+" errors="+report.summary.totalErrors+" warnings="+report.summary.totalWarnings;
  File.WriteAllText(path+".build.txt",result);Debug.Log(result);if(report.summary.result!=BuildResult.Succeeded)throw new Exception(result);
 }
}
